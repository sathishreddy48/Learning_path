Graphs, Union-Find & Shortest Paths

Most graph problems are not labelled as graph problems. A grid is a graph, a set of prerequisites is a graph, a word ladder is a graph, and a state machine is a graph — so the skill is recognising that nodes and edges are hiding in the problem, then picking the right traversal.

Representations: adjacency list, matrix and implicit graphs

Three ways to hold a graph, and the choice follows from density. An adjacency list is O(V + E) space and iterating a node's neighbours is O(degree) — the default. An adjacency matrix is O(V²) space but answers "is there an edge?" in O(1), which is worth it only for dense graphs or Floyd-Warshall. An implicit graph has no data structure at all: the neighbours are computed from the current state, which is how grids, word ladders and puzzle states work.

Recognising the implicit case is the valuable part — "minimum number of moves" over any kind of state is BFS over a graph you never build.

static readonly (int R, int C)[] Dirs = [(0,1), (1,0), (0,-1), (-1,0)];

static IEnumerable<(int R, int C)> Neighbours((int R, int C) cell, char[][] grid)
{
    foreach (var (dr, dc) in Dirs)
    {
        int r = cell.R + dr, c = cell.C + dc;
        if (r >= 0 && r < grid.Length && c >= 0 && c < grid[0].Length && grid[r][c] != '#')
            yield return (r, c);
    }
}

// A WORD LADDER is a graph where neighbours are "words one letter away"
static IEnumerable<string> WordNeighbours(string word, HashSet<string> dictionary)
{
    var chars = word.ToCharArray();
    for (int i = 0; i < chars.Length; i++)
    {
        char original = chars[i];
        for (char c = 'a'; c <= 'z'; c++)
        {
            if (c == original) continue;
            chars[i] = c;
            var candidate = new string(chars);
            if (dictionary.Contains(candidate)) yield return candidate;
        }
        chars[i] = original;
    }
}

BFS and DFS on graphs and grids

Represent graphs as adjacency lists (Dictionary<int, List<int>>) or treat grids as implicit graphs. BFS (queue) for shortest path in unweighted graphs and level order; DFS (recursion or explicit stack) for connectivity, cycles and topological sort. Canonical: number of islands, clone graph, course schedule.

// BFS shortest path (unweighted)
static int ShortestPath(Dictionary<int, List<int>> g, int src, int dst)
{
    var dist = new Dictionary<int, int> { [src] = 0 };
    var q = new Queue<int>(); q.Enqueue(src);
    while (q.Count > 0)
    {
        var u = q.Dequeue();
        if (u == dst) return dist[u];
        foreach (var v in g.GetValueOrDefault(u, new()))
            if (dist.TryAdd(v, dist[u] + 1)) q.Enqueue(v);
    }
    return -1;
}

// DFS on a grid: number of islands
static int NumIslands(char[][] grid)
{
    int rows = grid.Length, cols = grid[0].Length, count = 0;
    void Sink(int r, int c)
    {
        if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] != '1') return;
        grid[r][c] = '0';
        Sink(r + 1, c); Sink(r - 1, c); Sink(r, c + 1); Sink(r, c - 1);
    }
    for (int r = 0; r < rows; r++) for (int c = 0; c < cols; c++) if (grid[r][c] == '1') { count++; Sink(r, c); }
    return count;
}

// Topological sort (Kahn) — course schedule
static bool CanFinish(int n, int[][] prereqs)
{
    var adj = new List<int>[n]; var indeg = new int[n];
    for (int i = 0; i < n; i++) adj[i] = new();
    foreach (var p in prereqs) { adj[p[1]].Add(p[0]); indeg[p[0]]++; }
    var q = new Queue<int>(Enumerable.Range(0, n).Where(i => indeg[i] == 0));
    int seen = 0;
    while (q.Count > 0) { var u = q.Dequeue(); seen++; foreach (var v in adj[u]) if (--indeg[v] == 0) q.Enqueue(v); }
    return seen == n;                       // false = cycle
}

// Dijkstra with PriorityQueue (.NET 6+)
static int[] Dijkstra(List<(int to, int w)>[] g, int src)
{
    var dist = Enumerable.Repeat(int.MaxValue, g.Length).ToArray(); dist[src] = 0;
    var pq = new PriorityQueue<int, int>(); pq.Enqueue(src, 0);
    while (pq.TryDequeue(out var u, out var d))
    {
        if (d > dist[u]) continue;
        foreach (var (v, w) in g[u]) if (d + w < dist[v]) { dist[v] = d + w; pq.Enqueue(v, dist[v]); }
    }
    return dist;
}
from collections import deque, defaultdict
import heapq

def shortest_path(g: dict[int, list[int]], src: int, dst: int) -> int:
    dist = {src: 0}; q = deque([src])
    while q:
        u = q.popleft()
        if u == dst: return dist[u]
        for v in g.get(u, []):
            if v not in dist:
                dist[v] = dist[u] + 1; q.append(v)
    return -1

def num_islands(grid: list[list[str]]) -> int:
    rows, cols = len(grid), len(grid[0]); count = 0
    def sink(r, c):
        if r < 0 or c < 0 or r >= rows or c >= cols or grid[r][c] != "1": return
        grid[r][c] = "0"
        for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)): sink(r + dr, c + dc)
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1": count += 1; sink(r, c)
    return count

def can_finish(n: int, prereqs: list[list[int]]) -> bool:
    adj = defaultdict(list); indeg = [0] * n
    for a, b in prereqs: adj[b].append(a); indeg[a] += 1
    q = deque(i for i in range(n) if indeg[i] == 0); seen = 0
    while q:
        u = q.popleft(); seen += 1
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0: q.append(v)
    return seen == n

def dijkstra(g: list[list[tuple[int, int]]], src: int) -> list[int]:
    dist = [float("inf")] * len(g); dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue
        for v, w in g[u]:
            if d + w < dist[v]:
                dist[v] = d + w; heapq.heappush(pq, (dist[v], v))
    return dist
Interview angle

Know which one each question wants. BFS for shortest path in an unweighted graph and for anything level-by-level; DFS for reachability, components, cycle detection and anything needing the full path. The two bugs interviewers watch for: marking visited at dequeue rather than at enqueue, which lets the same node enter the queue many times, and recursing without a visited set so a cycle loops forever.

Connected components, flood fill and bipartite checking

Three variations on "traverse from every unvisited node". Counting components is a loop over all nodes, starting a traversal from each unvisited one and counting the starts. Flood fill and "number of islands" are the same thing on a grid. Bipartite checking adds a two-colouring: colour each node the opposite of its parent, and the graph is bipartite exactly when no edge joins two nodes of the same colour.

// COUNT CONNECTED COMPONENTS
static int CountComponents(int n, int[][] edges)
{
    var adj = Build(n, edges);
    var seen = new bool[n];
    int components = 0;

    for (int start = 0; start < n; start++)
    {
        if (seen[start]) continue;
        components++;                            // a new component begins here

        var stack = new Stack<int>();
        stack.Push(start);
        seen[start] = true;
        while (stack.Count > 0)
            foreach (var next in adj[stack.Pop()])
                if (!seen[next]) { seen[next] = true; stack.Push(next); }
    }
    return components;
}

// NUMBER OF ISLANDS - the same loop on an implicit grid, sinking as it goes
static int NumIslands(char[][] grid)
{
    int rows = grid.Length, cols = grid[0].Length, count = 0;

    void Sink(int r, int c)
    {
        if (r < 0 || r >= rows || c < 0 || c >= cols || grid[r][c] != '1') return;
        grid[r][c] = '0';                        // the grid IS the visited set
        Sink(r + 1, c); Sink(r - 1, c); Sink(r, c + 1); Sink(r, c - 1);
    }

    for (int r = 0; r < rows; r++)
        for (int c = 0; c < cols; c++)
            if (grid[r][c] == '1') { count++; Sink(r, c); }
    return count;
}
// Mutating the input is O(1) space but destructive - SAY that you are doing it
// and offer a separate visited array if the caller needs the grid intact.
// On a 1000x1000 all-land grid the recursion is 10^6 deep: use an explicit

// BIPARTITE - two-colour it; an edge between same-coloured nodes disproves it
static bool IsBipartite(int[][] graph)
{
    var colour = new int[graph.Length];          // 0 = uncoloured, 1 and -1
    for (int start = 0; start < graph.Length; start++)
    {
        if (colour[start] != 0) continue;        // every COMPONENT must be checked

        var queue = new Queue<int>();
        queue.Enqueue(start);
        colour[start] = 1;

        while (queue.Count > 0)
        {
            int node = queue.Dequeue();
            foreach (var next in graph[node])
            {
                if (colour[next] == colour[node]) return false;   // odd cycle
                if (colour[next] != 0) continue;
                colour[next] = -colour[node];
                queue.Enqueue(next);
            }
        }
    }
    return true;
}
// A graph is bipartite iff it contains no odd-length cycle. Worth saying -
// it is the mathematical fact the algorithm is detecting.

// CYCLE DETECTION, UNDIRECTED: a cycle exists if you reach a visited node that
// is not the one you came from
static bool HasCycleUndirected(int n, List<int>[] adj)
{
    var seen = new bool[n];
    bool Dfs(int node, int parent)
    {
        seen[node] = true;
        foreach (var next in adj[node])
        {
            if (!seen[next]) { if (Dfs(next, node)) return true; }
            else if (next != parent) return true;          // the `parent` check
        }
        return false;
    }
    for (int i = 0; i < n; i++) if (!seen[i] && Dfs(i, -1)) return true;
    return false;
}

// CYCLE DETECTION, DIRECTED: three colours. A back-edge to a node still ON THE
// STACK is a cycle; an edge to a finished node is not.
static bool HasCycleDirected(int n, List<int>[] adj)
{
    var state = new int[n];                      // 0 unvisited, 1 in progress, 2 done
    bool Dfs(int node)
    {
        state[node] = 1;
        foreach (var next in adj[node])
        {
            if (state[next] == 1) return true;   // back-edge: a cycle
            if (state[next] == 0 && Dfs(next)) return true;
        }
        state[node] = 2;
        return false;
    }
    for (int i = 0; i < n; i++) if (state[i] == 0 && Dfs(i)) return true;
    return false;
}

from collections import deque

def count_components(n: int, edges: list[list[int]]) -> int:
    adj = build(n, edges)
    seen = [False] * n
    components = 0
    for start in range(n):
        if seen[start]:
            continue
        components += 1
        stack = [start]
        seen[start] = True
        while stack:
            for nxt in adj[stack.pop()]:
                if not seen[nxt]:
                    seen[nxt] = True
                    stack.append(nxt)
    return components

def num_islands(grid: list[list[str]]) -> int:
    rows, cols = len(grid), len(grid[0])
    count = 0

    def sink(r: int, c: int) -> None:
        stack = [(r, c)]                        # iterative: 10^6 depth is common
        while stack:
            r, c = stack.pop()
            if not (0 <= r < rows and 0 <= c < cols) or grid[r][c] != "1":
                continue
            grid[r][c] = "0"
            stack.extend(((r+1,c), (r-1,c), (r,c+1), (r,c-1)))

    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1":
                count += 1
                sink(r, c)
    return count

def is_bipartite(graph: list[list[int]]) -> bool:
    colour = [0] * len(graph)
    for start in range(len(graph)):
        if colour[start]:
            continue
        queue = deque([start])
        colour[start] = 1
        while queue:
            node = queue.popleft()
            for nxt in graph[node]:
                if colour[nxt] == colour[node]:
                    return False
                if not colour[nxt]:
                    colour[nxt] = -colour[node]
                    queue.append(nxt)
    return True

def has_cycle_undirected(n: int, adj: list[list[int]]) -> bool:
    seen = [False] * n
    def dfs(node: int, parent: int) -> bool:
        seen[node] = True
        for nxt in adj[node]:
            if not seen[nxt]:
                if dfs(nxt, node):
                    return True
            elif nxt != parent:
                return True
        return False
    return any(not seen[i] and dfs(i, -1) for i in range(n))

def has_cycle_directed(n: int, adj: list[list[int]]) -> bool:
    UNVISITED, IN_PROGRESS, DONE = 0, 1, 2
    state = [UNVISITED] * n
    def dfs(node: int) -> bool:
        state[node] = IN_PROGRESS
        for nxt in adj[node]:
            if state[nxt] == IN_PROGRESS:
                return True
            if state[nxt] == UNVISITED and dfs(nxt):
                return True
        state[node] = DONE
        return False
    return any(state[i] == UNVISITED and dfs(i) for i in range(n))
Interview angle

The detail that separates answers is that directed and undirected cycle detection are different algorithms: undirected needs the "ignore the node I came from" parent check, directed needs three states because an edge to an already-finished node is not a cycle. Using the undirected version on a directed graph is a common and quietly wrong answer. For islands, say explicitly whether you are mutating the input, and note that a large grid needs an iterative traversal because the recursion can go a million frames deep.

Topological sort: Kahn and DFS

A linear ordering of a directed acyclic graph where every edge points forwards. The giveaway wording is "prerequisites", "dependencies", "build order", "course schedule" — and the ordering exists if and only if there is no cycle, so topological sort doubles as cycle detection on a directed graph.

Two implementations. Kahn's is BFS on in-degrees: repeatedly take a node with in-degree zero and remove it; if you finish with fewer than V nodes, there was a cycle. DFS pushes each node onto a stack after its descendants, and the reversed finish order is the answer. Kahn's is easier to get right and tells you about the cycle for free, so prefer it.

// KAHN - BFS on in-degrees. The version to default to.
static int[] TopoSort(int n, int[][] edges)      // edges: [from, to]
{
    var adj = new List<int>[n];
    for (int i = 0; i < n; i++) adj[i] = [];
    var inDegree = new int[n];

    foreach (var e in edges) { adj[e[0]].Add(e[1]); inDegree[e[1]]++; }

    var queue = new Queue<int>();
    for (int i = 0; i < n; i++) if (inDegree[i] == 0) queue.Enqueue(i);

    var order = new List<int>(n);
    while (queue.Count > 0)
    {
        int node = queue.Dequeue();
        order.Add(node);
        foreach (var next in adj[node])
            if (--inDegree[next] == 0) queue.Enqueue(next);   // now unblocked
    }

    // fewer than n nodes emitted => something never reached in-degree 0 => a cycle
    return order.Count == n ? order.ToArray() : [];
}
// O(V + E). The cycle check is free, which is why "can you finish all courses?"
// is just `TopoSort(...).Length == n`.

// COURSE SCHEDULE - the canonical phrasing
static bool CanFinish(int numCourses, int[][] prerequisites)
    => TopoSort(numCourses, prerequisites.Select(p => new[] { p[1], p[0] }).ToArray())
       .Length == numCourses;
// Note the swap: [course, prerequisite] means the EDGE goes prerequisite -> course.
// Getting this direction backwards is the most common bug in the problem.

// DFS VERSION - reverse finish order
static int[] TopoSortDfs(int n, List<int>[] adj)
{
    var state = new int[n];                      // 0 unvisited, 1 in progress, 2 done
    var order = new Stack<int>();

    bool Dfs(int node)
    {
        state[node] = 1;
        foreach (var next in adj[node])
        {
            if (state[next] == 1) return false;              // cycle
            if (state[next] == 0 && !Dfs(next)) return false;
        }
        state[node] = 2;
        order.Push(node);                        // pushed AFTER its descendants
        return true;
    }

    for (int i = 0; i < n; i++)
        if (state[i] == 0 && !Dfs(i)) return [];
    return order.ToArray();                      // Stack enumerates top-first
}

// ALIEN DICTIONARY - the hard, satisfying application. Derive edges by comparing
// adjacent words at their first differing character, then topologically sort.
static string AlienOrder(string[] words)
{
    var adj = new Dictionary<char, HashSet<char>>();
    var inDegree = new Dictionary<char, int>();
    foreach (var w in words) foreach (var c in w)
    { adj.TryAdd(c, []); inDegree.TryAdd(c, 0); }

    for (int i = 0; i + 1 < words.Length; i++)
    {
        string a = words[i], b = words[i + 1];
        if (a.Length > b.Length && a.StartsWith(b)) return "";   // invalid input
        for (int j = 0; j < Math.Min(a.Length, b.Length); j++)
        {
            if (a[j] == b[j]) continue;
            if (adj[a[j]].Add(b[j])) inDegree[b[j]]++;           // first difference
            break;                                                // ONLY the first
        }
    }

    var queue = new Queue<char>(inDegree.Where(kv => kv.Value == 0).Select(kv => kv.Key));
    var sb = new StringBuilder();
    while (queue.Count > 0)
    {
        char c = queue.Dequeue();
        sb.Append(c);
        foreach (var next in adj[c]) if (--inDegree[next] == 0) queue.Enqueue(next);
    }
    return sb.Length == inDegree.Count ? sb.ToString() : "";     // cycle => no order
}
from collections import deque, defaultdict
import heapq

def topo_sort(n: int, edges: list[list[int]]) -> list[int]:
    adj: list[list[int]] = [[] for _ in range(n)]
    in_degree = [0] * n
    for u, v in edges:
        adj[u].append(v)
        in_degree[v] += 1

    queue = deque(i for i in range(n) if in_degree[i] == 0)
    order: list[int] = []
    while queue:
        node = queue.popleft()
        order.append(node)
        for nxt in adj[node]:
            in_degree[nxt] -= 1
            if in_degree[nxt] == 0:
                queue.append(nxt)

    return order if len(order) == n else []      # short => a cycle

def can_finish(num_courses: int, prerequisites: list[list[int]]) -> bool:
    # [course, prereq] means the edge is prereq -> course
    edges = [[prereq, course] for course, prereq in prerequisites]
    return len(topo_sort(num_courses, edges)) == num_courses

            if state[nxt] == UNVISITED and not dfs(nxt):
                return False
        state[node] = DONE
        order.append(node)                       # appended AFTER its descendants
        return True

    for i in range(n):
        if state[i] == UNVISITED and not dfs(i):
            return []
    return order[::-1]                           # reversed finish order

# Lexicographically smallest: a heap instead of a deque
def topo_sort_smallest(n: int, edges: list[list[int]]) -> list[int]:
    adj: list[list[int]] = [[] for _ in range(n)]
    in_degree = [0] * n
    for u, v in edges:
        adj[u].append(v); in_degree[v] += 1

    ready = [i for i in range(n) if in_degree[i] == 0]
    heapq.heapify(ready)
    order = []
    while ready:
        node = heapq.heappop(ready)
        order.append(node)
        for nxt in adj[node]:
            in_degree[nxt] -= 1
            if in_degree[nxt] == 0:
                heapq.heappush(ready, nxt)
    return order if len(order) == n else []

Union-Find (disjoint set union)

A structure for "are these two things in the same group?" and "merge these two groups", both in effectively O(1). It is the right answer whenever connectivity is being built up incrementally — where a DFS would have to re-run after every added edge.

Two optimisations make it fast and you should include both: path compression (point every node directly at the root during Find) and union by rank or size (attach the smaller tree under the larger). Together they give O(α(n)) amortised, where α is the inverse Ackermann function — under 5 for any input that exists.

public class UnionFind
{
    private readonly int[] _parent;
    private readonly int[] _size;
    public int Components { get; private set; }

    public UnionFind(int n)
    {
        _parent = new int[n];
        _size = new int[n];
        for (int i = 0; i < n; i++) { _parent[i] = i; _size[i] = 1; }
        Components = n;
    }

    // PATH COMPRESSION - every node on the way up is repointed at the root
    public int Find(int x)
    {
        while (_parent[x] != x)
        {
            _parent[x] = _parent[_parent[x]];    // path HALVING: one line, iterative,
            x = _parent[x];                      // no recursion to blow the stack
        }
        return x;
    }

    // UNION BY SIZE - the smaller tree hangs off the larger
    public bool Union(int a, int b)
    {
        int rootA = Find(a), rootB = Find(b);
        if (rootA == rootB) return false;        // already together

        if (_size[rootA] < _size[rootB]) (rootA, rootB) = (rootB, rootA);
        _parent[rootB] = rootA;
        _size[rootA] += _size[rootB];
        Components--;
        return true;                             // true == "this edge merged two groups"
    }

    public bool Connected(int a, int b) => Find(a) == Find(b);
    public int  SizeOf(int x) => _size[Find(x)];
}
// O(alpha(n)) amortised per operation - effectively constant.
// Union returning a bool is what makes Kruskal and cycle detection one-liners.

// WHEN UNION-FIND BEATS DFS: edges arrive one at a time and you must answer
// connectivity after each. DFS is O(V+E) per query; union-find is ~O(1).
// WHEN IT DOES NOT: union-find cannot SPLIT a set, and it cannot give you the
// path between two nodes - only whether one exists.
class UnionFind:
    __slots__ = ("parent", "size", "components")

    def __init__(self, n: int) -> None:
        self.parent = list(range(n))
        self.size = [1] * n
        self.components = n

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> bool:
        root_a, root_b = self.find(a), self.find(b)
        if root_a == root_b:
            return False
        if self.size[root_a] < self.size[root_b]:
            root_a, root_b = root_b, root_a
        self.parent[root_b] = root_a
        self.size[root_a] += self.size[root_b]
        self.components -= 1
        return True

    def connected(self, a: int, b: int) -> bool:
        return self.find(a) == self.find(b)

    def size_of(self, x: int) -> int:
        return self.size[self.find(x)]


def count_components(n: int, edges: list[list[int]]) -> int:
    uf = UnionFind(n)
    for u, v in edges:
        uf.union(u, v)
    return uf.components

def has_cycle(n: int, edges: list[list[int]]) -> bool:
    uf = UnionFind(n)
    return any(not uf.union(u, v) for u, v in edges)

def valid_tree(n: int, edges: list[list[int]]) -> bool:
    if len(edges) != n - 1:
        return False
    uf = UnionFind(n)
    return all(uf.union(u, v) for u, v in edges)

# For non-integer items, map to indices first - or use a dict-backed version
class UnionFindDict:
    def __init__(self) -> None:
        self.parent: dict = {}

    def find(self, x):
        self.parent.setdefault(x, x)
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        self.parent[rb] = ra
        return True

# ACCOUNTS MERGE - union emails that share an account, then group by root
from collections import defaultdict
def accounts_merge(accounts: list[list[str]]) -> list[list[str]]:
    uf = UnionFindDict()
    owner: dict[str, str] = {}
    for name, *emails in accounts:
        for email in emails:
            owner[email] = name
            uf.union(emails[0], email)

    groups: dict[str, list[str]] = defaultdict(list)
    for email in owner:
        groups[uf.find(email)].append(email)

    return [[owner[root]] + sorted(emails) for root, emails in groups.items()]

Dijkstra and 0-1 BFS

Shortest path with non-negative weights. It is BFS with a priority queue instead of a plain queue: always expand the closest unfinalised node, and because weights are non-negative, the first time you pop a node you have its final distance.

That non-negativity condition is the question. With a negative edge the greedy finalisation is invalid — a longer-looking path could later be reduced — and you need Bellman-Ford instead.

The implementation detail: .NET's PriorityQueue has no decrease-key, so use lazy deletion — push the improved distance as a new entry and skip stale pops. And if every weight is 0 or 1, skip the heap entirely: a deque with zero-weight edges pushed to the front gives the same answer in O(V + E).

// DIJKSTRA with lazy deletion
static int[] Dijkstra(int n, List<(int To, int Weight)>[] adj, int source)
{
    var dist = new int[n];
    Array.Fill(dist, int.MaxValue);
    dist[source] = 0;

    var pq = new PriorityQueue<int, int>();      // (node, distance)
    pq.Enqueue(source, 0);

    while (pq.TryDequeue(out int node, out int d))
    {
        if (d > dist[node]) continue;            // LAZY DELETION: a stale entry

        foreach (var (next, weight) in adj[node])
        {
            int candidate = d + weight;
            if (candidate >= dist[next]) continue;
            dist[next] = candidate;
            pq.Enqueue(next, candidate);         // push a NEW entry, do not update
        }
    }
    return dist;
}
// O((V + E) log V). The `if (d > dist[node]) continue;` line is what makes the
// lazy approach correct - without it you reprocess stale entries and the
// complexity degrades. It is the line interviewers look for.

// RECONSTRUCTING THE PATH - keep a predecessor array
static List<int> ShortestPath(int n, List<(int, int)>[] adj, int source, int target)
{
    var dist = new int[n]; Array.Fill(dist, int.MaxValue); dist[source] = 0;
    var prev = new int[n]; Array.Fill(prev, -1);

    var pq = new PriorityQueue<int, int>();
    pq.Enqueue(source, 0);
    while (pq.TryDequeue(out int node, out int d))
    {
        if (d > dist[node]) continue;
        if (node == target) break;               // early exit: we are finalised
        foreach (var (next, w) in adj[node])
            if (d + w < dist[next]) { dist[next] = d + w; prev[next] = node; pq.Enqueue(next, d + w); }
    }

    if (dist[target] == int.MaxValue) return [];          // unreachable
    var path = new List<int>();
    for (int at = target; at != -1; at = prev[at]) path.Add(at);
    path.Reverse();
    return path;
}

// 0-1 BFS - every weight is 0 or 1, so a DEQUE replaces the heap: O(V + E)
static int[] ZeroOneBfs(int n, List<(int To, int Weight)>[] adj, int source)
{
    var dist = new int[n];
    Array.Fill(dist, int.MaxValue);
    dist[source] = 0;

    var deque = new LinkedList<int>();
    deque.AddFirst(source);

    while (deque.Count > 0)
    {
        int node = deque.First!.Value;
        deque.RemoveFirst();

        foreach (var (next, weight) in adj[node])
        {
            if (dist[node] + weight >= dist[next]) continue;
            dist[next] = dist[node] + weight;
            if (weight == 0) deque.AddFirst(next);        // free: same "layer"
            else             deque.AddLast(next);         // costs 1: next layer
        }
    }
    return dist;
}
// Classic use: "minimum obstacles to remove to reach the end" on a grid, where
// moving into an empty cell costs 0 and breaking a wall costs 1.

// VARIANTS the same loop solves, with the relaxation changed:
//   PATH WITH MAXIMUM PROBABILITY  - a MAX-heap, multiply instead of add
//   NETWORK DELAY TIME             - Dijkstra, answer = max over all dist
//   CHEAPEST FLIGHTS WITH K STOPS  - Dijkstra breaks (the stop limit is a second
//                                    dimension); use Bellman-Ford with k rounds,
//                                    or make the state (node, stopsUsed)
//   SWIM IN RISING WATER / PATH WITH MINIMUM EFFORT
//                                  - "minimise the maximum edge on the path":
//                                    relax with max(d, w) instead of d + w

// A* is Dijkstra with a heuristic added to the priority: priority = dist +
// estimatedRemaining. It is correct whenever the heuristic never overestimates.
import heapq
from collections import deque

def dijkstra(n: int, adj: list[list[tuple[int, int]]], source: int) -> list[float]:
    dist = [float("inf")] * n
    dist[source] = 0
    pq: list[tuple[int, int]] = [(0, source)]        # (distance, node)

    while pq:
        d, node = heapq.heappop(pq)
        if d > dist[node]:                           # stale entry
            continue
        for nxt, weight in adj[node]:
            candidate = d + weight
            if candidate < dist[nxt]:
                dist[nxt] = candidate
                heapq.heappush(pq, (candidate, nxt))
    return dist
# Distance FIRST in the tuple, so the heap orders by it.

def shortest_path(n, adj, source: int, target: int) -> list[int]:
    dist = [float("inf")] * n
    dist[source] = 0
    prev = [-1] * n
    pq = [(0, source)]

    while pq:
        d, node = heapq.heappop(pq)
        if d > dist[node]:
            continue
        if node == target:
            break
        for nxt, w in adj[node]:
            if d + w < dist[nxt]:
                dist[nxt] = d + w
                prev[nxt] = node
                heapq.heappush(pq, (d + w, nxt))

    if dist[target] == float("inf"):
        return []
    path, at = [], target
    while at != -1:
        path.append(at)
        at = prev[at]
    return path[::-1]

def zero_one_bfs(n: int, adj, source: int) -> list[float]:
    dist = [float("inf")] * n
    dist[source] = 0
    dq = deque([source])
    while dq:
        node = dq.popleft()
        for nxt, weight in adj[node]:
            if dist[node] + weight < dist[nxt]:
                dist[nxt] = dist[node] + weight
                if weight == 0: dq.appendleft(nxt)   # same layer
                else:           dq.append(nxt)       # next layer
    return dist

# "Minimise the maximum edge" - change only the relaxation
def minimum_effort(heights: list[list[int]]) -> int:
    rows, cols = len(heights), len(heights[0])
    effort = [[float("inf")] * cols for _ in range(rows)]
    effort[0][0] = 0
    pq = [(0, 0, 0)]
    while pq:
        e, r, c = heapq.heappop(pq)
        if (r, c) == (rows - 1, cols - 1):
            return e
        if e > effort[r][c]:
            continue
        for dr, dc in ((1,0), (-1,0), (0,1), (0,-1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols:
                step = max(e, abs(heights[nr][nc] - heights[r][c]))   # MAX, not +
                if step < effort[nr][nc]:
                    effort[nr][nc] = step
                    heapq.heappush(pq, (step, nr, nc))
    return 0
Interview angle

State the precondition before writing anything: Dijkstra requires non-negative weights, because it finalises a node the first time it is popped and a negative edge could later reduce an already-finalised distance. The implementation detail that gets looked for is lazy deletion — neither PriorityQueue nor heapq has decrease-key, so you push a new entry and skip stale pops with if (d > dist[node]) continue;. 0-1 BFS is a strong thing to offer unprompted when all weights are 0 or 1: a deque instead of a heap, O(V + E) instead of O(E log V).

Bellman-Ford, Floyd-Warshall and MSTs

The rest of the shortest-path toolkit, mostly needed so you can say why you are not using them. Bellman-Ford relaxes every edge V−1 times: O(V·E), slower than Dijkstra but it tolerates negative weights and detects negative cycles (if a V-th pass still improves something, one exists). Floyd-Warshall is all-pairs in O(V³) and is fifteen lines, which makes it the right answer for small dense graphs.

A minimum spanning tree answers a different question: connect every node at minimum total cost. Kruskal sorts the edges and adds any that does not create a cycle (union-find does the check); Prim grows one tree with a heap. Kruskal is easier to write if you already have union-find.

// BELLMAN-FORD - handles negative weights, detects negative cycles. O(V*E).
static (int[] Dist, bool HasNegativeCycle) BellmanFord(int n, int[][] edges, int source)
{
    var dist = new int[n];
    Array.Fill(dist, int.MaxValue);
    dist[source] = 0;

    for (int pass = 0; pass < n - 1; pass++)     // V-1 passes is always enough
    {
        bool changed = false;
        foreach (var e in edges)
        {
            if (dist[e[0]] == int.MaxValue) continue;          // unreachable
            if (dist[e[0]] + e[2] >= dist[e[1]]) continue;
            dist[e[1]] = dist[e[0]] + e[2];
            changed = true;
        }
        if (!changed) break;                     // early exit when it settles
    }

    // a V-th pass that still improves something proves a negative cycle
    foreach (var e in edges)
        if (dist[e[0]] != int.MaxValue && dist[e[0]] + e[2] < dist[e[1]])
            return (dist, true);

    return (dist, false);
}
// WHY V-1 PASSES: any shortest path has at most V-1 edges, and pass k
// guarantees every shortest path of k edges is correct. That is the proof.

// CHEAPEST FLIGHTS WITHIN K STOPS - Bellman-Ford shines here, because
// "at most k stops" is exactly "k+1 relaxation passes"
static int FindCheapestPrice(int n, int[][] flights, int src, int dst, int k)
{
    var dist = new int[n];
    Array.Fill(dist, int.MaxValue);
    dist[src] = 0;

    for (int i = 0; i <= k; i++)
    {
        var snapshot = (int[])dist.Clone();      // use the PREVIOUS round's values,
        foreach (var f in flights)               // or one pass can use multiple hops
        {
            if (snapshot[f[0]] == int.MaxValue) continue;
            dist[f[1]] = Math.Min(dist[f[1]], snapshot[f[0]] + f[2]);
        }
    }
    return dist[dst] == int.MaxValue ? -1 : dist[dst];
}
// The clone is essential and is the bug people ship: without it, one pass can
// chain several flights and you exceed the stop limit.

// FLOYD-WARSHALL - all pairs, O(V^3), fifteen lines
static int[,] FloydWarshall(int n, int[][] edges)
{
    const int INF = 1_000_000_000;
    var d = new int[n, n];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) d[i, j] = i == j ? 0 : INF;
    foreach (var e in edges) d[e[0], e[1]] = Math.Min(d[e[0], e[1]], e[2]);

    for (int k = 0; k < n; k++)                  // the INTERMEDIATE node is OUTERMOST
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i, k] + d[k, j] < d[i, j]) d[i, j] = d[i, k] + d[k, j];

    return d;
}
// The k loop MUST be outermost - it is the DP dimension ("paths using only the
// first k nodes as intermediates"). Putting it inside gives wrong answers, and
// it is the single most common Floyd-Warshall bug.
// A negative d[i,i] afterwards means a negative cycle through i.

// KRUSKAL - sort the edges, add any that merges two components. Needs union-find.
static int MstKruskal(int n, int[][] edges)      // edges: [u, v, weight]
{
    Array.Sort(edges, (a, b) => a[2].CompareTo(b[2]));
    var uf = new UnionFind(n);
    int total = 0, used = 0;

    foreach (var e in edges)
    {
        if (!uf.Union(e[0], e[1])) continue;     // would close a cycle: skip
        total += e[2];
        if (++used == n - 1) break;              // a tree has exactly n-1 edges
    }
    return used == n - 1 ? total : -1;           // -1 == the graph is disconnected
}
// O(E log E) for the sort, which dominates. Union returning a bool is what
// makes the cycle check one line.

// PRIM - grow a single tree from any node, always taking the cheapest edge out
static int MstPrim(int n, List<(int To, int Weight)>[] adj)
{
    var inTree = new bool[n];
    var pq = new PriorityQueue<int, int>();
    pq.Enqueue(0, 0);
    int total = 0, count = 0;

    while (pq.TryDequeue(out int node, out int weight))
    {
        if (inTree[node]) continue;              // lazy deletion again
        inTree[node] = true;
        total += weight;
        count++;
        foreach (var (next, w) in adj[node])
            if (!inTree[next]) pq.Enqueue(next, w);
    }
    return count == n ? total : -1;
}
// O(E log V). Prefer Prim on a DENSE graph, Kruskal on a SPARSE one or when
// you already have union-find to hand.

// THE CHOICE, in one table
//   unweighted shortest path          BFS                 O(V + E)
//   weights all 0 or 1                0-1 BFS (deque)     O(V + E)
//   non-negative weights              Dijkstra            O(E log V)
//   negative weights / cycle check    Bellman-Ford        O(V * E)
//   all pairs, small dense graph      Floyd-Warshall      O(V^3)
//   connect everything, min cost      Kruskal / Prim      O(E log E)
import heapq

def bellman_ford(n: int, edges: list[list[int]], source: int):
    dist = [float("inf")] * n
    dist[source] = 0

    for _ in range(n - 1):
        changed = False
        for u, v, w in edges:
            if dist[u] != float("inf") and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                changed = True
        if not changed:
            break

    has_negative_cycle = any(
        dist[u] != float("inf") and dist[u] + w < dist[v] for u, v, w in edges)
    return dist, has_negative_cycle

def find_cheapest_price(n, flights, src, dst, k) -> int:
    dist = [float("inf")] * n
    dist[src] = 0
    for _ in range(k + 1):
        snapshot = dist[:]                       # the clone is ESSENTIAL
        for u, v, w in flights:
            if snapshot[u] + w < dist[v]:
                dist[v] = snapshot[u] + w
    return -1 if dist[dst] == float("inf") else dist[dst]

def floyd_warshall(n: int, edges: list[list[int]]) -> list[list[float]]:
    INF = float("inf")
    d = [[0 if i == j else INF for j in range(n)] for i in range(n)]
    for u, v, w in edges:
        d[u][v] = min(d[u][v], w)

    for k in range(n):                           # intermediate node: OUTERMOST
        dk = d[k]
        for i in range(n):
            dik = d[i][k]
            if dik == INF:
                continue
            di = d[i]
            for j in range(n):
                if dik + dk[j] < di[j]:
                    di[j] = dik + dk[j]
    return d

def mst_kruskal(n: int, edges: list[list[int]]) -> int:
    edges.sort(key=lambda e: e[2])
    uf = UnionFind(n)
    total = used = 0
    for u, v, w in edges:
        if uf.union(u, v):
            total += w
            used += 1
            if used == n - 1:
                break
    return total if used == n - 1 else -1

def mst_prim(n: int, adj: list[list[tuple[int, int]]]) -> int:
    in_tree = [False] * n
    pq: list[tuple[int, int]] = [(0, 0)]
    total = count = 0
    while pq:
        weight, node = heapq.heappop(pq)
        if in_tree[node]:
            continue
        in_tree[node] = True
        total += weight
        count += 1
        for nxt, w in adj[node]:
            if not in_tree[nxt]:
                heapq.heappush(pq, (w, nxt))
    return total if count == n else -1

#   unweighted               BFS
#   weights 0 or 1           0-1 BFS with a deque
#   non-negative             Dijkstra
#   negative / cycle check   Bellman-Ford
#   all pairs, small dense   Floyd-Warshall
#   minimum spanning tree    Kruskal (sparse) / Prim (dense)

You are unlikely to be asked to implement these cold; you are likely to be asked which one and why. Have the selection table ready, and have the two reasons Bellman-Ford exists: negative weights, and negative-cycle detection via a V-th pass that still improves something. Two concrete bug-level details worth volunteering: in Floyd-Warshall the intermediate-node loop must be outermost (it is the DP dimension), and in "cheapest flights within k stops" each round must relax against a snapshot of the previous round or one pass chains several hops and silently exceeds the limit.

Practice
  • Implement Bellman-Ford and detect a negative cycle with the extra pass.
  • Write Floyd-Warshall, then move the k loop inside and find the wrong answer it produces.
  • Build an MST with both Kruskal and Prim and check they agree on the total.
