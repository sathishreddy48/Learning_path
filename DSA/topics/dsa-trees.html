Trees, Tries & Heaps

Trees are where recursion stops being a technique and becomes the natural shape of the code. Almost every problem here is one of four skeletons — traverse, aggregate upward, search downward, or process level by level — and recognising which one you need is most of the work.

Traversals: recursive and iterative

Four orders, and you should be able to write each one both ways. Pre-order (node, left, right) copies or serialises a tree. In-order (left, node, right) visits a BST in sorted order — which is why so many BST problems reduce to "do an in-order walk". Post-order (left, right, node) is for anything where a node needs its children's results first — heights, sums, deletion. Level-order is BFS.

The recursive versions are four lines each. The iterative ones are what interviewers ask for, and post-order is the awkward one: either push a visited flag with each node, or do a reversed pre-order and reverse the output.

public class TreeNode(int val)
{
    public int Val = val;
    public TreeNode? Left, Right;
}

// ---------- RECURSIVE: say these, then write the iterative ones ----------
static void PreOrder(TreeNode? n, List<int> acc)
{
    if (n is null) return;
    acc.Add(n.Val); PreOrder(n.Left, acc); PreOrder(n.Right, acc);
}
static void InOrder(TreeNode? n, List<int> acc)
{
    if (n is null) return;
    InOrder(n.Left, acc); acc.Add(n.Val); InOrder(n.Right, acc);
}
static void PostOrder(TreeNode? n, List<int> acc)
{
    if (n is null) return;
    PostOrder(n.Left, acc); PostOrder(n.Right, acc); acc.Add(n.Val);
}
// All O(n) time, O(h) space on the call stack.

// ---------- ITERATIVE ----------

// PRE-ORDER: push right BEFORE left, so left pops first
static List<int> PreOrderIter(TreeNode? root)
{
    var result = new List<int>();
    if (root is null) return result;

    var stack = new Stack<TreeNode>();
    stack.Push(root);
    while (stack.Count > 0)
    {
        var n = stack.Pop();
        result.Add(n.Val);
        if (n.Right is not null) stack.Push(n.Right);     // right first
        if (n.Left  is not null) stack.Push(n.Left);
    }
    return result;
}

// IN-ORDER: walk the left spine, then pop and go right. This is also the
// BST-iterator skeleton, so it is worth knowing cold.
static List<int> InOrderIter(TreeNode? root)
{
    var result = new List<int>();
    var stack = new Stack<TreeNode>();
    var node = root;

    while (node is not null || stack.Count > 0)
    {
        while (node is not null) { stack.Push(node); node = node.Left; }
        node = stack.Pop();
        result.Add(node.Val);
        node = node.Right;
    }
    return result;
}

// POST-ORDER, trick version: pre-order with the children swapped, then reverse
static List<int> PostOrderIter(TreeNode? root)
{
    var result = new List<int>();
    if (root is null) return result;

    var stack = new Stack<TreeNode>();
    stack.Push(root);
    while (stack.Count > 0)
    {
        var n = stack.Pop();
        result.Add(n.Val);
        if (n.Left  is not null) stack.Push(n.Left);      // LEFT first this time
        if (n.Right is not null) stack.Push(n.Right);
    }
    result.Reverse();                                     // node-right-left reversed
    return result;                                        //   == left-right-node
}

// POST-ORDER, honest version: a visited flag, for when you must process in
// order without buffering (e.g. freeing nodes)
static List<int> PostOrderFlag(TreeNode? root)
{
    var result = new List<int>();
    var stack = new Stack<(TreeNode Node, bool Visited)>();
    if (root is not null) stack.Push((root, false));

    while (stack.Count > 0)
    {
        var (node, visited) = stack.Pop();
        if (visited) { result.Add(node.Val); continue; }

        stack.Push((node, true));                          // re-push, now marked
        if (node.Right is not null) stack.Push((node.Right, false));
        if (node.Left  is not null) stack.Push((node.Left,  false));
    }
    return result;
}

// MORRIS TRAVERSAL - O(1) space, by temporarily rewiring the tree. Know that it
// exists and what it costs (it mutates during the walk) rather than memorising it.

// ---------- THE POST-ORDER SKELETON, which is most tree problems ----------
// "compute something for a node from its children's results"
static int Height(TreeNode? n)
    => n is null ? 0 : 1 + Math.Max(Height(n.Left), Height(n.Right));

static int CountNodes(TreeNode? n)
    => n is null ? 0 : 1 + CountNodes(n.Left) + CountNodes(n.Right);

// DIAMETER - the pattern where the RETURN value and the ANSWER differ
static int Diameter(TreeNode? root)
{
    int best = 0;
    int Depth(TreeNode? n)
    {
        if (n is null) return 0;
        int l = Depth(n.Left), r = Depth(n.Right);
        best = Math.Max(best, l + r);        // the answer: a path THROUGH this node
        return 1 + Math.Max(l, r);           // the return: depth, for the parent
    }
    Depth(root);
    return best;
}
// Returning one thing while accumulating another is the single most reusable
// tree idiom - "max path sum", "longest univalue path", "balanced check" are
// all this shape.
class TreeNode:
    __slots__ = ("val", "left", "right")
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

# RECURSIVE - generators make these read nicely and stay lazy
def pre_order(n):
    if n:
        yield n.val
        yield from pre_order(n.left)
        yield from pre_order(n.right)

def in_order(n):
    if n:
        yield from in_order(n.left)
        yield n.val
        yield from in_order(n.right)

def post_order(n):
    if n:
        yield from post_order(n.left)
        yield from post_order(n.right)
        yield n.val

# ITERATIVE
def pre_order_iter(root) -> list[int]:
    result, stack = [], [root] if root else []
    while stack:
        n = stack.pop()
        result.append(n.val)
        if n.right: stack.append(n.right)      # right first
        if n.left:  stack.append(n.left)
    return result

def in_order_iter(root) -> list[int]:
    result, stack, node = [], [], root
    while node or stack:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        result.append(node.val)
        node = node.right
    return result

def post_order_iter(root) -> list[int]:
    result, stack = [], [root] if root else []
    while stack:
        n = stack.pop()
        result.append(n.val)
        if n.left:  stack.append(n.left)       # left first
        if n.right: stack.append(n.right)
    return result[::-1]

def post_order_flag(root) -> list[int]:
    result, stack = [], [(root, False)] if root else []
    while stack:
        node, visited = stack.pop()
        if visited:
            result.append(node.val)
            continue
        stack.append((node, True))
        if node.right: stack.append((node.right, False))
        if node.left:  stack.append((node.left,  False))
    return result

# THE POST-ORDER SKELETON
def height(n) -> int:
    return 0 if n is None else 1 + max(height(n.left), height(n.right))

def diameter(root) -> int:
    best = 0
    def depth(n) -> int:
        nonlocal best                          # `nonlocal`, or you rebind a local
        if n is None:
            return 0
        l, r = depth(n.left), depth(n.right)
        best = max(best, l + r)                # the answer
        return 1 + max(l, r)                   # the return value
    depth(root)
    return best

Interview angle

Be able to produce the iterative in-order without hesitating — it is the most-asked of the three and the basis of the BST-iterator question. For post-order, the reversed-pre-order trick is the fast answer and the visited-flag version is the honest one; say which you are writing and why. The genuinely reusable idea is the diameter shape: a recursion that returns one value to its parent while accumulating a different answer in an enclosing variable. Once you see it, max-path-sum and balanced-tree checking stop being separate problems.

Practice
  • Write all four traversals iteratively, including post-order both ways.
  • Solve "diameter of a binary tree" and "maximum path sum" with the same return-vs-accumulate shape.
  • Convert a recursive traversal to iterative and confirm it handles a 100,000-node degenerate tree.

Level-order BFS, depth and views

When a problem mentions levels, depth, nearest or minimum number of steps, it wants BFS. The implementation detail that makes level problems easy is to record the queue size at the start of each level and consume exactly that many nodes — that one line separates the levels without any extra bookkeeping.

// LEVEL ORDER - the snapshot-the-count idiom
static IList<IList<int>> LevelOrder(TreeNode? root)
{
    var result = new List<IList<int>>();
    if (root is null) return result;

    var queue = new Queue<TreeNode>();
    queue.Enqueue(root);

    while (queue.Count > 0)
    {
        int size = queue.Count;                  // <- everything currently queued
        var level = new List<int>(size);         //    is exactly this level

        for (int i = 0; i < size; i++)
        {
            var node = queue.Dequeue();
            level.Add(node.Val);
            if (node.Left  is not null) queue.Enqueue(node.Left);
            if (node.Right is not null) queue.Enqueue(node.Right);
        }
        result.Add(level);
    }
    return result;
}

// MINIMUM DEPTH - BFS beats DFS here, because you can STOP at the first leaf
static int MinDepth(TreeNode? root)
{
    if (root is null) return 0;
    var queue = new Queue<(TreeNode Node, int Depth)>();
    queue.Enqueue((root, 1));

    while (queue.Count > 0)
    {
        var (node, depth) = queue.Dequeue();
        if (node.Left is null && node.Right is null) return depth;   // first leaf
        if (node.Left  is not null) queue.Enqueue((node.Left,  depth + 1));
        if (node.Right is not null) queue.Enqueue((node.Right, depth + 1));
    }
    return 0;
}
// DFS would visit the whole tree. On a tree with one deep branch and one
// shallow leaf, that is the difference between O(n) and O(1). Worth saying.

// RIGHT SIDE VIEW - the last node of each level
static IList<int> RightSideView(TreeNode? root)
{
    var result = new List<int>();
    if (root is null) return result;

    var queue = new Queue<TreeNode>();
    queue.Enqueue(root);
    while (queue.Count > 0)
    {
        int size = queue.Count;
        for (int i = 0; i < size; i++)
        {
            var node = queue.Dequeue();
            if (i == size - 1) result.Add(node.Val);     // the last one on this level
            if (node.Left  is not null) queue.Enqueue(node.Left);
            if (node.Right is not null) queue.Enqueue(node.Right);
        }
    }
    return result;
}

// ZIGZAG - the same loop, reversing alternate levels
static IList<IList<int>> Zigzag(TreeNode? root)
{
    var result = new List<IList<int>>();
    if (root is null) return result;

    var queue = new Queue<TreeNode>();
    queue.Enqueue(root);
    bool leftToRight = true;

    while (queue.Count > 0)
    {
        int size = queue.Count;
        var level = new int[size];
        for (int i = 0; i < size; i++)
        {
            var node = queue.Dequeue();
            level[leftToRight ? i : size - 1 - i] = node.Val;   // write backwards
            if (node.Left  is not null) queue.Enqueue(node.Left);
            if (node.Right is not null) queue.Enqueue(node.Right);
        }
        result.Add(level);
        leftToRight = !leftToRight;
    }
    return result;
}

// VERTICAL ORDER - the column is (parent column -1 / +1); sort at the end
static IList<IList<int>> VerticalOrder(TreeNode? root)
{
    var columns = new SortedDictionary<int, List<(int Row, int Val)>>();
    var queue = new Queue<(TreeNode Node, int Row, int Col)>();
    if (root is not null) queue.Enqueue((root, 0, 0));

    while (queue.Count > 0)
    {
        var (node, row, col) = queue.Dequeue();
        if (!columns.TryGetValue(col, out var bucket)) columns[col] = bucket = [];
        bucket.Add((row, node.Val));
        if (node.Left  is not null) queue.Enqueue((node.Left,  row + 1, col - 1));
        if (node.Right is not null) queue.Enqueue((node.Right, row + 1, col + 1));
    }
    return columns.Values
        .Select(b => (IList<int>)b.OrderBy(x => x.Row).ThenBy(x => x.Val)
                                  .Select(x => x.Val).ToList())
        .ToList();
}

// CONNECT NEXT RIGHT POINTERS in O(1) space: use the level you have already
// linked as the queue for the next one, instead of an actual queue.
from collections import deque, defaultdict

def level_order(root) -> list[list[int]]:
    result = []
    if not root:
        return result
    queue = deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):            # snapshot the count
            node = queue.popleft()
            level.append(node.val)
            if node.left:  queue.append(node.left)
            if node.right: queue.append(node.right)
        result.append(level)
    return result

def min_depth(root) -> int:
    if not root:
        return 0
    queue = deque([(root, 1)])
    while queue:
        node, depth = queue.popleft()
        if not node.left and not node.right:
            return depth                       # the FIRST leaf, so stop here
        if node.left:  queue.append((node.left,  depth + 1))
        if node.right: queue.append((node.right, depth + 1))
    return 0

def right_side_view(root) -> list[int]:
    result = []
    if not root:
        return result
    queue = deque([root])
    while queue:
        size = len(queue)
        for i in range(size):
            node = queue.popleft()
            if i == size - 1:
                result.append(node.val)
            if node.left:  queue.append(node.left)
            if node.right: queue.append(node.right)
    return result

def zigzag(root) -> list[list[int]]:
    result = []
    if not root:
        return result
    queue = deque([root])
    left_to_right = True
    while queue:
        level = deque()
        for _ in range(len(queue)):
            node = queue.popleft()
            if left_to_right: level.append(node.val)
            else:             level.appendleft(node.val)      # deque, so O(1)
            if node.left:  queue.append(node.left)
            if node.right: queue.append(node.right)
        result.append(list(level))
        left_to_right = not left_to_right
    return result

def vertical_order(root) -> list[list[int]]:
    columns: dict[int, list[tuple[int, int]]] = defaultdict(list)
    queue = deque([(root, 0, 0)]) if root else deque()
    while queue:
        node, row, col = queue.popleft()
        columns[col].append((row, node.val))
        if node.left:  queue.append((node.left,  row + 1, col - 1))
        if node.right: queue.append((node.right, row + 1, col + 1))
    return [[v for _, v in sorted(columns[c])] for c in sorted(columns)]
Interview angle

Say "snapshot the queue size at the top of each level" before you write the loop — it is the one idiom that makes every level-order variant trivial, and it reads as having done this before. The other point worth volunteering: minimum depth is a BFS problem, not a DFS one, because BFS can return at the first leaf while DFS must explore everything. On a tree with one very deep branch that is a real difference, not a theoretical one.

Practice
  • Write level-order and then derive right-side view and zigzag from it without restarting.
  • Solve minimum depth with both BFS and DFS and construct the input where BFS wins.
  • Solve vertical order and decide how ties within a cell should be broken.

BSTs: search, insert, validate and k-th smallest

The BST property is global, not local: every node in the left subtree is less than the node, and every node in the right is greater — not just the immediate children. Validating by comparing a node with its children is the classic wrong answer, and it is wrong for a tree like [5, 1, 6, null, null, 3, 7] where 3 sits in the right subtree of 5.

The correct validation passes a bounded range down the recursion. The other reliable fact: an in-order traversal of a BST is sorted, which turns k-th smallest, "is it a BST", and "recover a swapped pair" into one walk.

// SEARCH / INSERT / the shape all BST operations share: O(h)
static TreeNode? Search(TreeNode? n, int target)
{
    while (n is not null && n.Val != target) n = target < n.Val ? n.Left : n.Right;
    return n;
}

static TreeNode Insert(TreeNode? n, int val)
{
    if (n is null) return new TreeNode(val);
    if (val < n.Val) n.Left  = Insert(n.Left,  val);
    else             n.Right = Insert(n.Right, val);
    return n;
}

// DELETE - three cases, and the third is the one to get right
static TreeNode? Delete(TreeNode? n, int key)
{
    if (n is null) return null;

    if      (key < n.Val) n.Left  = Delete(n.Left,  key);
    else if (key > n.Val) n.Right = Delete(n.Right, key);
    else
    {
        if (n.Left  is null) return n.Right;        // 0 or 1 child
        if (n.Right is null) return n.Left;

        // two children: replace with the in-order SUCCESSOR (leftmost of the
        // right subtree), then delete that successor from the right subtree
        var successor = n.Right;
        while (successor.Left is not null) successor = successor.Left;
        n.Val = successor.Val;
        n.Right = Delete(n.Right, successor.Val);
    }
    return n;
}

// VALIDATE - bounds, not neighbours. THIS is the question.
static bool IsValidBst(TreeNode? n, long min = long.MinValue, long max = long.MaxValue)
{
    if (n is null) return true;
    if (n.Val <= min || n.Val >= max) return false;
    return IsValidBst(n.Left, min, n.Val) && IsValidBst(n.Right, n.Val, max);
}
// `long` for the bounds, because a node holding int.MinValue breaks an int version.
// The WRONG answer, which looks right:
//   return n.Left.Val < n.Val && n.Right.Val > n.Val && recurse...
//   -> accepts [5,1,6,null,null,3,7], because 3 < 6 locally but 3 < 5 globally.

// Or validate by checking the in-order walk is strictly increasing
static bool IsValidBstInOrder(TreeNode? root)
{
    TreeNode? prev = null;
    var stack = new Stack<TreeNode>();
    var node = root;
    while (node is not null || stack.Count > 0)
    {
        while (node is not null) { stack.Push(node); node = node.Left; }
        node = stack.Pop();
        if (prev is not null && node.Val <= prev.Val) return false;
        prev = node;
        node = node.Right;
    }
    return true;
}

// K-TH SMALLEST - stop the in-order walk at k. O(h + k), not O(n).
static int KthSmallest(TreeNode? root, int k)
{
    var stack = new Stack<TreeNode>();
    var node = root;
    while (node is not null || stack.Count > 0)
    {
        while (node is not null) { stack.Push(node); node = node.Left; }
        node = stack.Pop();
        if (--k == 0) return node.Val;
        node = node.Right;
    }
    throw new ArgumentException("k is larger than the tree");
}
// The follow-up: "what if kth-smallest is called often and the tree is modified?"
// -> store a subtree size on each node, and the query becomes O(h).

// SORTED ARRAY -> BALANCED BST: the middle is the root, recurse on each half
static TreeNode? FromSorted(int[] a, int lo, int hi)
{
    if (lo > hi) return null;
    int mid = lo + (hi - lo) / 2;
    return new TreeNode(a[mid]) { Left  = FromSorted(a, lo, mid - 1),
                                  Right = FromSorted(a, mid + 1, hi) };
}

// IN-ORDER SUCCESSOR, without a parent pointer: the last node where you went left
static TreeNode? Successor(TreeNode? root, TreeNode target)
{
    TreeNode? best = null;
    var node = root;
    while (node is not null)
    {
        if (target.Val < node.Val) { best = node; node = node.Left; }
        else                        node = node.Right;
    }
    return best;
}

// BALANCE: everything above is O(h), and h is O(n) on sorted insertions.
// AVL and red-black trees keep h at O(log n) by rotating. You will almost never
// be asked to implement one - but know that SortedDictionary and SortedSet are
// red-black trees, which is why they are O(log n) and Dictionary is O(1).
def search(n, target):
    while n and n.val != target:
        n = n.left if target < n.val else n.right
    return n

def insert(n, val):
    if n is None:
        return TreeNode(val)
    if val < n.val: n.left  = insert(n.left,  val)
    else:           n.right = insert(n.right, val)
    return n

def delete(n, key):
    if n is None:
        return None
    if key < n.val:
        n.left = delete(n.left, key)
    elif key > n.val:
        n.right = delete(n.right, key)
    else:
        if n.left is None:  return n.right
        if n.right is None: return n.left
        successor = n.right                       # leftmost of the right subtree
        while successor.left:
            successor = successor.left
        n.val = successor.val
        n.right = delete(n.right, successor.val)
    return n

def is_valid_bst(n, low=float("-inf"), high=float("inf")) -> bool:
    if n is None:
        return True
    if not (low < n.val < high):
        return False
    return is_valid_bst(n.left, low, n.val) and is_valid_bst(n.right, n.val, high)
# float("inf") avoids C#'s int.MinValue problem entirely

def is_valid_bst_in_order(root) -> bool:
    prev = None
    stack, node = [], root
    while node or stack:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        if prev is not None and node.val <= prev:
            return False
        prev = node.val
        node = node.right
    return True

def kth_smallest(root, k: int) -> int:
    stack, node = [], root
    while node or stack:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        k -= 1
        if k == 0:
            return node.val
        node = node.right
    raise ValueError("k exceeds the tree size")

# Or, with the generator from the traversals section - lazy, so it stops early
from itertools import islice
def kth_smallest_gen(root, k: int) -> int:
    return next(islice(in_order(root), k - 1, k))

def from_sorted(a: list[int], lo: int = 0, hi: int | None = None):
    if hi is None:
        hi = len(a) - 1
    if lo > hi:
        return None
    mid = (lo + hi) // 2
    return TreeNode(a[mid], from_sorted(a, lo, mid - 1), from_sorted(a, mid + 1, hi))

def successor(root, target):
    best, node = None, root
    while node:
        if target.val < node.val:
            best, node = node, node.left          # remember where we went left
        else:
            node = node.right
    return best

Interview angle

"Validate a BST" is asked constantly and the local comparison is the trap it is designed to catch — have the counterexample ready ([5,1,6,null,null,3,7]) rather than just asserting that bounds are needed. Mention the int.MinValue edge case in the bounded version, because it is the second trap. For k-th smallest, note the walk can stop at k for O(h + k), and have the follow-up answer ready: if the query is frequent and the tree mutates, store subtree sizes on the nodes.

Practice
  • Validate a BST with bounds and construct the tree that defeats the local comparison.
  • Find the k-th smallest, then extend the node to carry a subtree size and make it O(h).
  • Delete a node with two children and confirm the tree is still a valid BST.

Lowest common ancestor and serialisation

Two staples. LCA in a plain binary tree is the cleanest small recursion in the whole set: return the node if you find either target, recurse both sides, and the node where both sides come back non-null is the answer. In a BST it is even simpler — walk down while both targets are on the same side, and the first node that splits them is the LCA.

Serialise/deserialise tests whether you understand what information a traversal preserves. Pre-order with null markers determines a tree uniquely; pre-order alone does not, and in-order alone never does.

// LCA in a general binary tree. Assumes both nodes exist in the tree.
static TreeNode? Lca(TreeNode? root, TreeNode p, TreeNode q)
{
    if (root is null || root == p || root == q) return root;

    var left  = Lca(root.Left,  p, q);
    var right = Lca(root.Right, p, q);

    if (left is not null && right is not null) return root;   // they split here
    return left ?? right;                                     // both are below one side
}
// O(n) time, O(h) space. If the nodes might NOT be present, you need to count
// matches rather than return early - say so, because it changes the code.

// LCA in a BST - no recursion needed, O(h)
static TreeNode? LcaBst(TreeNode? node, TreeNode p, TreeNode q)
{
    while (node is not null)
    {
        if      (p.Val < node.Val && q.Val < node.Val) node = node.Left;
        else if (p.Val > node.Val && q.Val > node.Val) node = node.Right;
        else return node;                      // they diverge here, or one IS node
    }
    return null;
}

// WITH PARENT POINTERS: walk up from both, the same two-pointer trick as
// "intersection of two linked lists"
static TreeNode? LcaWithParents(TreeNode a, TreeNode b)
{
    TreeNode? p = a, q = b;
    while (p != q)
    {
        p = p?.Parent ?? b;                    // switch to the other list on null
        q = q?.Parent ?? a;
    }
    return p;
}

// Pre-order WITH null markers. Pre-order alone does not determine a tree;
// pre-order + null markers does, which is the insight being tested.
public class Codec
{
    public string Serialize(TreeNode? root)
    {
        var sb = new StringBuilder();
        void Write(TreeNode? n)
        {
            if (n is null) { sb.Append("#,"); return; }
            sb.Append(n.Val).Append(',');
            Write(n.Left);
            Write(n.Right);
        }
        Write(root);
        return sb.ToString();
    }

    public TreeNode? Deserialize(string data)
    {
        var tokens = data.Split(',', StringSplitOptions.RemoveEmptyEntries);
        int i = 0;

        TreeNode? Read()
        {
            var token = tokens[i++];
            if (token == "#") return null;
            return new TreeNode(int.Parse(token))
            {
                Left  = Read(),                  // ORDER MATTERS: the children are
                Right = Read(),                  // consumed left-then-right
            };
        }
        return Read();
    }
}
// Careful in C#: object-initialiser assignments run in source order, so
// Left = Read() before Right = Read() is correct here - but if you wrote
// `new TreeNode(v) { Right = Read(), Left = Read() }` it would silently
// deserialise mirrored. Being aware of evaluation order is the subtle bit.

// LEVEL-ORDER serialisation is the LeetCode input format, and worth knowing
// because you will be handed strings in it: "[1,2,3,null,null,4,5]"

// BUILD FROM TWO TRAVERSALS - pre-order + in-order determines a tree uniquely
// (assuming distinct values); post-order + in-order also does; pre + post does NOT.
static TreeNode? BuildTree(int[] preorder, int[] inorder)
{
    var indexOf = new Dictionary<int, int>();
    for (int i = 0; i < inorder.Length; i++) indexOf[inorder[i]] = i;   // O(1) lookup

    int p = 0;
    TreeNode? Build(int lo, int hi)
    {
        if (lo > hi) return null;
        int rootVal = preorder[p++];             // pre-order gives the root
        int mid = indexOf[rootVal];              // in-order splits the subtrees
        return new TreeNode(rootVal)
        {
            Left  = Build(lo, mid - 1),
            Right = Build(mid + 1, hi),
        };
    }
    return Build(0, inorder.Length - 1);
}
// The dictionary is what makes it O(n) rather than O(n^2) - scanning the
// in-order array for the root each time is the naive version.
def lca(root, p, q):
    if root is None or root is p or root is q:
        return root
    left, right = lca(root.left, p, q), lca(root.right, p, q)
    if left and right:
        return root
    return left or right

def lca_bst(node, p, q):

        if   p.val < node.val and q.val < node.val: node = node.left
        elif p.val > node.val and q.val > node.val: node = node.right
        else: return node
    return None

def lca_with_parents(a, b):

        q = q.parent if q else a

def build_tree(preorder: list[int], inorder: list[int]):
    index_of = {v: i for i, v in enumerate(inorder)}     # O(1) lookup
    pre = iter(preorder)

    def build(lo: int, hi: int):
        if lo > hi:
            return None
        root_val = next(pre)
        mid = index_of[root_val]
        node = TreeNode(root_val)
        node.left  = build(lo, mid - 1)
        node.right = build(mid + 1, hi)
        return node

    return build(0, len(inorder) - 1)

Tries: prefix search and word problems

A trie stores strings by sharing prefixes: each node is a character, each path is a prefix, and a flag marks where a complete word ends. Lookup and insert are O(k) in the length of the word, independent of how many words are stored — which is the property a hash map cannot match, because a hash map can only answer exact-match questions.

Reach for it when you see "prefix", "autocomplete", "dictionary of words", or a grid word-search with many target words. The last one is the real payoff: searching a board for 10,000 words one at a time is hopeless, but one DFS guided by a trie prunes the moment the current path is not a prefix of anything.

public class Trie
{
    private sealed class Node
    {
        // an array of 26 is faster than a Dictionary for lowercase ASCII, and
        // much heavier for a large alphabet - say which you chose and why
        public readonly Node?[] Children = new Node?[26];
        public bool IsWord;
    }

    private readonly Node _root = new();

    public void Insert(string word)
    {
        var node = _root;
        foreach (var c in word)
        {
            int i = c - 'a';
            node = node.Children[i] ??= new Node();
        }
        node.IsWord = true;
    }

    public bool Search(string word)  => Find(word) is { IsWord: true };
    public bool StartsWith(string p) => Find(p) is not null;

    private Node? Find(string prefix)
    {
        var node = _root;
        foreach (var c in prefix)
        {
            node = node.Children[c - 'a'];
            if (node is null) return null;
        }
        return node;
    }

    // AUTOCOMPLETE: walk to the prefix, then collect everything below it
    public List<string> WithPrefix(string prefix, int limit = 10)
    {
        var results = new List<string>();
        var start = Find(prefix);
        if (start is null) return results;

        var sb = new StringBuilder(prefix);
        void Collect(Node node)
        {
            if (results.Count == limit) return;
            if (node.IsWord) results.Add(sb.ToString());
            for (int i = 0; i < 26; i++)
            {
                if (node.Children[i] is not { } child) continue;
                sb.Append((char)('a' + i));
                Collect(child);
                sb.Length--;                     // backtrack
            }
        }
        Collect(start);
        return results;
    }
}
// Insert and search: O(k). Space: O(total characters), shared across prefixes.

// WILDCARD SEARCH - '.' matches any character, so branch on every child
public bool SearchWildcard(string word)
{
    bool Match(Node node, int i)
    {
        if (i == word.Length) return node.IsWord;
        if (word[i] != '.') return node.Children[word[i] - 'a'] is { } c && Match(c, i + 1);

        foreach (var child in node.Children)     // '.' - try them all
            if (child is not null && Match(child, i + 1)) return true;
        return false;
    }
    return Match(_root, 0);
}

// WORD SEARCH II - the payoff. One DFS over the board, pruned by the trie.
// Searching 10,000 words one at a time is hopeless; this is one pass.
static IList<string> FindWords(char[][] board, string[] words)
{
    var trie = new TrieNode();
    foreach (var w in words) trie.Insert(w);        // build once

    var found = new HashSet<string>();
    int rows = board.Length, cols = board[0].Length;

    void Dfs(int r, int c, TrieNode node)
    {
        if (r < 0 || r >= rows || c < 0 || c >= cols) return;

        char ch = board[r][c];
        if (ch == '#' || node.Children[ch - 'a'] is not { } next) return;
        //                ^ the prune: this path is not a prefix of ANY word

        if (next.Word is not null)
        {
            found.Add(next.Word);
            next.Word = null;                       // de-duplicate without a set scan
        }

        board[r][c] = '#';                          // mark visited, in place
        Dfs(r + 1, c, next); Dfs(r - 1, c, next);
        Dfs(r, c + 1, next); Dfs(r, c - 1, next);
        board[r][c] = ch;                           // restore
    }

    for (int r = 0; r < rows; r++)
        for (int c = 0; c < cols; c++) Dfs(r, c, trie);

    return found.ToList();
}
// Storing the whole WORD on the terminal node (rather than a bool) avoids
// rebuilding the string during the DFS - a small but real optimisation.

// OTHER USES worth naming: longest common prefix of a set, IP routing tables
// (longest prefix match), the "replace words with their root" problem, and

class TrieNode:
    __slots__ = ("children", "is_word")
    def __init__(self) -> None:
        self.children: dict[str, "TrieNode"] = {}    # dict: any alphabet, sparse
        self.is_word = False

class Trie:
    def __init__(self) -> None:
        self.root = TrieNode()

    def insert(self, word: str) -> None:
        node = self.root
        for c in word:
            node = node.children.setdefault(c, TrieNode())
        node.is_word = True

    def _find(self, prefix: str) -> TrieNode | None:
        node = self.root
        for c in prefix:
            node = node.children.get(c)
            if node is None:
                return None
        return node

    def search(self, word: str) -> bool:
        node = self._find(word)
        return node is not None and node.is_word

    def starts_with(self, prefix: str) -> bool:
        return self._find(prefix) is not None

    def with_prefix(self, prefix: str, limit: int = 10) -> list[str]:
        results: list[str] = []
        start = self._find(prefix)
        if start is None:
            return results

        def collect(node: TrieNode, path: list[str]) -> None:
            if len(results) == limit:
                return
            if node.is_word:
                results.append(prefix + "".join(path))
            for c, child in node.children.items():
                path.append(c)
                collect(child, path)
                path.pop()                            # backtrack

        collect(start, [])
        return results

def find_words(board: list[list[str]], words: list[str]) -> list[str]:
    root: dict = {}
    for w in words:                                   # build the trie
        node = root
        for c in w:
            node = node.setdefault(c, {})
        node["$"] = w                                 # store the whole word

    rows, cols = len(board), len(board[0])
    found: set[str] = set()

    def dfs(r: int, c: int, node: dict) -> None:
        ch = board[r][c]
        nxt = node.get(ch)
        if nxt is None:                               # the prune
            return
        word = nxt.get("$")
        if word:
            found.add(word)
            del nxt["$"]                              # do not find it twice

        board[r][c] = "#"
        for dr, dc in ((1,0), (-1,0), (0,1), (0,-1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and board[nr][nc] != "#":
                dfs(nr, nc, nxt)
        board[r][c] = ch

    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    return list(found)

Heaps and PriorityQueue

A heap gives O(log n) insert and O(1) peek of the min/max — perfect for “top k”, “k-th largest”, merging k sorted lists, and scheduling. .NET 6 added PriorityQueue<TElement, TPriority> (min-heap; negate for max). Python's heapq is a min-heap over lists.

// K-th largest: keep a min-heap of size k
static int KthLargest(int[] nums, int k)
{
    var pq = new PriorityQueue<int, int>();
    foreach (var n in nums) { pq.Enqueue(n, n); if (pq.Count > k) pq.Dequeue(); }
    return pq.Peek();
}

static ListNode? MergeK(ListNode?[] lists)
{
    var pq = new PriorityQueue<ListNode, int>();
    foreach (var l in lists) if (l is not null) pq.Enqueue(l, l.Val);
    var dummy = new ListNode(0); var tail = dummy;

        if (node.Next is not null) pq.Enqueue(node.Next, node.Next.Val);

Know the three things a heap is for: the k largest or smallest (a bounded heap of size k gives O(n log k), beating a full sort), a running median (two heaps facing each other), and the frontier in Dijkstra or A*. The .NET specifics worth saying: PriorityQueue is a min-heap, it is not stable for equal priorities, and EnqueueDequeue is cheaper than an enqueue followed by a dequeue. Top-k, merge-k and the two-heap median

Three heap patterns cover most of what gets asked. Bounded heap for top-k: keep a min-heap of size k and evict the smallest, giving O(n log k) and constant memory in k — the right answer for a stream. Merge k sorted sequences: a heap holding one element from each source, which is O(N log k). Two heaps for a running median: a max-heap of the lower half and a min-heap of the upper half, kept within one element of each other, so the median is always at the tops.

// 1. TOP-K with a BOUNDED heap - O(n log k), and it works on a stream
static int[] TopK(int[] nums, int k)
{
    var heap = new PriorityQueue<int, int>();        // min-heap: the smallest is on top
    foreach (var n in nums)
    {
        heap.Enqueue(n, n);
        if (heap.Count > k) heap.Dequeue();          // evict the smallest
    }
    return heap.UnorderedItems.Select(x => x.Element).ToArray();
}
// Counter-intuitive but correct: to find the k LARGEST you use a MIN-heap, so
// that the easiest element to discard is the smallest of your current best k.
// Being able to explain that inversion is the point of the question.

// EnqueueDequeue avoids growing to k+1 and shrinking back
foreach (var n in nums)
{
    if (heap.Count < k) heap.Enqueue(n, n);
    else if (n > heap.Peek()) heap.EnqueueDequeue(n, n);
}

// K CLOSEST POINTS TO THE ORIGIN - the same shape, with a computed priority
static int[][] KClosest(int[][] points, int k)
{
    var heap = new PriorityQueue<int[], int>();
    foreach (var p in points)
    {
        int d = p[0] * p[0] + p[1] * p[1];           // no sqrt: it is monotonic
        heap.Enqueue(p, -d);                          // negate for a max-heap
        if (heap.Count > k) heap.Dequeue();           // evict the FURTHEST
    }
    return heap.UnorderedItems.Select(x => x.Element).ToArray();
}

// 2. MERGE K SORTED LISTS - O(N log k)
static int[] MergeK(int[][] lists)
{
    var heap = new PriorityQueue<(int Value, int List, int Index), int>();
    for (int i = 0; i < lists.Length; i++)
        if (lists[i].Length > 0) heap.Enqueue((lists[i][0], i, 0), lists[i][0]);

    var result = new List<int>();
    while (heap.TryDequeue(out var item, out _))
    {
        result.Add(item.Value);
        int next = item.Index + 1;
        if (next < lists[item.List].Length)
            heap.Enqueue((lists[item.List][next], item.List, next),
                         lists[item.List][next]);
    }
    return result.ToArray();
}

// merging a terabyte of sorted files with k in memory, not N.

// 3. TWO HEAPS - a running median in O(log n) per insert, O(1) per query
public class MedianFinder
{
    private readonly PriorityQueue<int, int> _low = new();    // MAX-heap (negated)
    private readonly PriorityQueue<int, int> _high = new();   // min-heap

    public void Add(int num)
    {
        _low.Enqueue(num, -num);                       // always insert into low
        _high.Enqueue(_low.Dequeue(), _low.Peek());    // move its largest to high

        if (_high.Count > _low.Count)                  // rebalance: low may hold
        {                                              // one extra, high never
            var x = _high.Dequeue();
            _low.Enqueue(x, -x);
        }
    }

    public double Median() => _low.Count > _high.Count
        ? _low.Peek()
        : (_low.Peek() + _high.Peek()) / 2.0;
}

// |_low| - |_high| is 0 or 1. Push-then-pop-then-rebalance maintains it without
// any comparisons, which is why this formulation is the one to memorise.

def k_closest(points: list[list[int]], k: int) -> list[list[int]]:
    heap: list[tuple[int, list[int]]] = []
    for p in points:

    return [p for _, p in heap]

def merge_k(lists: list[list[int]]) -> list[int]:
    heap = [(lst[0], i, 0) for i, lst in enumerate(lists) if lst]
    heapq.heapify(heap)
    result = []
    while heap:
        value, li, idx = heapq.heappop(heap)
        result.append(value)
        if idx + 1 < len(lists[li]):
            heapq.heappush(heap, (lists[li][idx + 1], li, idx + 1))
    return result

# 3. TWO HEAPS
class MedianFinder:
    def __init__(self) -> None:
        self.low: list[int] = []             # max-heap, values negated
        self.high: list[int] = []            # min-heap

    def add(self, num: int) -> None:
        heapq.heappush(self.low, -num)
        heapq.heappush(self.high, -heapq.heappop(self.low))   # largest of low -> high
        if len(self.high) > len(self.low):                    # rebalance
            heapq.heappush(self.low, -heapq.heappop(self.high))

    def median(self) -> float:
        if len(self.low) > len(self.high):
            return float(-self.low[0])
        return (-self.low[0] + self.high[0]) / 2

# 4. TASK SCHEDULER - a max-heap of remaining counts plus a cooldown queue
from collections import deque
def least_interval(tasks: list[str], n: int) -> int:
    heap = [-c for c in Counter(tasks).values()]
    heapq.heapify(heap)
    cooldown: deque[tuple[int, int]] = deque()      # (available_at, remaining)
    time = 0
    while heap or cooldown:
        time += 1
        if cooldown and cooldown[0][0] == time:
            heapq.heappush(heap, cooldown.popleft()[1])
        if heap:
            remaining = heapq.heappop(heap) + 1
            if remaining:
                cooldown.append((time + n + 1, remaining))
    return time
Interview angle

The question worth rehearsing an answer to is the inversion: to find the k largest you keep a min-heap, because the thing you want to discard cheaply is the smallest of your current best k. Candidates who have only memorised the code get this backwards under pressure. For the running median, state the invariant before the code — every element of the low half ≤ every element of the high half, sizes differing by at most one — and have the hard follow-up ready: a sliding-window median needs arbitrary removal, which a binary heap cannot do, so you use lazy deletion or a balanced tree.

What .NET gives you: SortedSet, SortedDictionary and SortedList

Worth a minute of preparation, because the common complaint that "C# has no TreeMap" is wrong and knowing the three options is a cheap signal. SortedDictionary and SortedSet are red-black trees: O(log n) insert, delete and lookup, and they maintain order. SortedList is an array pair: O(log n) lookup but O(n) insert, and much less memory — it is for build-once-then-read-many.

The one everybody forgets is SortedSet.GetViewBetween, which gives a live range view in O(log n) and is exactly what range queries and "nearest neighbour" problems want. Python, by contrast, genuinely has no balanced tree in the standard library — bisect on a list gives O(log n) search but O(n) insert, and sortedcontainers is the standard third-party answer.

// SortedSet<T> - a red-black tree. Ordered, unique, O(log n).
var set = new SortedSet<int>([5, 1, 3]);
set.Min; set.Max;                               // O(log n)
set.Add(4); set.Remove(1);                      // O(log n)

// The underused one: a LIVE view over a range, O(log n) to create
var inRange = set.GetViewBetween(2, 4);         // {3, 4}
inRange.Count;                                   // the count in that range
// Mutating the view mutates the set. Useful, and occasionally surprising.

// "The nearest value to x" - what a Dictionary cannot answer at all
var atLeast = set.GetViewBetween(x, int.MaxValue).Min;     // ceiling
var atMost  = set.GetViewBetween(int.MinValue, x).Max;     // floor

// With a comparer, which turns it into a sorted multiset over objects
var byDeadline = new SortedSet<Task>(
    Comparer<Task>.Create((a, b) =>
        a.Deadline != b.Deadline ? a.Deadline.CompareTo(b.Deadline)
                                 : a.Id.CompareTo(b.Id)));   // tiebreak, or it
                                                             // deduplicates!
// A SortedSet DROPS elements its comparer calls equal. Always include a unique
// tiebreaker when you are using it as a sorted multiset - this is a real bug.

// SortedDictionary<K,V> - a red-black tree keyed by K
var sd = new SortedDictionary<int, string>();
sd[5] = "five";                                 // O(log n) insert
foreach (var (k, v) in sd) { }                  // in key order

// SortedList<K,V> - two parallel ARRAYS. Different trade-off entirely.
var sl = new SortedList<int, string>();
sl[5] = "five";                                 // O(n) insert - it shifts
var v = sl[5];                                  // O(log n) lookup
var byIndex = sl.Values[0];                     // O(1) BY INDEX - the unique feature
//   Use when: you build it once and then read a lot, or you need indexed access.
//   Memory: much lower than SortedDictionary (no node objects).

// THE CHOICE, in one table
//   need                                   use
//   ------------------------------------   -------------------------
//   O(1) lookup, no order                  Dictionary / HashSet
//   ordered iteration, frequent writes     SortedDictionary / SortedSet
//   ordered, built once, read often        SortedList
//   only the min or max                    PriorityQueue
//   indexed access AND order               SortedList
//   range queries                          SortedSet.GetViewBetween

// WHAT IS MISSING: there is no built-in skip list, no order-statistic tree
// (so no "the k-th element" in O(log n)), and SortedSet has no indexer.
// For "k-th smallest with updates" you augment a BST with subtree sizes or use
// a Fenwick tree - worth naming if the problem demands it.

// FENWICK (binary indexed) TREE - prefix sums with updates, in ~15 lines.
// Worth having if "range sum with point updates" comes up.
public class Fenwick(int n)
{
    private readonly int[] _tree = new int[n + 1];

    public void Add(int i, int delta)                      // O(log n)
    {
        for (i++; i < _tree.Length; i += i & -i) _tree[i] += delta;
    }
    public int PrefixSum(int i)                            // O(log n)
    {
        int sum = 0;
        for (i++; i > 0; i -= i & -i) sum += _tree[i];
        return sum;
    }
    public int RangeSum(int l, int r) => PrefixSum(r) - PrefixSum(l - 1);
}

# Fenwick tree, same ~15 lines
class Fenwick:
    def __init__(self, n: int) -> None:
        self.tree = [0] * (n + 1)

    def add(self, i: int, delta: int) -> None:
        i += 1
        while i < len(self.tree):
            self.tree[i] += delta
            i += i & -i

    def prefix_sum(self, i: int) -> int:
        i += 1
        total = 0
        while i > 0:
            total += self.tree[i]
            i -= i & -i
        return total

    def range_sum(self, l: int, r: int) -> int:
        return self.prefix_sum(r) - (self.prefix_sum(l - 1) if l else 0)
Interview angle

Mostly a knowledge check, with one real trap: a SortedSet silently drops elements its comparer considers equal, so using it as a sorted multiset without a unique tiebreaker loses data. Knowing GetViewBetween exists is a small differentiator for range queries. In Python, say plainly that there is no balanced tree in the stdlib, give bisect with its O(n) insert, and name sortedcontainers as the standard answer — that reads as accurate rather than evasive.
