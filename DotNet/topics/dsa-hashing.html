Hashing, Sets & Counting

Trading space for time. A dictionary or a set turns "have I seen this before?" from an O(n) scan into an O(1) lookup, and that one substitution solves a startling proportion of interview problems. This page covers the idiomatic forms and the two places the trick breaks down.

Dictionary frequency map

The single most common interview structure: count things in a Dictionary<TKey,int> then answer a question about the counts. Canonical: two-sum, first unique character, top-k frequent elements.

// Two Sum: O(n) with a value→index map
static int[] TwoSum(int[] nums, int target)
{
    var seen = new Dictionary<int, int>();
    for (int i = 0; i < nums.Length; i++)
    {
        if (seen.TryGetValue(target - nums[i], out var j)) return new[] { j, i };
        seen[nums[i]] = i;
    }
    return Array.Empty<int>();
}

// First unique character
static int FirstUnique(string s)
{
    var counts = new Dictionary<char, int>();
    foreach (var c in s) counts[c] = counts.GetValueOrDefault(c) + 1;
    for (int i = 0; i < s.Length; i++) if (counts[s[i]] == 1) return i;
    return -1;
}

// Top-k frequent: count, then sort or heap
static int[] TopK(int[] nums, int k) =>
    nums.GroupBy(n => n)
        .OrderByDescending(g => g.Count())
        .Take(k)
        .Select(g => g.Key)
        .ToArray();

HashSet: membership, dedup and "visited"

When you only need to know whether something is present, a set is the answer and it comes up in three guises: membership tests inside a loop (which is what turns O(n·m) into O(n+m)), deduplication, and the "visited" set that makes graph traversal terminate.

The C# detail worth knowing: HashSet<T>.Add returns false if the element was already present, so you get the test and the insert in one lookup instead of Contains followed by Add.

// The substitution that fixes most accidental O(n^2)
foreach (var x in a) if (b.Contains(x)) { }          // List.Contains: O(n * m)

var lookup = b.ToHashSet();
foreach (var x in a) if (lookup.Contains(x)) { }     // O(n + m)

// Add returns false when it was already there - one lookup, not two
var seen = new HashSet<int>();
foreach (var x in nums)
    if (!seen.Add(x)) return x;                      // first duplicate

// Deduplicate while PRESERVING order (a plain HashSet does not promise order)
static List<T> Distinct<T>(IEnumerable<T> items)
{
    var seen = new HashSet<T>();
    var result = new List<T>();
    foreach (var item in items) if (seen.Add(item)) result.Add(item);
    return result;
}
//  LINQ's .Distinct() already preserves order; say so rather than reimplementing it.

// LONGEST CONSECUTIVE SEQUENCE - the classic "set makes it O(n)" problem.
// Sorting is O(n log n); this is O(n) by only starting a run at its true start.
static int LongestConsecutive(int[] nums)
{
    var set = nums.ToHashSet();
    int best = 0;

    foreach (var n in set)
    {
        if (set.Contains(n - 1)) continue;       // not the start of a run: skip
        int length = 1;
        while (set.Contains(n + length)) length++;
        best = Math.Max(best, length);
    }
    return best;
}

Grouping by a derived key

A whole family of problems reduces to: compute a canonical key for each item, then bucket by it. Anagrams group by their sorted letters; points on a line group by slope; strings group by their shape. The work is choosing the key, and the key must be cheap and must collide exactly when the items should group.

For anagrams there are two keys worth knowing: the sorted string (O(k log k) per word, trivially correct) and the 26-element count vector rendered as a string (O(k), better when words are long).

// GROUP ANAGRAMS - key by the sorted letters
static IList<IList<string>> GroupAnagrams(string[] words)
{
    var groups = new Dictionary<string, List<string>>();
    foreach (var word in words)
    {
        var key = string.Concat(word.OrderBy(c => c));         // O(k log k)
        if (!groups.TryGetValue(key, out var bucket))
            groups[key] = bucket = [];
        bucket.Add(word);
    }
    return groups.Values.Cast<IList<string>>().ToList();
}

// The O(k) key: a count vector, when words are long enough for it to matter
static string CountKey(string word)
{
    var counts = new int[26];
    foreach (var c in word) counts[c - 'a']++;
    return string.Join(',', counts);        // or build it with a StringBuilder
}
// O(n * k) total instead of O(n * k log k). Mention the trade-off, then pick.

// LINQ says it directly, and in an interview this is fine once you have shown
// you know what it costs

// ISOMORPHIC STRINGS / WORD PATTERN - the key is the SHAPE
static string Shape(string s)
{
    var map = new Dictionary<char, int>();
    var sb = new StringBuilder();
    foreach (var c in s)
    {
        if (!map.TryGetValue(c, out int id)) map[c] = id = map.Count;
        sb.Append(id).Append(',');
    }
    return sb.ToString();
}
// "egg" -> "0,1,1,"   "add" -> "0,1,1,"   same shape, so isomorphic

// MAX POINTS ON A LINE - key by the normalised slope from each point
static int MaxPoints(int[][] points)
{
    if (points.Length <= 2) return points.Length;
    int best = 0;

    for (int i = 0; i < points.Length; i++)
    {
        var slopes = new Dictionary<(int, int), int>();
        for (int j = i + 1; j < points.Length; j++)
        {
            int dy = points[j][1] - points[i][1], dx = points[j][0] - points[i][0];
            int g = Gcd(Math.Abs(dy), Math.Abs(dx));
            if (g != 0) { dy /= g; dx /= g; }
            if (dx < 0 || (dx == 0 && dy < 0)) { dx = -dx; dy = -dy; }   // canonicalise
            var key = (dy, dx);
            slopes[key] = slopes.GetValueOrDefault(key) + 1;
            best = Math.Max(best, slopes[key]);
        }
    }
    return best + 1;
}
static int Gcd(int a, int b) => b == 0 ? a : Gcd(b, a % b);

        groups["".join(sorted(word))].append(word)        # O(k log k) key

# The O(k) key: a tuple of counts, which is hashable for free
def group_anagrams_linear(words: list[str]) -> list[list[str]]:
    groups: dict[tuple, list[str]] = defaultdict(list)
    for word in words:
        counts = [0] * 26
        for c in word:
            counts[ord(c) - ord("a")] += 1
        groups[tuple(counts)].append(word)                # tuple: hashable
    return list(groups.values())

# ISOMORPHIC - key by shape
def shape(s: str) -> tuple:
    mapping: dict[str, int] = {}
    return tuple(mapping.setdefault(c, len(mapping)) for c in s)

# MAX POINTS ON A LINE
def max_points(points: list[list[int]]) -> int:
    if len(points) <= 2:
        return len(points)
    best = 0
    for i, (x1, y1) in enumerate(points):
        slopes: dict[tuple[int, int], int] = defaultdict(int)
        for x2, y2 in points[i + 1:]:
            dy, dx = y2 - y1, x2 - x1
            g = gcd(abs(dy), abs(dx)) or 1
            dy, dx = dy // g, dx // g
            if dx < 0 or (dx == 0 and dy < 0):            # canonicalise direction
                dy, dx = -dy, -dx
            slopes[(dy, dx)] += 1
            best = max(best, slopes[(dy, dx)])
    return best + 1
# Integer pairs as the key - never a float slope

Prefix sums plus a map: subarray counting

The single most useful compound technique on this page, and the one that most often appears as the "hard" question. The pattern: as you sweep, keep a running prefix sum and a map from prefix value → how many times it has occurred. A subarray summing to k ends at i exactly when prefix[i] - k has been seen before, so the count of such earlier prefixes is the number of qualifying subarrays ending here.

The trick generalises further than people expect: with prefix % k as the key it counts subarrays divisible by k, and by mapping 0→−1 and 1→+1 it counts the longest subarray with equal zeroes and ones.

// SUBARRAY SUM EQUALS K - count, O(n)
static int SubarraySum(int[] nums, int k)
{
    var counts = new Dictionary<int, int> { [0] = 1 };    // the empty prefix: vital
    int sum = 0, result = 0;

    foreach (var n in nums)
    {
        sum += n;
        // every earlier prefix equal to (sum - k) ends a qualifying subarray here
        result += counts.GetValueOrDefault(sum - k);
        counts[sum] = counts.GetValueOrDefault(sum) + 1;
    }
    return result;
}
// The [0] = 1 seed is what makes a subarray starting at index 0 count. Omitting
// it is the single most common bug in this pattern.

// The same shape, asking for the LONGEST rather than the count: store the FIRST
// index at which each prefix appeared, and never overwrite it.
static int LongestSubarraySum(int[] nums, int k)
{
    var firstIndex = new Dictionary<int, int> { [0] = -1 };
    int sum = 0, best = 0;

    for (int i = 0; i < nums.Length; i++)
    {
        sum += nums[i];
        if (firstIndex.TryGetValue(sum - k, out int j)) best = Math.Max(best, i - j);
        if (!firstIndex.ContainsKey(sum)) firstIndex[sum] = i;   // FIRST only
    }
    return best;
}

// CONTIGUOUS ARRAY - longest subarray with equal 0s and 1s.
// Map 0 -> -1, 1 -> +1, and the question becomes "longest subarray summing to 0".
static int FindMaxLength(int[] nums)
{
    var firstIndex = new Dictionary<int, int> { [0] = -1 };
    int sum = 0, best = 0;

    for (int i = 0; i < nums.Length; i++)
    {
        sum += nums[i] == 0 ? -1 : 1;
        if (firstIndex.TryGetValue(sum, out int j)) best = Math.Max(best, i - j);
        else firstIndex[sum] = i;
    }
    return best;
}

// SUBARRAY SUMS DIVISIBLE BY K - key on the remainder instead of the sum
static int SubarraysDivByK(int[] nums, int k)
{
    var counts = new int[k];
    counts[0] = 1;
    int sum = 0, result = 0;

    foreach (var n in nums)
    {
        sum = ((sum + n) % k + k) % k;      // non-negative remainder: C# % can be < 0
        result += counts[sum]++;
    }
    return result;
}

// CONTINUOUS SUBARRAY SUM - is there a subarray of length >= 2 summing to a
// multiple of k? Same remainder map, but store the index and require a gap.
static bool CheckSubarraySum(int[] nums, int k)
{
    var firstIndex = new Dictionary<int, int> { [0] = -1 };
    int sum = 0;
    for (int i = 0; i < nums.Length; i++)
    {
        sum = ((sum + nums[i]) % k + k) % k;
        if (firstIndex.TryGetValue(sum, out int j)) { if (i - j >= 2) return true; }
        else firstIndex[sum] = i;
    }
    return false;
}

// WHY NOT A SLIDING WINDOW? Because these arrays contain NEGATIVE numbers, so
// the sum is not monotonic as the window grows and shrinking the left edge does
// not reliably reduce it. With all-positive values a sliding window works and is
// O(1) space - knowing which applies is the real question here.
from collections import defaultdict

def subarray_sum(nums: list[int], k: int) -> int:
    counts: dict[int, int] = defaultdict(int)
    counts[0] = 1                      # the empty prefix - do not omit
    total = result = 0
    for n in nums:
        total += n
        result += counts[total - k]
        counts[total] += 1
    return result

def longest_subarray_sum(nums: list[int], k: int) -> int:
    first: dict[int, int] = {0: -1}
    total = best = 0
    for i, n in enumerate(nums):
        total += n
        if total - k in first:
            best = max(best, i - first[total - k])
        if total not in first:         # FIRST occurrence only
            first[total] = i
    return best

def find_max_length(nums: list[int]) -> int:
    """Longest subarray with equal 0s and 1s."""
    first: dict[int, int] = {0: -1}
    total = best = 0
    for i, n in enumerate(nums):
        total += 1 if n else -1
        if total in first:
            best = max(best, i - first[total])
        else:
            first[total] = i
    return best

def subarrays_div_by_k(nums: list[int], k: int) -> int:
    counts = [0] * k
    counts[0] = 1
    total = result = 0
    for n in nums:
        total = (total + n) % k        # Python's % is always non-negative for k>0
        result += counts[total]
        counts[total] += 1
    return result

def check_subarray_sum(nums: list[int], k: int) -> bool:
    first: dict[int, int] = {0: -1}
    total = 0
    for i, n in enumerate(nums):
        total = (total + n) % k
        if total in first:
            if i - first[total] >= 2:
                return True
        else:
            first[total] = i
    return False

# Sliding window only works when all values are positive - then the sum is
# monotonic in the window size. With negatives you need the prefix map.
Interview angle

This is often the problem that decides the round, and there are three things to get right. Seed the map with {0: 1} (or {0: -1} for the longest variant) or subarrays starting at index 0 are missed. Store the first index and never overwrite it when you want the longest. And be ready for "why not a sliding window?" — because negative values break the monotonicity a window relies on, which is exactly why the map is needed.

Custom keys, GetHashCode and tuples

When the key is not a primitive, you need a type that hashes and compares correctly. Getting it wrong is a silent bug: entries go into the dictionary and can never be found again, which looks like a logic error rather than a hashing one.

The contract is short — equal objects must produce equal hashes, and the hash must not change while the object is a key. In practice: use a tuple, a record or a value tuple and let the compiler generate it.

// Value tuples hash and compare STRUCTURALLY - the default answer for a
// composite key, and no code to write
var visited = new HashSet<(int R, int C)>();
visited.Add((1, 2));
visited.Contains((1, 2));                // true

var memo = new Dictionary<(int Index, int Remaining), long>();

// Records: the same, with names
public readonly record struct Cell(int Row, int Col);
var grid = new Dictionary<Cell, char>();

// By hand, when you must. Three members, all of them:
public sealed class Point : IEquatable<Point>
{
    public int X { get; }                // IMMUTABLE - the hash must not change
    public int Y { get; }
    public Point(int x, int y) => (X, Y) = (x, y);

    public bool Equals(Point? other) => other is not null && X == other.X && Y == other.Y;
    public override bool Equals(object? o) => Equals(o as Point);
    public override int GetHashCode() => HashCode.Combine(X, Y);
}

Interview angle

The question is "what makes a good hash key?" and the complete answer is: immutable, with Equals and GetHashCode consistent, and the hash stable for as long as the object is a key. Have the failure mode concrete — mutate a hashed field and the entry is in the collection but unfindable. The detail most people miss in C# is that arrays compare by reference, so int[] as a dictionary key silently never matches; in Python the equivalent is that lists are unhashable, which at least fails loudly.

Practice
  • Use an int[] as a dictionary key in C# and watch every lookup miss.
  • Put a mutable object in a HashSet, change a hashed field, and prove it is lost.
  • Write a structural comparer for an array key and verify lookups now succeed.

Bucketing and top-k

"The k most frequent" has three solutions and knowing all three with their complexities is a strong answer: sort the counts O(n log n); a bounded min-heap of size k O(n log k); or bucket sort by frequency O(n), which works because a frequency cannot exceed n, so you can index buckets by count directly.

The bucket version is the one that impresses, because it beats the "obvious optimal" heap answer and the reason is a constraint on the data rather than a cleverer algorithm.

// 1. SORT - O(n log n). Fine, and the first thing to say.
static int[] TopKSort(int[] nums, int k) =>
    nums.GroupBy(n => n)
        .OrderByDescending(g => g.Count())
        .Take(k).Select(g => g.Key).ToArray();

// 2. BOUNDED MIN-HEAP - O(n log k). Better when k is much smaller than n,
//    and the right answer for a STREAM, where you cannot hold everything.
static int[] TopKHeap(int[] nums, int k)
{
    var counts = new Dictionary<int, int>();
    foreach (var n in nums) counts[n] = counts.GetValueOrDefault(n) + 1;

    var heap = new PriorityQueue<int, int>();        // min-heap on frequency
    foreach (var (value, count) in counts)
    {
        heap.Enqueue(value, count);
        if (heap.Count > k) heap.Dequeue();          // evict the least frequent
    }
    return heap.UnorderedItems.Select(x => x.Element).ToArray();
}

// 3. BUCKET BY FREQUENCY - O(n). A count cannot exceed n, so index by it.
static int[] TopKBucket(int[] nums, int k)
{
    var counts = new Dictionary<int, int>();
    foreach (var n in nums) counts[n] = counts.GetValueOrDefault(n) + 1;

    var buckets = new List<int>[nums.Length + 1];    // buckets[f] = values seen f times
    foreach (var (value, count) in counts)
        (buckets[count] ??= []).Add(value);

    var result = new List<int>(k);
    for (int f = buckets.Length - 1; f >= 1 && result.Count < k; f--)
        if (buckets[f] is { } bucket)
            foreach (var v in bucket)
            {
                result.Add(v);
                if (result.Count == k) break;
            }
    return result.ToArray();
}
// O(n) time, O(n) space. The insight is that frequency is BOUNDED by n, which
// is what makes a counting-sort-style bucket legal.

// The same bucketing idea elsewhere:

// SORT CHARACTERS BY FREQUENCY
static string FrequencySort(string s)
{
    var counts = new Dictionary<char, int>();
    foreach (var c in s) counts[c] = counts.GetValueOrDefault(c) + 1;

    var buckets = new List<char>[s.Length + 1];
    foreach (var (c, n) in counts) (buckets[n] ??= []).Add(c);

    var sb = new StringBuilder(s.Length);
    for (int f = buckets.Length - 1; f >= 1; f--)
        if (buckets[f] is { } bucket)
            foreach (var c in bucket) sb.Append(c, f);
    return sb.ToString();
}

// TOP K FREQUENT WORDS - ties broken alphabetically, so the heap comparer
// has to handle two keys. This is where a heap beats bucketing.
var heap = new PriorityQueue<string, (int Count, string Word)>(
    Comparer<(int Count, string Word)>.Create((a, b) =>
        a.Count != b.Count ? a.Count.CompareTo(b.Count)
                           : b.Word.CompareTo(a.Word)));   // reverse for the min-heap
from collections import Counter
import heapq

# 1. SORT / Counter.most_common - O(n log n), and one line
def top_k_sort(nums: list[int], k: int) -> list[int]:
    return [v for v, _ in Counter(nums).most_common(k)]

# 2. HEAP - O(n log k)
def top_k_heap(nums: list[int], k: int) -> list[int]:
    counts = Counter(nums)
    return heapq.nlargest(k, counts.keys(), key=counts.get)
# or by hand, which is what to write if asked not to use the library:
def top_k_heap_manual(nums: list[int], k: int) -> list[int]:
    counts = Counter(nums)
    heap: list[tuple[int, int]] = []
    for value, count in counts.items():
        heapq.heappush(heap, (count, value))
        if len(heap) > k:
            heapq.heappop(heap)          # evict the least frequent
    return [v for _, v in heap]

# 3. BUCKET - O(n)
def top_k_bucket(nums: list[int], k: int) -> list[int]:
    counts = Counter(nums)
    buckets: list[list[int]] = [[] for _ in range(len(nums) + 1)]
    for value, count in counts.items():
        buckets[count].append(value)

    result: list[int] = []
    for freq in range(len(buckets) - 1, 0, -1):
        for value in buckets[freq]:
            result.append(value)
            if len(result) == k:
                return result
    return result

def frequency_sort(s: str) -> str:
    counts = Counter(s)
    buckets: list[list[str]] = [[] for _ in range(len(s) + 1)]
    for c, n in counts.items():
        buckets[n].append(c)
    return "".join(c * freq
                   for freq in range(len(buckets) - 1, 0, -1)
                   for c in buckets[freq])

# Ties broken alphabetically - negate the count so one sort key does both
def top_k_words(words: list[str], k: int) -> list[str]:
    counts = Counter(words)
    return heapq.nsmallest(k, counts.keys(), key=lambda w: (-counts[w], w))

The LRU cache

A near-universal interview question, and the one that most clearly tests whether you can combine two structures. The requirement is O(1) Get and O(1) Put with eviction of the least recently used entry — which no single structure gives you.

The answer is a hash map plus a doubly linked list: the map gives O(1) lookup of the node, and because you hold the node, unlinking and re-inserting it at the front is O(1) too. A singly linked list does not work, because removing a node requires its predecessor.

// Dictionary<K, LinkedListNode<V>> + LinkedList<V>.
// The map finds the node in O(1); holding the node makes removal O(1).
public class LruCache<TKey, TValue> where TKey : notnull
{
    private readonly int _capacity;
    private readonly Dictionary<TKey, LinkedListNode<(TKey Key, TValue Value)>> _map;
    private readonly LinkedList<(TKey Key, TValue Value)> _order = new();
    //   front = most recently used, back = least recently used

    public LruCache(int capacity)
    {
        _capacity = capacity > 0 ? capacity : throw new ArgumentOutOfRangeException();
        _map = new Dictionary<TKey, LinkedListNode<(TKey, TValue)>>(capacity);
    }

    public bool TryGet(TKey key, out TValue value)
    {
        if (!_map.TryGetValue(key, out var node)) { value = default!; return false; }

        _order.Remove(node);                 // O(1) - we hold the node
        _order.AddFirst(node);               // O(1) - reuses the same node object
        value = node.Value.Value;
        return true;
    }

    public void Put(TKey key, TValue value)
    {
        if (_map.TryGetValue(key, out var existing))
        {
            existing.Value = (key, value);    // update in place
            _order.Remove(existing);
            _order.AddFirst(existing);
            return;
        }

        if (_map.Count == _capacity)
        {
            var lru = _order.Last!;          // evict from the BACK
            _order.RemoveLast();
            _map.Remove(lru.Value.Key);      // this is why the node stores the KEY
        }

        _map[key] = _order.AddFirst((key, value));
    }
}
// Storing the key inside the node is the detail people miss: on eviction you
// have the node and need to remove its entry from the map, which needs the key.

// Why not a singly linked list? Removing a node needs its PREDECESSOR, so you
// would scan - O(n). Why not a List<T>? Removing from the middle shifts - O(n).
// Why not SortedDictionary by timestamp? O(log n), not O(1).

// If asked to write the list by hand (common, to stop you using the BCL):
public class Node
{
    public int Key, Value;
    public Node? Prev, Next;
}
// Use sentinel head and tail nodes - they remove every null check from the
// unlink/insert code, which is where the bugs are.
private readonly Node _head = new(), _tail = new();
// constructor: _head.Next = _tail; _tail.Prev = _head;

private void Unlink(Node n) { n.Prev!.Next = n.Next; n.Next!.Prev = n.Prev; }
private void PushFront(Node n)
{
    n.Next = _head.Next; n.Prev = _head;
    _head.Next!.Prev = n; _head.Next = n;
}

// LFU is the harder follow-up: a frequency map plus a map from frequency to a
// list of nodes at that frequency, plus a running minimum frequency.
from collections import OrderedDict

# OrderedDict is the hash-map-plus-linked-list, already built. move_to_end and
# popitem are both O(1), which is exactly the pair you need.
class LRUCache:
    def __init__(self, capacity: int) -> None:
        self.capacity = capacity
        self.cache: OrderedDict[int, int] = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)             # O(1), to the MRU end
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)      # O(1), evict the LRU end

# Say that OrderedDict IS a dict plus a doubly linked list before you use it -
# otherwise it looks like you are dodging the question. Then offer the manual
# version if they want it.

class Node:
    __slots__ = ("key", "value", "prev", "next")
    def __init__(self, key=0, value=0):
        self.key, self.value = key, value
        self.prev = self.next = None

class LRUCacheManual:
    def __init__(self, capacity: int) -> None:
        self.capacity = capacity
        self.map: dict[int, Node] = {}
        self.head, self.tail = Node(), Node()    # sentinels: no null checks anywhere
        self.head.next, self.tail.prev = self.tail, self.head

    def _unlink(self, node: Node) -> None:
        node.prev.next, node.next.prev = node.next, node.prev

    def _push_front(self, node: Node) -> None:
        node.next, node.prev = self.head.next, self.head
        self.head.next.prev = self.head.next = node

    def get(self, key: int) -> int:
        node = self.map.get(key)
        if node is None:
            return -1
        self._unlink(node)
        self._push_front(node)
        return node.value

    def put(self, key: int, value: int) -> None:
        node = self.map.get(key)
        if node:
            node.value = value
            self._unlink(node)
            self._push_front(node)
            return
        if len(self.map) == self.capacity:
            lru = self.tail.prev
            self._unlink(lru)
            del self.map[lru.key]                # the node stores the key for this
        node = Node(key, value)
        self.map[key] = node
        self._push_front(node)

# functools.lru_cache is the decorator version, for memoising a function
from functools import lru_cache
@lru_cache(maxsize=128)
def expensive(n: int) -> int: ...

When hashing is the wrong answer

A dictionary is the right answer often enough that reaching for it becomes automatic — and interviewers probe exactly that. There are four situations where it is wrong, and naming one unprompted is a strong signal.

  • You need order. A hash map has none. "The k smallest", "the next larger key", "a range query" all want a sorted structure or a heap.
  • O(1) space is required. "Find the duplicate without modifying the array and in constant space" exists precisely to rule out a set — the answer is cycle detection or index marking.
  • The alphabet is tiny and fixed. For lowercase letters an int[26] beats a dictionary on both time and memory, with no hashing at all.
  • Keys share a prefix structure. Autocomplete and prefix search want a trie; a hash map can only answer exact-match questions.
// 1. YOU NEED ORDER ---------------------------------------------------------
// "the next key larger than x" - a dictionary cannot answer this

// 2. O(1) SPACE IS REQUIRED -------------------------------------------------
// "find the duplicate, do not modify the array, O(1) space" - the constraints
// are there to rule out both sorting and a HashSet. Floyd's cycle detection:
static int FindDuplicate(int[] nums)
{
    int slow = nums[0], fast = nums[0];
    do { slow = nums[slow]; fast = nums[nums[fast]]; } while (slow != fast);
    slow = nums[0];
    while (slow != fast) { slow = nums[slow]; fast = nums[fast]; }
    return slow;
}

// "find all numbers missing from 1..n, O(1) space" - mark by negating in place
static IList<int> FindDisappeared(int[] nums)
{
    foreach (var n in nums)
    {
        int i = Math.Abs(n) - 1;
        if (nums[i] > 0) nums[i] = -nums[i];     // the array IS the visited set
    }
    var missing = new List<int>();
    for (int i = 0; i < nums.Length; i++) if (nums[i] > 0) missing.Add(i + 1);
    return missing;
}

// "the single number, every other appears twice" - XOR, O(1) space
static int SingleNumber(int[] nums) => nums.Aggregate(0, (a, b) => a ^ b);

// 3. A TINY FIXED ALPHABET --------------------------------------------------
static bool IsAnagram(string a, string b)
{
    if (a.Length != b.Length) return false;
    var counts = new int[26];                    // no hashing, no allocation
    for (int i = 0; i < a.Length; i++) { counts[a[i] - 'a']++; counts[b[i] - 'a']--; }
    return counts.All(c => c == 0);
}

def find_duplicate(nums: list[int]) -> int:
    slow = fast = nums[0]
    while True:
        slow, fast = nums[slow], nums[nums[fast]]
        if slow == fast:
            break
    slow = nums[0]
    while slow != fast:
        slow, fast = nums[slow], nums[fast]
    return slow

def find_disappeared(nums: list[int]) -> list[int]:
    for n in nums:
        i = abs(n) - 1
        if nums[i] > 0:
            nums[i] = -nums[i]             # mark in place
    return [i + 1 for i, n in enumerate(nums) if n > 0]

def single_number(nums: list[int]) -> int:
    return reduce(xor, nums, 0)

def is_anagram(a: str, b: str) -> bool:
    if len(a) != len(b):
        return False
    counts = [0] * 26
    for ca, cb in zip(a, b):
        counts[ord(ca) - 97] += 1
        counts[ord(cb) - 97] -= 1
    return not any(counts)
