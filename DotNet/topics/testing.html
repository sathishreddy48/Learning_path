xUnit basics and Arrange-Act-Assert

A test class per unit, a [Fact] per behaviour, named so a failure reads like a sentence. Structure every test as Arrange / Act / Assert. xUnit creates a new instance of the class per test, so the constructor is your setup and IDisposable is teardown.

dotnet new xunit -n Shop.Tests
dotnet add Shop.Tests reference Shop
dotnet add Shop.Tests package FluentAssertions
dotnet add Shop.Tests package Moq
dotnet test --filter "FullyQualifiedName~CartTests"
public class CartTests : IDisposable
{
    private readonly Cart _cart = new();          // fresh per test (constructor = setup)

    [Fact]
    public void Add_SameSkuTwice_MergesIntoOneLine()
    {
        // Arrange
        var sku = "A1";

        // Act
        _cart.Add(sku, qty: 1);
        _cart.Add(sku, qty: 2);

        // Assert
        Assert.Single(_cart.Lines);
        Assert.Equal(3, _cart.Lines[0].Qty);
    }

    [Fact]
    public void Add_ZeroQty_Throws()
    {
        var ex = Assert.Throws<ArgumentOutOfRangeException>(() => _cart.Add("A1", 0));
        Assert.Equal("qty", ex.ParamName);
    }

    [Fact]
    public async Task Checkout_EmptyCart_ReturnsFailure()
    {
        var result = await _cart.CheckoutAsync();
        Assert.False(result.Ok);
    }

    public void Dispose() { /* teardown */ }
}

Data-driven tests: [Theory], [InlineData], [MemberData]

One test body, many inputs. [InlineData] for literals, [MemberData]/[ClassData] for objects. Each case shows up as its own test in the runner. pytest's @pytest.mark.parametrize is identical.

public class PasswordPolicyTests
{
    [Theory]
    [InlineData("short", false)]
    [InlineData("longenoughbutnodigits", false)]
    [InlineData("correct-horse-battery-9", true)]
    public void IsStrong_ReturnsExpected(string password, bool expected)
    {
        Assert.Equal(expected, PasswordPolicy.IsStrong(password));
    }

    public static IEnumerable<object[]> Discounts => new[]
    {
        new object[] { new Order(Total: 10m), 10m },
        new object[] { new Order(Total: 100m), 90m },
    };

    [Theory, MemberData(nameof(Discounts))]
    public void Price_AppliesDiscountOver50(Order order, decimal expected)
    {
        Assert.Equal(expected, Pricing.Price(order));
    }
}

Isolating with Moq

Mock the interfaces around the unit so the test runs fast and deterministically. Set up return values, verify calls, and — importantly — do not over-verify; assert outcomes, not implementation details. Python's unittest.mock is built in.

public class OrderServiceTests
{
    private readonly Mock<IOrderRepo> _repo = new();
    private readonly Mock<IEmailSender> _email = new();
    private readonly Mock<IClock> _clock = new();
    private readonly OrderService _sut;

    public OrderServiceTests()
    {
        _clock.Setup(c => c.UtcNow).Returns(new DateTime(2024, 1, 1));
        _sut = new OrderService(_repo.Object, _email.Object, _clock.Object);
    }

    [Fact]
    public async Task Place_ValidOrder_SavesAndEmails()
    {
        _repo.Setup(r => r.Add(It.IsAny<Order>())).ReturnsAsync(42);

        var id = await _sut.Place(new OrderDto("A1", 2, "a@x.com"));

        Assert.Equal(42, id);
        _repo.Verify(r => r.Add(It.Is<Order>(o => o.Sku == "A1" && o.Created == new DateTime(2024, 1, 1))), Times.Once);
        _email.Verify(e => e.Send("a@x.com", It.IsAny<string>(), It.IsAny<string>()), Times.Once);
    }

    [Fact]
    public async Task Place_WhenRepoThrows_DoesNotEmail()
    {
        _repo.Setup(r => r.Add(It.IsAny<Order>())).ThrowsAsync(new DbException("down"));

        await Assert.ThrowsAsync<DbException>(() => _sut.Place(new OrderDto("A1", 2, "a@x.com")));

        _email.VerifyNoOtherCalls();
    }
}
// Alternatives: NSubstitute (terser syntax), hand-written fakes for simple interfaces
from unittest.mock import Mock, AsyncMock, ANY
from datetime import datetime
import pytest
from shop.orders import OrderService, OrderDto, Order

@pytest.fixture
def deps():
    repo = Mock(); repo.add = AsyncMock(return_value=42)
    email = Mock(); email.send = AsyncMock()
    clock = Mock(); clock.utc_now.return_value = datetime(2024, 1, 1)
    return repo, email, clock

@pytest.mark.asyncio
async def test_place_valid_order_saves_and_emails(deps):
    repo, email, clock = deps
    sut = OrderService(repo, email, clock)

    id = await sut.place(OrderDto("A1", 2, "a@x.com"))

    assert id == 42
    repo.add.assert_awaited_once()
    order = repo.add.await_args.args[0]
    assert order.sku == "A1" and order.created == datetime(2024, 1, 1)
    email.send.assert_awaited_once_with("a@x.com", ANY, ANY)

@pytest.mark.asyncio
async def test_place_when_repo_raises_does_not_email(deps):
    repo, email, clock = deps
    repo.add.side_effect = RuntimeError("down")
    sut = OrderService(repo, email, clock)

    with pytest.raises(RuntimeError):
        await sut.place(OrderDto("A1", 2, "a@x.com"))

    email.send.assert_not_awaited()

Readable assertions with FluentAssertions

FluentAssertions turns Assert.Equal(expected, actual) into actual.Should().Be(expected) with far better failure messages, especially for collections and object graphs. pytest already rewrites assert to show detailed diffs.

using FluentAssertions;

result.Should().NotBeNull();
result.Total.Should().Be(30m).And.BePositive();
result.Lines.Should().HaveCount(2).And.OnlyContain(l => l.Qty > 0);
result.Lines.Select(l => l.Sku).Should().BeEquivalentTo(new[] { "A1", "B2" });   // order-insensitive
result.Should().BeEquivalentTo(expectedDto, o => o.Excluding(x => x.Created));  // deep structural compare
name.Should().StartWith("Ord").And.HaveLength(8);

Func<Task> act = () => service.Place(badOrder);
await act.Should().ThrowAsync<ValidationException>().WithMessage("*qty*");

// Events
using var monitor = ticker.Monitor();
ticker.Price = 120;
monitor.Should().Raise("PriceChanged");

// Execution time
Func<Task> slow = () => service.Report();
await slow.Should().CompleteWithinAsync(TimeSpan.FromSeconds(1));
import pytest, re

assert result is not None
assert result.total == 30 and result.total > 0

assert sorted(l.sku for l in result.lines) == ["A1", "B2"]
assert {k: v for k, v in vars(result).items() if k != "created"} == vars(expected_dto)

One integration test with WebApplicationFactory

Boot the real API in memory, swap the database for SQLite/Testcontainers, and call it with HttpClient. One or two of these per feature catch the wiring bugs unit tests cannot: routing, DI, serialization, middleware, auth. FastAPI's TestClient does the same.

// dotnet add package Microsoft.AspNetCore.Mvc.Testing
public class ApiFactory : WebApplicationFactory<Program>
{
    protected override void ConfigureWebHost(IWebHostBuilder b)
    {
        b.UseEnvironment("Testing");
        b.ConfigureServices(services =>
        {
            services.RemoveAll<DbContextOptions<AppDb>>();
            services.AddDbContext<AppDb>(o => o.UseSqlite("DataSource=:memory:"));
            services.AddScoped<IEmailSender, FakeEmailSender>();                 // no real email
            services.AddAuthentication("Test").AddScheme<AuthenticationSchemeOptions, TestAuthHandler>("Test", _ => { });
        });
    }
}

public class OrdersApiTests(ApiFactory factory) : IClassFixture<ApiFactory>
{
    private readonly HttpClient _client = factory.CreateClient();

    [Fact]
    public async Task Post_then_Get_roundtrips()
    {
        var create = await _client.PostAsJsonAsync("/orders", new { sku = "A1", qty = 2 });
        create.StatusCode.Should().Be(HttpStatusCode.Created);
        var location = create.Headers.Location!;

        var order = await _client.GetFromJsonAsync<OrderDto>(location);
        order!.Sku.Should().Be("A1");
    }

    [Fact]
    public async Task Post_invalid_returns_ProblemDetails()
    {
        var resp = await _client.PostAsJsonAsync("/orders", new { sku = "", qty = 0 });
        resp.StatusCode.Should().Be(HttpStatusCode.BadRequest);
        var problem = await resp.Content.ReadFromJsonAsync<ValidationProblemDetails>();
        problem!.Errors.Should().ContainKey("Qty");
    }
}

import pytest
from fastapi.testclient import TestClient
from shop.main import app
from shop.deps import get_db, get_email_sender

@pytest.fixture
def client():
    engine = create_engine("sqlite://", connect_args={"check_same_thread": False})
    Base.metadata.create_all(engine)
    TestSession = sessionmaker(bind=engine)

    def override_db():
        with TestSession() as s: yield s
    app.dependency_overrides[get_db] = override_db
    app.dependency_overrides[get_email_sender] = lambda: FakeEmailSender()
    with TestClient(app) as c:
        yield c
    app.dependency_overrides.clear()

def test_post_then_get_roundtrips(client):
    create = client.post("/orders", json={"sku": "A1", "qty": 2})
    assert create.status_code == 201
    location = create.headers["location"]

    order = client.get(location).json()
    assert order["sku"] == "A1"

def test_post_invalid_returns_problem(client):
    resp = client.post("/orders", json={"sku": "", "qty": 0})
    assert resp.status_code in (400, 422)
    assert any(e["loc"][-1] == "qty" for e in resp.json()["detail"])
