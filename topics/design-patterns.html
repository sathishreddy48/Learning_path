S — Single Responsibility

A class should have one reason to change. The smell: a class that talks to the database, formats output and sends email. Split by axis of change — persistence, presentation and notification evolve independently.

// ✗ Before: three reasons to change
public class InvoiceService
{
    public void Process(Invoice inv)
    {
        using var conn = new SqlConnection(cs); /* save */ ;
        var html = $"<h1>Invoice {inv.Id}</h1><p>{inv.Total:C}</p>";
        new SmtpClient().Send("billing@x.com", inv.Email, "Invoice", html);
    }
}

// ✓ After: each part has one job and can be tested/replaced alone
public interface IInvoiceRepository { Task Save(Invoice inv); }
public interface IInvoiceRenderer  { string Render(Invoice inv); }
public interface IEmailSender      { Task Send(string to, string subject, string body); }

public class InvoiceService(IInvoiceRepository repo, IInvoiceRenderer renderer, IEmailSender email)
{
    public async Task Process(Invoice inv)
    {
        await repo.Save(inv);
        await email.Send(inv.Email, "Invoice", renderer.Render(inv));
    }
}
# ✗ Before
class InvoiceService:
    def process(self, inv):
        db.save(inv)
        html = f"<h1>Invoice {inv.id}</h1><p>{inv.total:.2f}</p>"
        smtplib.SMTP("mail").sendmail("billing@x.com", inv.email, html)

# ✓ After — protocols describe the collaborators
from typing import Protocol

class InvoiceRepository(Protocol):
    def save(self, inv) -> None: ...
class InvoiceRenderer(Protocol):
    def render(self, inv) -> str: ...
class EmailSender(Protocol):
    def send(self, to: str, subject: str, body: str) -> None: ...

class InvoiceService:
    def __init__(self, repo: InvoiceRepository, renderer: InvoiceRenderer, email: EmailSender):
        self.repo, self.renderer, self.email = repo, renderer, email

    def process(self, inv) -> None:
        self.repo.save(inv)
        self.email.send(inv.email, "Invoice", self.renderer.render(inv))

O — Open/Closed

Open for extension, closed for modification: add behaviour by adding new types, not by editing a growing switch. The usual mechanism is polymorphism (Strategy) — or, in .NET, registering additional implementations in DI and iterating IEnumerable<T>.

// ✗ Every new discount edits this method
decimal Apply(Order o, string kind) => kind switch
{
    "seasonal" => o.Total * 0.9m,
    "loyalty"  => o.Total - 5,
    _ => o.Total
};

// ✓ New discount = new class; nothing existing changes
public interface IDiscount { bool AppliesTo(Order o); decimal Apply(decimal total); }

public class SeasonalDiscount : IDiscount
{
    public bool AppliesTo(Order o) => DateTime.UtcNow.Month == 12;
    public decimal Apply(decimal total) => total * 0.9m;
}
public class LoyaltyDiscount : IDiscount
{
    public bool AppliesTo(Order o) => o.Customer.Orders > 10;
    public decimal Apply(decimal total) => total - 5;
}

public class PricingService(IEnumerable<IDiscount> discounts)
{
    public decimal Price(Order o) =>
        discounts.Where(d => d.AppliesTo(o)).Aggregate(o.Total, (t, d) => d.Apply(t));
}
// builder.Services.AddScoped<IDiscount, SeasonalDiscount>();  AddScoped<IDiscount, LoyaltyDiscount>();
from typing import Protocol, Callable
from datetime import datetime

class Discount(Protocol):
    def applies_to(self, order) -> bool: ...
    def apply(self, total: float) -> float: ...

class SeasonalDiscount:
    def applies_to(self, order): return datetime.utcnow().month == 12
    def apply(self, total): return total * 0.9

class LoyaltyDiscount:
    def applies_to(self, order): return order.customer.orders > 10
    def apply(self, total): return total - 5

class PricingService:
    def __init__(self, discounts: list[Discount]): self.discounts = discounts
    def price(self, order) -> float:
        total = order.total
        for d in self.discounts:
            if d.applies_to(order): total = d.apply(total)
        return total

# Or a registry of plain functions — often enough in Python
DISCOUNTS: list[tuple[Callable, Callable]] = []
def discount(pred):
    def reg(fn): DISCOUNTS.append((pred, fn)); return fn
    return reg

@discount(lambda o: datetime.utcnow().month == 12)
def seasonal(total): return total * 0.9

L — Liskov Substitution

Subtypes must be usable wherever the base type is expected without surprising the caller: no strengthened preconditions, no weakened postconditions, no exceptions the base would not throw. The classic violation is Square : Rectangle; the .NET one is a read-only collection that throws on Add.

// ✗ Square breaks Rectangle's contract (setting Width changes Height)
public class Rectangle { public virtual int Width { get; set; } public virtual int Height { get; set; } public int Area => Width * Height; }
public class Square : Rectangle
{
    public override int Width  { set { base.Width = value; base.Height = value; } }
    public override int Height { set { base.Width = value; base.Height = value; } }
}
void Test(Rectangle r) { r.Width = 4; r.Height = 5; Debug.Assert(r.Area == 20); }   // fails for Square

// ✓ Model the real relationship: both are Shapes; neither is the other
public interface IShape { int Area { get; } }
public record Rectangle(int Width, int Height) : IShape { public int Area => Width * Height; }
public record Square(int Side) : IShape { public int Area => Side * Side; }

// ✗ .NET example: a subtype that throws where the base doesn't
public class ReadOnlyList<T> : List<T>
{
    public new void Add(T item) => throw new NotSupportedException();   // callers with List<T> break
}
// ✓ Expose IReadOnlyList<T> instead, so the type itself says what is allowed
# ✗ Square breaks Rectangle's contract
class Rectangle:
    def __init__(self, w, h): self._w, self._h = w, h
    @property
    def width(self): return self._w
    @width.setter
    def width(self, v): self._w = v
    @property
    def height(self): return self._h
    @height.setter
    def height(self, v): self._h = v
    @property
    def area(self): return self._w * self._h

class Square(Rectangle):
    @Rectangle.width.setter
    def width(self, v): self._w = self._h = v
    @Rectangle.height.setter
    def height(self, v): self._w = self._h = v

def test(r: Rectangle):
    r.width, r.height = 4, 5
    assert r.area == 20          # fails for Square

# ✓ Separate immutable shapes sharing a protocol
from dataclasses import dataclass
@dataclass(frozen=True)
class Rectangle:
    width: int; height: int
    @property
    def area(self): return self.width * self.height

@dataclass(frozen=True)
class Square:
    side: int
    @property
    def area(self): return self.side ** 2

# Duck typing makes LSP about behaviour: if it quacks like the base, it must quack correctly.

I — Interface Segregation

Clients should not depend on methods they do not use. Fat interfaces force NotImplementedException stubs and make mocking painful. Split into role interfaces; a class can implement several.

// ✗ One fat interface; a read-only report source must stub Save/Delete
public interface IRepository<T>
{
    Task<T?> Get(int id); Task<List<T>> All(); Task Save(T e); Task Delete(int id); Task<int> Count();
}

// ✓ Role interfaces; compose what you need
public interface IReadRepository<T>  { Task<T?> Get(int id); Task<List<T>> All(); }
public interface IWriteRepository<T> { Task Save(T e); Task Delete(int id); }

public class SqlRepository<T> : IReadRepository<T>, IWriteRepository<T> { /* full */ }
public class ReportSource : IReadRepository<Report> { /* only reads */ }

public class ReportController(IReadRepository<Report> reports) { }   // cannot accidentally write

// .NET BCL does this: IEnumerable → ICollection → IList; IReadOnlyList vs IList
from typing import Protocol, Optional

# ✓ Small protocols; implement any subset
class Reader(Protocol):
    def get(self, id: int) -> Optional[object]: ...
    def all(self) -> list: ...

class Writer(Protocol):
    def save(self, e) -> None: ...
    def delete(self, id: int) -> None: ...

class SqlRepository:                  # satisfies both structurally
    def get(self, id): ...
    def all(self): ...
    def save(self, e): ...
    def delete(self, id): ...

class ReportSource:                   # satisfies Reader only
    def get(self, id): ...
    def all(self): ...

class ReportController:
    def __init__(self, reports: Reader): self.reports = reports

# Python's ABCs are already segregated: collections.abc.Sized, Iterable, Container, Mapping…

D — Dependency Inversion

High-level policy should not depend on low-level details; both depend on abstractions, and the abstraction is owned by the high-level side. In practice: constructor-inject interfaces, let the composition root (DI container) choose implementations. This is what makes code testable.

// ✗ High-level OrderService constructs and depends on a concrete SmtpMailer
public class OrderService
{
    private readonly SmtpMailer _mailer = new("smtp.example.com");
    public void Place(Order o) { /* ... */ _mailer.Send(o.Email, "Thanks"); }
}

// ✓ Depend on an abstraction the domain owns; details plug in from outside
public interface INotifier { Task Notify(string to, string message); }     // defined next to OrderService

public class OrderService(INotifier notifier)
{
    public async Task Place(Order o) { /* ... */ await notifier.Notify(o.Email, "Thanks"); }
}

// Infrastructure implements it
public class SmtpNotifier(SmtpClient smtp) : INotifier { public Task Notify(string to, string m) => smtp.SendAsync(to, m); }
public class FakeNotifier : INotifier { public List<string> Sent = new(); public Task Notify(string to, string m) { Sent.Add(m); return Task.CompletedTask; } }

// Composition root
builder.Services.AddScoped<INotifier, SmtpNotifier>();
builder.Services.AddScoped<OrderService>();

// Test
var fake = new FakeNotifier();
await new OrderService(fake).Place(order);
Assert.Single(fake.Sent);

# ✓ Abstraction owned by the domain module

    def notify(self, to: str, message: str) -> None: ...

class OrderService:
    def __init__(self, notifier: Notifier): self.notifier = notifier
    def place(self, order):
        ...
        self.notifier.notify(order.email, "Thanks")

# Infrastructure implementations

    def __init__(self, smtp): self.smtp = smtp

class FakeNotifier:

# Composition root (a function; or a container like `dependency-injector`)

    return OrderService(SmtpNotifier(smtplib.SMTP(settings.smtp_host)))

# Test

Singleton

One instance, globally accessible. In modern .NET you almost never write one — register the type as a singleton in DI and inject it. If you must, use Lazy<T> for thread-safe lazy initialisation. In Python, a module-level instance is a singleton.

// Thread-safe lazy singleton (when DI is not available)
public sealed class Config
{
    private static readonly Lazy<Config> _instance = new(() => new Config());
    public static Config Instance => _instance.Value;
    private Config() { /* load once */ }
    public string ConnectionString { get; } = "...";
}
var cs = Config.Instance.ConnectionString;

// Preferred: the container owns the lifetime, the class stays testable
builder.Services.AddSingleton<IClock, SystemClock>();
public class OrderService(IClock clock) { }

// Pitfalls: hidden global state, hard to test, captive dependencies if the singleton
// takes scoped services (see Web API → DI).
# A module-level object is the idiomatic singleton (modules are imported once)
# config.py
class Config:
    def __init__(self): self.connection_string = "..."
config = Config()

# elsewhere: from config import config

# Lazy, cached factory
from functools import lru_cache
@lru_cache(maxsize=None)
def get_config() -> Config:
    return Config()

# Class-based (rarely needed)
class Singleton:
    _instance = None
    def __new__(cls, *a, **kw):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

Factory Method and Abstract Factory

Factories centralise object creation so callers depend on an interface, not a constructor. Factory Method: one method that returns the right subtype. Abstract Factory: a family of related objects (e.g. all the parts for one database provider). In .NET, IHttpClientFactory, ILoggerFactory and Func<T> injection are factories.

public interface IExporter { byte[] Export(Report r); }
public class CsvExporter : IExporter { public byte[] Export(Report r) => ...; }
public class PdfExporter : IExporter { public byte[] Export(Report r) => ...; }

// Factory method
public static class ExporterFactory
{
    public static IExporter Create(string format) => format.ToLowerInvariant() switch
    {
        "csv" => new CsvExporter(),
        "pdf" => new PdfExporter(),
        _ => throw new NotSupportedException(format)
    };
}

// DI-friendly factory: resolve by key
public interface IExporterFactory { IExporter For(string format); }
public class ExporterFactory(IServiceProvider sp) : IExporterFactory
{
    public IExporter For(string format) => sp.GetRequiredKeyedService<IExporter>(format);
}
builder.Services.AddKeyedTransient<IExporter, CsvExporter>("csv");
builder.Services.AddKeyedTransient<IExporter, PdfExporter>("pdf");

// Abstract factory: a family that must match
public interface IDbFactory { IDbConnection Connection(); IQueryBuilder Builder(); }
public class SqlServerFactory : IDbFactory { /* SqlConnection + T-SQL builder */ }
public class PostgresFactory  : IDbFactory { /* NpgsqlConnection + PG builder */ }
from typing import Protocol, Callable

class Exporter(Protocol):
    def export(self, report) -> bytes: ...

class CsvExporter:
    def export(self, report): ...
class PdfExporter:
    def export(self, report): ...

# Factory method as a dict of constructors
EXPORTERS: dict[str, Callable[[], Exporter]] = {"csv": CsvExporter, "pdf": PdfExporter}

def create_exporter(fmt: str) -> Exporter:
    try:
        return EXPORTERS[fmt.lower()]()
    except KeyError:
        raise ValueError(f"unsupported format {fmt}")

# Register new ones without touching the factory
def register(fmt: str):
    def deco(cls): EXPORTERS[fmt] = cls; return cls
    return deco

@register("xlsx")
class XlsxExporter:
    def export(self, report): ...

# Abstract factory: a family of related objects
class DbFactory(Protocol):
    def connection(self): ...
    def builder(self): ...
class PostgresFactory:
    def connection(self): return psycopg.connect(dsn)
    def builder(self): return PgQueryBuilder()

Builder

Builders assemble complex objects step by step with a fluent API and validate at Build(). .NET is full of them: WebApplicationBuilder, StringBuilder, ConfigurationBuilder, ModelBuilder. Python usually prefers keyword arguments and dataclasses; builders appear for query/DSL construction.

public class HttpRequestBuilder
{
    private string _method = "GET";
    private string? _url;
    private readonly Dictionary<string, string> _headers = new();
    private string? _body;

    public HttpRequestBuilder Method(string m) { _method = m; return this; }
    public HttpRequestBuilder Url(string u) { _url = u; return this; }
    public HttpRequestBuilder Header(string k, string v) { _headers[k] = v; return this; }
    public HttpRequestBuilder Json(object o) { _body = JsonSerializer.Serialize(o); _headers["Content-Type"] = "application/json"; return this; }

    public HttpRequestMessage Build()
    {
        if (_url is null) throw new InvalidOperationException("Url is required");
        var req = new HttpRequestMessage(new HttpMethod(_method), _url);
        foreach (var (k, v) in _headers) if (k != "Content-Type") req.Headers.Add(k, v);
        if (_body is not null) req.Content = new StringContent(_body, Encoding.UTF8, "application/json");
        return req;
    }
}

var req = new HttpRequestBuilder().Method("POST").Url("https://api/x").Header("X-Trace", "1").Json(new { a = 1 }).Build();

Adapter

Adapter wraps an existing class so it satisfies the interface your code expects — the way to bring a third-party SDK or legacy class under your own abstraction without changing either side.

// Your abstraction
public interface IPaymentGateway { Task<PaymentResult> Charge(decimal amount, string token); }

// Third-party SDK with a different shape
public class StripeClient { public Task<StripeCharge> CreateCharge(long cents, string source, string currency) => ...; }

// Adapter
public class StripeAdapter(StripeClient stripe) : IPaymentGateway
{
    public async Task<PaymentResult> Charge(decimal amount, string token)
    {
        var c = await stripe.CreateCharge((long)(amount * 100), token, "usd");
        return new PaymentResult(c.Id, c.Status == "succeeded");
    }
}
builder.Services.AddScoped<IPaymentGateway, StripeAdapter>();

// Also: wrapping static/legacy APIs so they can be mocked
public interface IFileSystem { string ReadAllText(string path); }
public class FileSystem : IFileSystem { public string ReadAllText(string p) => File.ReadAllText(p); }
from typing import Protocol
from dataclasses import dataclass

@dataclass
class PaymentResult: id: str; succeeded: bool

class PaymentGateway(Protocol):
    def charge(self, amount: float, token: str) -> PaymentResult: ...

class StripeClient:                                    # third-party shape
    def create_charge(self, cents: int, source: str, currency: str): ...

class StripeAdapter:
    def __init__(self, stripe: StripeClient): self.stripe = stripe
    def charge(self, amount, token):
        c = self.stripe.create_charge(int(amount * 100), token, "usd")
        return PaymentResult(c.id, c.status == "succeeded")

# Function adapter for a one-method interface
def adapt_stripe(stripe: StripeClient):

        c = stripe.create_charge(int(amount * 100), token, "usd")
        return PaymentResult(c.id, c.status == "succeeded")
    return charge

Decorator

Decorator wraps an object with the same interface to add behaviour — caching, logging, retries, authorisation — without touching the original. Chain as many as you like. Scrutor's .Decorate<>() makes registration easy. Python decorators are the function-level version of the same idea.

public interface IProductRepo { Task<Product?> Get(int id); }

public class SqlProductRepo(AppDb db) : IProductRepo
{
    public Task<Product?> Get(int id) => db.Products.FindAsync(id).AsTask();
}

// Caching decorator
public class CachedProductRepo(IProductRepo inner, IMemoryCache cache) : IProductRepo
{
    public Task<Product?> Get(int id) => cache.GetOrCreateAsync($"product:{id}", e =>
    {
        e.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5);
        return inner.Get(id);
    });
}

// Logging decorator
public class LoggedProductRepo(IProductRepo inner, ILogger<LoggedProductRepo> log) : IProductRepo
{
    public async Task<Product?> Get(int id)
    {
        var sw = Stopwatch.StartNew();
        try { return await inner.Get(id); }
        finally { log.LogInformation("Get({Id}) took {Ms}ms", id, sw.ElapsedMilliseconds); }
    }
}

// Registration (Scrutor): outermost last
builder.Services.AddScoped<IProductRepo, SqlProductRepo>();
builder.Services.Decorate<IProductRepo, CachedProductRepo>();
builder.Services.Decorate<IProductRepo, LoggedProductRepo>();   // Logged(Cached(Sql))
import functools, time, logging
from typing import Protocol

class ProductRepo(Protocol):
    def get(self, id: int): ...

class SqlProductRepo:
    def __init__(self, db): self.db = db
    def get(self, id): return self.db.get(Product, id)

# Object decorator (same interface, wraps inner)
class CachedProductRepo:
    def __init__(self, inner: ProductRepo): self.inner, self.cache = inner, {}
    def get(self, id):
        if id not in self.cache:
            self.cache[id] = self.inner.get(id)
        return self.cache[id]

class LoggedProductRepo:
    def __init__(self, inner: ProductRepo): self.inner = inner
    def get(self, id):
        t0 = time.perf_counter()
        try: return self.inner.get(id)
        finally: logging.info("get(%s) took %.1fms", id, (time.perf_counter() - t0) * 1000)

repo = LoggedProductRepo(CachedProductRepo(SqlProductRepo(db)))

# Function decorators are the same pattern at function level
def timed(fn):
    @functools.wraps(fn)
    def wrapper(*a, **kw):
        t0 = time.perf_counter()
        try: return fn(*a, **kw)
        finally: logging.info("%s took %.1fms", fn.__name__, (time.perf_counter() - t0) * 1000)
    return wrapper

@timed
@functools.lru_cache
def get_product(id): ...

Facade

A Facade gives a simple entry point over a messy subsystem. Application-service classes in layered architectures are facades: one PlaceOrder() that orchestrates inventory, payment and notification so controllers stay thin.

public class CheckoutFacade(IInventory inventory, IPaymentGateway payments, IOrderRepo orders, INotifier notifier)
{
    public async Task<OrderResult> PlaceOrder(Cart cart, string paymentToken)
    {
        if (!await inventory.Reserve(cart.Items)) return OrderResult.OutOfStock();
        var payment = await payments.Charge(cart.Total, paymentToken);
        if (!payment.Succeeded) { await inventory.Release(cart.Items); return OrderResult.PaymentFailed(); }
        var order = await orders.Create(cart, payment.Id);
        await notifier.Notify(cart.Email, $"Order {order.Id} confirmed");
        return OrderResult.Success(order.Id);
    }
}

// Controller stays a thin adapter to HTTP
app.MapPost("/checkout", async (CheckoutDto dto, CheckoutFacade checkout) =>
    (await checkout.PlaceOrder(dto.Cart, dto.Token)) switch
    {
        { Ok: true } r => Results.Created($"/orders/{r.OrderId}", r),
        var r => Results.UnprocessableEntity(r.Reason)
    });
class CheckoutFacade:
    def __init__(self, inventory, payments, orders, notifier):
        self.inventory, self.payments, self.orders, self.notifier = inventory, payments, orders, notifier

    def place_order(self, cart, payment_token):
        if not self.inventory.reserve(cart.items):
            return OrderResult.out_of_stock()
        payment = self.payments.charge(cart.total, payment_token)
        if not payment.succeeded:
            self.inventory.release(cart.items)
            return OrderResult.payment_failed()
        order = self.orders.create(cart, payment.id)
        self.notifier.notify(cart.email, f"Order {order.id} confirmed")
        return OrderResult.success(order.id)

@app.post("/checkout", status_code=201)
def checkout(dto: CheckoutDto, facade: CheckoutFacade = Depends(get_checkout)):
    result = facade.place_order(dto.cart, dto.token)
    if not result.ok:
        raise HTTPException(422, result.reason)
    return result

Strategy

Strategy makes an algorithm swappable at runtime: define an interface, implement variants, inject the one you want. In C# a Func<> parameter is a lightweight strategy (OrderBy(keySelector)). In Python, just pass a function.

public interface IShippingStrategy { decimal Cost(Order o); }
public class FlatRate : IShippingStrategy    { public decimal Cost(Order o) => 5m; }
public class ByWeight : IShippingStrategy    { public decimal Cost(Order o) => o.WeightKg * 1.2m; }
public class FreeOver50 : IShippingStrategy  { public decimal Cost(Order o) => o.Total > 50 ? 0 : 5m; }

public class ShippingCalculator(IShippingStrategy strategy)
{
    public decimal Calculate(Order o) => strategy.Cost(o);
}

// Choose at runtime
IShippingStrategy s = o.Country == "US" ? new FreeOver50() : new ByWeight();
var cost = new ShippingCalculator(s).Calculate(o);

// Lightweight: delegates
Func<Order, decimal> flat = _ => 5m;
decimal Calculate(Order o, Func<Order, decimal> strategy) => strategy(o);

// Also Comparer<T> for sort strategies
var byName = Comparer<Person>.Create((a, b) => string.Compare(a.Name, b.Name, StringComparison.Ordinal));
people.Sort(byName);
from typing import Callable

# Strategies are just callables

Observer

Observer lets many subscribers react to a change without the subject knowing them. C# has it built in as event, plus IObservable<T> (Rx) for push streams. Beware of leaks: subscribers must unsubscribe or they keep the subject alive.

public class StockTicker
{
    public event EventHandler<PriceChangedEventArgs>? PriceChanged;

    private decimal _price;
    public decimal Price
    {
        get => _price;
        set
        {
            if (value == _price) return;
            var old = _price; _price = value;
            PriceChanged?.Invoke(this, new PriceChangedEventArgs(old, value));   // ?. handles no subscribers
        }
    }
}
public record PriceChangedEventArgs(decimal Old, decimal New);

from typing import Callable
from dataclasses import dataclass

@dataclass
class PriceChanged: old: float; new: float

class StockTicker:
    def __init__(self):
        self._price = 0.0
        self._subscribers: list[Callable[[PriceChanged], None]] = []

    def subscribe(self, fn):   self._subscribers.append(fn); return lambda: self._subscribers.remove(fn)

    @property
    def price(self): return self._price
    @price.setter
    def price(self, value):
        if value == self._price: return
        old, self._price = self._price, value
        for fn in list(self._subscribers):
            fn(PriceChanged(old, value))

Command

Command turns a request into an object: it can be queued, logged, retried and undone. Undo/redo stacks, job queues and CQRS commands all use it. In .NET, MediatR's IRequest handlers are commands with a dispatcher.

public interface ICommand { void Execute(); void Undo(); }

public class MoveCommand(Shape shape, int dx, int dy) : ICommand
{
    public void Execute() => shape.Move(dx, dy);
    public void Undo() => shape.Move(-dx, -dy);
}

public class CommandHistory
{
    private readonly Stack<ICommand> _done = new(), _undone = new();
    public void Run(ICommand c) { c.Execute(); _done.Push(c); _undone.Clear(); }
    public void Undo() { if (_done.TryPop(out var c)) { c.Undo(); _undone.Push(c); } }
    public void Redo() { if (_undone.TryPop(out var c)) { c.Execute(); _done.Push(c); } }
}

history.Undo();

from typing import Protocol

class Command(Protocol):
    def execute(self) -> None: ...
    def undo(self) -> None: ...

class MoveCommand:
    def __init__(self, shape, dx, dy): self.shape, self.dx, self.dy = shape, dx, dy
    def execute(self): self.shape.move(self.dx, self.dy)
    def undo(self): self.shape.move(-self.dx, -self.dy)

class CommandHistory:
    def __init__(self): self._done, self._undone = [], []
    def run(self, c: Command): c.execute(); self._done.append(c); self._undone.clear()
    def undo(self):
        if self._done: c = self._done.pop(); c.undo(); self._undone.append(c)
    def redo(self):
        if self._undone: c = self._undone.pop(); c.execute(); self._done.append(c)

history = CommandHistory()
history.run(MoveCommand(shape, 10, 0))
history.undo()

Mediator routes messages between components so they do not reference each other. In .NET the popular form is MediatR: controllers send a request object, exactly one handler processes it, and pipeline behaviours (validation, logging, transactions) wrap every handler. It is the usual backbone for CQRS.

public class ValidationBehavior<TReq, TRes>(IEnumerable<IValidator<TReq>> validators) : IPipelineBehavior<TReq, TRes>
{
    public async Task<TRes> Handle(TReq req, RequestHandlerDelegate<TRes> next, CancellationToken ct)
    {
        var failures = validators.Select(v => v.Validate(req)).SelectMany(r => r.Errors).ToList();
        if (failures.Count > 0) throw new ValidationException(failures);
        return await next();
    }
}

from typing import Callable, Any
from dataclasses import dataclass

@dataclass
class CreateOrder: sku: str; qty: int

class Mediator:
    def __init__(self):
        self._handlers: dict[type, Callable] = {}
        self._behaviors: list[Callable] = []
        self._listeners: dict[type, list[Callable]] = {}

    def register(self, req_type, handler): self._handlers[req_type] = handler
    def add_behavior(self, b): self._behaviors.append(b)
    def subscribe(self, event_type, fn): self._listeners.setdefault(event_type, []).append(fn)

    def send(self, request) -> Any:
        handler = self._handlers[type(request)]
        call = lambda: handler(request)
        for b in reversed(self._behaviors):          # wrap like middleware
            call = (lambda nxt, beh: (lambda: beh(request, nxt)))(call, b)
        return call()

    def publish(self, event):
        for fn in self._listeners.get(type(event), []): fn(event)

def validation_behavior(request, next_):
    if getattr(request, "qty", 1) <= 0: raise ValueError("qty")
    return next_()

def create_order_handler(cmd: CreateOrder) -> int:
    order = Order(sku=cmd.sku, qty=cmd.qty); db.add(order); db.commit(); return order.id

mediator = Mediator()
mediator.add_behavior(validation_behavior)
mediator.register(CreateOrder, create_order_handler)

@app.post("/orders")
def create(cmd: CreateOrder): return mediator.send(cmd)

Dependency Injection and IoC containers

DI is the mechanism for DIP: dependencies are passed in, never new-ed inside. The container is the composition root that knows how to build the graph and manage lifetimes. Rules: constructor injection, depend on interfaces, no service locator, keep constructors cheap, register once. Python has no built-in container; FastAPI's Depends and plain factory functions cover most needs.

Repository and Unit of Work

Repository hides data access behind a collection-like interface; Unit of Work groups changes into one commit. EF Core's DbSet and DbContext already are these, so a generic repository over EF adds little. A repository earns its place when it hides complex queries behind intention-revealing names, or when you need to swap the store in tests.

// Intention-revealing repository (not a generic CRUD wrapper)
public interface IOrderRepository
{
    Task<Order?> GetWithLines(int id, CancellationToken ct = default);
    Task<IReadOnlyList<Order>> OverdueFor(int customerId, CancellationToken ct = default);
    void Add(Order o);
}

public class EfOrderRepository(AppDb db) : IOrderRepository
{
    public Task<Order?> GetWithLines(int id, CancellationToken ct) =>
        db.Orders.Include(o => o.Lines).FirstOrDefaultAsync(o => o.Id == id, ct);
    public async Task<IReadOnlyList<Order>> OverdueFor(int customerId, CancellationToken ct) =>
        await db.Orders.Where(o => o.CustomerId == customerId && o.Due < DateTime.UtcNow && !o.Paid).ToListAsync(ct);
    public void Add(Order o) => db.Orders.Add(o);
}

// Unit of Work: commit once per use case
public interface IUnitOfWork { Task<int> Commit(CancellationToken ct = default); }
public class EfUnitOfWork(AppDb db) : IUnitOfWork { public Task<int> Commit(CancellationToken ct) => db.SaveChangesAsync(ct); }

public class PlaceOrderHandler(IOrderRepository orders, IInventoryRepository inventory, IUnitOfWork uow)
{
    public async Task Handle(PlaceOrder cmd, CancellationToken ct)
    {
        var order = Order.Create(cmd);
        orders.Add(order);
        await inventory.Reserve(order.Lines, ct);
        await uow.Commit(ct);                       // one transaction for both repos (same DbContext)
    }
}
// Both repos and the UoW share the scoped AppDb — that is what makes the commit atomic.
from typing import Protocol, Optional
from sqlalchemy import select
from sqlalchemy.orm import selectinload, Session

class OrderRepository(Protocol):
    def get_with_lines(self, id: int) -> Optional[Order]: ...
    def overdue_for(self, customer_id: int) -> list[Order]: ...
    def add(self, order: Order) -> None: ...

class SqlOrderRepository:
    def __init__(self, session: Session): self.s = session
    def get_with_lines(self, id):
        return self.s.scalar(select(Order).options(selectinload(Order.lines)).where(Order.id == id))
    def overdue_for(self, customer_id):
        return self.s.scalars(select(Order).where(Order.customer_id == customer_id, Order.due < now(), Order.paid == False)).all()
    def add(self, order): self.s.add(order)

# Unit of Work as a context manager owning the session
class UnitOfWork:
    def __init__(self, session_factory): self._sf = session_factory
    def __enter__(self):
        self.session = self._sf()
        self.orders = SqlOrderRepository(self.session)
        self.inventory = SqlInventoryRepository(self.session)
        return self
    def __exit__(self, exc_type, exc, tb):
        if exc_type: self.session.rollback()
        self.session.close()
    def commit(self): self.session.commit()

def place_order(cmd, uow: UnitOfWork):
    with uow:
        order = Order.create(cmd)
        uow.orders.add(order)
        uow.inventory.reserve(order.lines)
        uow.commit()                     # one transaction
Interview angle

“Do you need a repository over EF Core?” Balanced answer: EF is already a repository/UoW; add your own only for intention-revealing queries, to keep EF out of the domain, or for test seams — and never expose IQueryable from it.

CQRS — an introduction

Command Query Responsibility Segregation: separate the write model (commands, validation, domain rules) from the read model (queries, DTOs, often denormalised). It starts as two folders and two handler types in one database; only some systems go on to separate stores and event sourcing. It pairs naturally with Mediator and Dapper-for-reads.

// Write side: command + handler operating on the domain model via EF
public record ShipOrder(int OrderId) : IRequest;
public class ShipOrderHandler(AppDb db, IPublisher events) : IRequestHandler<ShipOrder>
{
    public async Task Handle(ShipOrder cmd, CancellationToken ct)
    {
        var order = await db.Orders.Include(o => o.Lines).FirstAsync(o => o.Id == cmd.OrderId, ct);
        order.Ship();                                   // domain rule: throws if unpaid
        await db.SaveChangesAsync(ct);
        await events.Publish(new OrderShipped(order.Id), ct);
    }
}

// Read side: query + handler returning a DTO, optimised SQL, no tracking
public record OrderSummary(int Id, string Customer, decimal Total, string Status);
public record GetOrderSummaries(int CustomerId) : IRequest<IReadOnlyList<OrderSummary>>;
public class GetOrderSummariesHandler(IDbConnection conn) : IRequestHandler<GetOrderSummaries, IReadOnlyList<OrderSummary>>
{
    public async Task<IReadOnlyList<OrderSummary>> Handle(GetOrderSummaries q, CancellationToken ct) =>
        (await conn.QueryAsync<OrderSummary>(
            "SELECT o.Id, c.Name AS Customer, o.Total, o.Status FROM Orders o JOIN Customers c ON c.Id = o.CustomerId WHERE o.CustomerId = @CustomerId",
            new { q.CustomerId })).ToList();
}

from dataclasses import dataclass
from sqlalchemy import text

# Write side
@dataclass
class ShipOrder: order_id: int

def ship_order(cmd: ShipOrder, session, publish):
    order = session.get(Order, cmd.order_id)
    order.ship()                             # domain rule
    session.commit()
    publish(OrderShipped(order.id))

# Read side: DTO + raw SQL, no ORM objects
@dataclass
class OrderSummary: id: int; customer: str; total: float; status: str

@dataclass
class GetOrderSummaries: customer_id: int

def get_order_summaries(q: GetOrderSummaries, conn) -> list[OrderSummary]:
    rows = conn.execute(text(
        "SELECT o.id, c.name AS customer, o.total, o.status FROM orders o "
        "JOIN customers c ON c.id = o.customer_id WHERE o.customer_id = :cid"), {"cid": q.customer_id}).mappings()
    return [OrderSummary(**r) for r in rows]

Interview angle

CQRS is not event sourcing; you can have one without the other. Be able to say what problem it solves (different shapes and scaling needs for reads vs writes) and what it costs (eventual consistency, duplication).
