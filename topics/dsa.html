Arrays and strings

Know the basics cold: in-place reversal, building results with StringBuilder/list-join, Span<T> for slicing without allocation, and Array.Sort + custom comparers. Canonical: reverse words in a string; check anagram.

// Reverse a string in place (char array)
static string Reverse(string s)
{
    var a = s.ToCharArray();
    Array.Reverse(a);
    return new string(a);
}

// Reverse the words: "the sky is blue" → "blue is sky the"
static string ReverseWords(string s) =>
    string.Join(' ', s.Split(' ', StringSplitOptions.RemoveEmptyEntries).Reverse());

// Anagram check: O(n) with a fixed-size count array
static bool IsAnagram(string a, string b)
{
    if (a.Length != b.Length) return false;
    var counts = new int[26];
    for (int i = 0; i < a.Length; i++) { counts[a[i] - 'a']++; counts[b[i] - 'a']--; }
    return counts.All(c => c == 0);
}

// Rotate array right by k in place: reverse all, reverse first k, reverse rest
static void Rotate(int[] nums, int k)
{
    k %= nums.Length;
    Array.Reverse(nums); Array.Reverse(nums, 0, k); Array.Reverse(nums, k, nums.Length - k);
}

Dictionary frequency map

The single most common interview structure: count things in a Dictionary<TKey,int> then answer a question about the counts. Canonical: two-sum, first unique character, top-k frequent elements.

// Two Sum: O(n) with a value→index map
static int[] TwoSum(int[] nums, int target)
{
    var seen = new Dictionary<int, int>();
    for (int i = 0; i < nums.Length; i++)
    {
        if (seen.TryGetValue(target - nums[i], out var j)) return new[] { j, i };
        seen[nums[i]] = i;
    }
    return Array.Empty<int>();
}

// First unique character
static int FirstUnique(string s)
{
    var counts = new Dictionary<char, int>();
    foreach (var c in s) counts[c] = counts.GetValueOrDefault(c) + 1;
    for (int i = 0; i < s.Length; i++) if (counts[s[i]] == 1) return i;
    return -1;
}

// Top-k frequent: count, then sort or heap
static int[] TopK(int[] nums, int k) =>
    nums.GroupBy(n => n)
        .OrderByDescending(g => g.Count())
        .Take(k)
        .Select(g => g.Key)
        .ToArray();

Two pointers

Two indices moving toward each other (sorted input) or in the same direction (fast/slow). Turns O(n²) pair searches into O(n). Canonical: two-sum on sorted array, container with most water, remove duplicates in place.

// Pair with target sum in a sorted array
static (int, int)? PairSum(int[] a, int target)
{
    int l = 0, r = a.Length - 1;
    while (l < r)
    {
        int s = a[l] + a[r];
        if (s == target) return (l, r);
        if (s < target) l++; else r--;
    }
    return null;
}

// Container with most water
static int MaxArea(int[] h)
{
    int l = 0, r = h.Length - 1, best = 0;
    while (l < r)
    {
        best = Math.Max(best, Math.Min(h[l], h[r]) * (r - l));
        if (h[l] < h[r]) l++; else r--;
    }
    return best;
}

// Remove duplicates from sorted array in place (slow/fast)
static int RemoveDuplicates(int[] a)
{
    if (a.Length == 0) return 0;
    int w = 1;
    for (int r = 1; r < a.Length; r++) if (a[r] != a[w - 1]) a[w++] = a[r];
    return w;
}

// 3Sum: sort + fix one + two pointers, skipping duplicates
static IList<IList<int>> ThreeSum(int[] nums)
{
    Array.Sort(nums); var res = new List<IList<int>>();
    for (int i = 0; i < nums.Length - 2; i++)
    {
        if (i > 0 && nums[i] == nums[i - 1]) continue;
        int l = i + 1, r = nums.Length - 1;
        while (l < r)
        {
            int s = nums[i] + nums[l] + nums[r];
            if (s == 0) { res.Add(new[] { nums[i], nums[l], nums[r] }); l++; r--; while (l < r && nums[l] == nums[l - 1]) l++; }
            else if (s < 0) l++; else r--;
        }
    }
    return res;
}
def pair_sum(a: list[int], target: int):
    l, r = 0, len(a) - 1
    while l < r:
        s = a[l] + a[r]
        if s == target: return (l, r)
        if s < target: l += 1
        else: r -= 1
    return None

def max_area(h: list[int]) -> int:
    l, r, best = 0, len(h) - 1, 0
    while l < r:
        best = max(best, min(h[l], h[r]) * (r - l))
        if h[l] < h[r]: l += 1
        else: r -= 1
    return best

def remove_duplicates(a: list[int]) -> int:
    if not a: return 0
    w = 1
    for r in range(1, len(a)):
        if a[r] != a[w - 1]:
            a[w] = a[r]; w += 1
    return w

def three_sum(nums: list[int]) -> list[list[int]]:
    nums.sort(); res = []
    for i in range(len(nums) - 2):
        if i and nums[i] == nums[i - 1]: continue
        l, r = i + 1, len(nums) - 1
        while l < r:
            s = nums[i] + nums[l] + nums[r]
            if s == 0:
                res.append([nums[i], nums[l], nums[r]]); l += 1; r -= 1
                while l < r and nums[l] == nums[l - 1]: l += 1
            elif s < 0: l += 1
            else: r -= 1
    return res

Sliding window

A window [l, r] over a sequence, expanded with r and shrunk with l while a condition holds; state kept in a counter or running sum. Canonical: longest substring without repeats, minimum window substring, max sum subarray of size k.

// Longest substring without repeating characters
static int LengthOfLongestSubstring(string s)
{
    var last = new Dictionary<char, int>();
    int best = 0, l = 0;
    for (int r = 0; r < s.Length; r++)
    {
        if (last.TryGetValue(s[r], out var idx) && idx >= l) l = idx + 1;   // jump past the repeat
        last[s[r]] = r;
        best = Math.Max(best, r - l + 1);
    }
    return best;
}

// Fixed window: max sum of k consecutive
static int MaxSumK(int[] a, int k)
{
    int sum = a.Take(k).Sum(), best = sum;
    for (int r = k; r < a.Length; r++) { sum += a[r] - a[r - k]; best = Math.Max(best, sum); }
    return best;
}

// Minimum window substring containing all chars of t
static string MinWindow(string s, string t)
{
    var need = new Dictionary<char, int>();
    foreach (var c in t) need[c] = need.GetValueOrDefault(c) + 1;
    int missing = t.Length, l = 0, bestL = 0, bestLen = int.MaxValue;
    for (int r = 0; r < s.Length; r++)
    {
        if (need.TryGetValue(s[r], out var n)) { if (n > 0) missing--; need[s[r]] = n - 1; }
        while (missing == 0)
        {
            if (r - l + 1 < bestLen) { bestLen = r - l + 1; bestL = l; }
            if (need.TryGetValue(s[l], out var m)) { need[s[l]] = m + 1; if (m + 1 > 0) missing++; }
            l++;
        }
    }
    return bestLen == int.MaxValue ? "" : s.Substring(bestL, bestLen);
}
def length_of_longest_substring(s: str) -> int:
    last: dict[str, int] = {}
    best = l = 0
    for r, c in enumerate(s):
        if c in last and last[c] >= l:
            l = last[c] + 1
        last[c] = r
        best = max(best, r - l + 1)
    return best

def max_sum_k(a: list[int], k: int) -> int:
    s = best = sum(a[:k])
    for r in range(k, len(a)):
        s += a[r] - a[r - k]; best = max(best, s)
    return best

from collections import Counter
def min_window(s: str, t: str) -> str:
    need = Counter(t); missing = len(t)
    l = 0; best = (0, float("inf"))
    for r, c in enumerate(s):
        if need[c] > 0: missing -= 1
        need[c] -= 1
        while missing == 0:
            if r - l + 1 < best[1] - best[0]: best = (l, r + 1)
            need[s[l]] += 1
            if need[s[l]] > 0: missing += 1
            l += 1
    return "" if best[1] == float("inf") else s[best[0]:best[1]]

Stack and queue patterns

Stack: matching/nesting, undo, monotonic stacks for “next greater element”. Queue: BFS, sliding window with deque, rate limiters. Canonical: valid parentheses, daily temperatures, min-stack.

// Valid parentheses
static bool IsValid(string s)
{
    var pairs = new Dictionary<char, char> { [')'] = '(', [']'] = '[', ['}'] = '{' };
    var st = new Stack<char>();
    foreach (var c in s)
    {
        if (pairs.ContainsValue(c)) st.Push(c);
        else if (pairs.TryGetValue(c, out var open) && (st.Count == 0 || st.Pop() != open)) return false;
    }
    return st.Count == 0;
}

// Monotonic stack: days until a warmer temperature
static int[] DailyTemperatures(int[] t)
{
    var res = new int[t.Length]; var st = new Stack<int>();   // indices, decreasing temps
    for (int i = 0; i < t.Length; i++)
    {
        while (st.Count > 0 && t[st.Peek()] < t[i]) { var j = st.Pop(); res[j] = i - j; }
        st.Push(i);
    }
    return res;
}

// Min-stack: O(1) min via a parallel stack
public class MinStack
{
    private readonly Stack<int> _s = new(), _min = new();
    public void Push(int x) { _s.Push(x); _min.Push(_min.Count == 0 ? x : Math.Min(x, _min.Peek())); }
    public int Pop() { _min.Pop(); return _s.Pop(); }
    public int Min => _min.Peek();
}

// Queue via two stacks (amortised O(1))
public class QueueViaStacks<T>
{
    private readonly Stack<T> _in = new(), _out = new();
    public void Enqueue(T x) => _in.Push(x);
    public T Dequeue() { if (_out.Count == 0) while (_in.Count > 0) _out.Push(_in.Pop()); return _out.Pop(); }
}
def is_valid(s: str) -> bool:
    pairs = {")": "(", "]": "[", "}": "{"}
    st = []
    for c in s:
        if c in pairs.values(): st.append(c)
        elif c in pairs and (not st or st.pop() != pairs[c]): return False
    return not st

def daily_temperatures(t: list[int]) -> list[int]:
    res = [0] * len(t); st = []            # indices with decreasing temps
    for i, temp in enumerate(t):
        while st and t[st[-1]] < temp:
            j = st.pop(); res[j] = i - j
        st.append(i)
    return res

class MinStack:
    def __init__(self): self._s, self._min = [], []
    def push(self, x):
        self._s.append(x); self._min.append(x if not self._min else min(x, self._min[-1]))
    def pop(self): self._min.pop(); return self._s.pop()
    @property
    def min(self): return self._min[-1]

class QueueViaStacks:
    def __init__(self): self._in, self._out = [], []
    def enqueue(self, x): self._in.append(x)
    def dequeue(self):
        if not self._out:
            while self._in: self._out.append(self._in.pop())
        return self._out.pop()

Linked lists

Pointer manipulation under pressure: reverse, detect cycle (Floyd), find middle, merge sorted lists, remove nth from end. Use a dummy head to avoid edge cases. .NET has LinkedList<T> for real code; interviews want the hand-rolled node.

public class ListNode(int val, ListNode? next = null) { public int Val = val; public ListNode? Next = next; }

// Reverse iteratively
static ListNode? Reverse(ListNode? head)
{
    ListNode? prev = null;
    while (head is not null) { var next = head.Next; head.Next = prev; prev = head; head = next; }
    return prev;
}

// Cycle detection (fast/slow)
static bool HasCycle(ListNode? head)
{
    var slow = head; var fast = head;
    while (fast?.Next is not null) { slow = slow!.Next; fast = fast.Next.Next; if (slow == fast) return true; }
    return false;
}

// Middle node
static ListNode? Middle(ListNode? head)
{
    var slow = head; var fast = head;
    while (fast?.Next is not null) { slow = slow!.Next; fast = fast.Next.Next; }
    return slow;
}

// Merge two sorted lists with a dummy head
static ListNode? Merge(ListNode? a, ListNode? b)
{
    var dummy = new ListNode(0); var tail = dummy;
    while (a is not null && b is not null)
    {
        if (a.Val <= b.Val) { tail.Next = a; a = a.Next; } else { tail.Next = b; b = b.Next; }
        tail = tail.Next;
    }
    tail.Next = a ?? b;
    return dummy.Next;
}

// Remove nth from end: two pointers n apart
static ListNode? RemoveNthFromEnd(ListNode? head, int n)
{
    var dummy = new ListNode(0, head); ListNode? fast = dummy, slow = dummy;
    for (int i = 0; i <= n; i++) fast = fast!.Next;
    while (fast is not null) { fast = fast.Next; slow = slow!.Next; }
    slow!.Next = slow.Next!.Next;
    return dummy.Next;
}

        if slow is fast: return True
    return False

def middle(head):
    slow = fast = head
    while fast and fast.next:
        slow, fast = slow.next, fast.next.next
    return slow

def merge(a, b):
    dummy = tail = ListNode(0)
    while a and b:
        if a.val <= b.val: tail.next, a = a, a.next
        else: tail.next, b = b, b.next
        tail = tail.next
    tail.next = a or b
    return dummy.next

def remove_nth_from_end(head, n):
    dummy = ListNode(0, head); fast = slow = dummy
    for _ in range(n + 1): fast = fast.next
    while fast: fast, slow = fast.next, slow.next
    slow.next = slow.next.next
    return dummy.next

Binary search

Halve the search space each step on anything monotonic — sorted arrays, but also “smallest x such that f(x) is true” (answer-space search). Off-by-one bugs are the whole difficulty; settle on one template. .NET has Array.BinarySearch and Python has bisect for the plain case.

// Template: lower bound (first index where a[i] >= target)
static int LowerBound(int[] a, int target)
{
    int lo = 0, hi = a.Length;              // hi is exclusive
    while (lo < hi)
    {
        int mid = lo + (hi - lo) / 2;         // avoids overflow
        if (a[mid] < target) lo = mid + 1; else hi = mid;
    }
    return lo;                              // == a.Length if none
}

// Exact search
static int Search(int[] a, int target)
{
    int i = LowerBound(a, target);
    return i < a.Length && a[i] == target ? i : -1;
}

// Rotated sorted array
static int SearchRotated(int[] a, int t)
{
    int lo = 0, hi = a.Length - 1;
    while (lo <= hi)
    {
        int mid = (lo + hi) / 2;
        if (a[mid] == t) return mid;
        if (a[lo] <= a[mid]) { if (a[lo] <= t && t < a[mid]) hi = mid - 1; else lo = mid + 1; }
        else                 { if (a[mid] < t && t <= a[hi]) lo = mid + 1; else hi = mid - 1; }
    }
    return -1;
}

// Answer-space search: minimum eating speed to finish piles in h hours
static int MinEatingSpeed(int[] piles, int h)
{
    int lo = 1, hi = piles.Max();
    while (lo < hi)
    {
        int k = lo + (hi - lo) / 2;
        long hours = piles.Sum(p => (long)Math.Ceiling(p / (double)k));
        if (hours <= h) hi = k; else lo = k + 1;
    }
    return lo;
}

Sorting and comparers

Array.Sort/List.Sort are introsort (unstable, O(n log n)); LINQ OrderBy is stable. Write comparers with Comparison<T> or IComparer<T>. Know one O(n log n) algorithm well enough to write it (merge sort) and when counting/bucket sort wins.

// Custom ordering
people.Sort((a, b) => a.Age != b.Age ? a.Age.CompareTo(b.Age) : string.CompareOrdinal(a.Name, b.Name));
var sorted = people.OrderBy(p => p.Age).ThenBy(p => p.Name).ToList();     // stable
Array.Sort(keys, values);                                                  // parallel arrays
people.Sort(Comparer<Person>.Create((a, b) => b.Score.CompareTo(a.Score))); // descending

// IComparable<T> for a natural order
public record Version(int Major, int Minor) : IComparable<Version>
{
    public int CompareTo(Version? o) => o is null ? 1 : Major != o.Major ? Major.CompareTo(o.Major) : Minor.CompareTo(o.Minor);
}

// Merge sort (stable, O(n log n), O(n) extra)
static int[] MergeSort(int[] a)
{
    if (a.Length <= 1) return a;
    var mid = a.Length / 2;
    var l = MergeSort(a[..mid]); var r = MergeSort(a[mid..]);
    var res = new int[a.Length]; int i = 0, j = 0, k = 0;
    while (i < l.Length && j < r.Length) res[k++] = l[i] <= r[j] ? l[i++] : r[j++];
    while (i < l.Length) res[k++] = l[i++];
    while (j < r.Length) res[k++] = r[j++];
    return res;
}

// Counting sort when the range is small
static int[] CountingSort(int[] a, int max)
{
    var counts = new int[max + 1];
    foreach (var x in a) counts[x]++;
    var res = new int[a.Length]; int k = 0;
    for (int v = 0; v <= max; v++) while (counts[v]-- > 0) res[k++] = v;
    return res;
}

def merge_sort(a: list[int]) -> list[int]:
    if len(a) <= 1: return a
    mid = len(a) // 2
    l, r = merge_sort(a[:mid]), merge_sort(a[mid:])

BFS and DFS on graphs and grids

Represent graphs as adjacency lists (Dictionary<int, List<int>>) or treat grids as implicit graphs. BFS (queue) for shortest path in unweighted graphs and level order; DFS (recursion or explicit stack) for connectivity, cycles and topological sort. Canonical: number of islands, clone graph, course schedule.

// BFS shortest path (unweighted)
static int ShortestPath(Dictionary<int, List<int>> g, int src, int dst)
{
    var dist = new Dictionary<int, int> { [src] = 0 };
    var q = new Queue<int>(); q.Enqueue(src);
    while (q.Count > 0)
    {
        var u = q.Dequeue();
        if (u == dst) return dist[u];
        foreach (var v in g.GetValueOrDefault(u, new()))
            if (dist.TryAdd(v, dist[u] + 1)) q.Enqueue(v);
    }
    return -1;
}

// DFS on a grid: number of islands
static int NumIslands(char[][] grid)
{
    int rows = grid.Length, cols = grid[0].Length, count = 0;
    void Sink(int r, int c)
    {
        if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] != '1') return;
        grid[r][c] = '0';
        Sink(r + 1, c); Sink(r - 1, c); Sink(r, c + 1); Sink(r, c - 1);
    }
    for (int r = 0; r < rows; r++) for (int c = 0; c < cols; c++) if (grid[r][c] == '1') { count++; Sink(r, c); }
    return count;
}

// Topological sort (Kahn) — course schedule
static bool CanFinish(int n, int[][] prereqs)
{
    var adj = new List<int>[n]; var indeg = new int[n];
    for (int i = 0; i < n; i++) adj[i] = new();
    foreach (var p in prereqs) { adj[p[1]].Add(p[0]); indeg[p[0]]++; }
    var q = new Queue<int>(Enumerable.Range(0, n).Where(i => indeg[i] == 0));
    int seen = 0;
    while (q.Count > 0) { var u = q.Dequeue(); seen++; foreach (var v in adj[u]) if (--indeg[v] == 0) q.Enqueue(v); }
    return seen == n;                       // false = cycle
}

// Dijkstra with PriorityQueue (.NET 6+)
static int[] Dijkstra(List<(int to, int w)>[] g, int src)
{
    var dist = Enumerable.Repeat(int.MaxValue, g.Length).ToArray(); dist[src] = 0;
    var pq = new PriorityQueue<int, int>(); pq.Enqueue(src, 0);
    while (pq.TryDequeue(out var u, out var d))
    {
        if (d > dist[u]) continue;
        foreach (var (v, w) in g[u]) if (d + w < dist[v]) { dist[v] = d + w; pq.Enqueue(v, dist[v]); }
    }
    return dist;
}

def shortest_path(g: dict[int, list[int]], src: int, dst: int) -> int:
    dist = {src: 0}; q = deque([src])
    while q:
        u = q.popleft()
        if u == dst: return dist[u]
        for v in g.get(u, []):
            if v not in dist:
                dist[v] = dist[u] + 1; q.append(v)
    return -1

def num_islands(grid: list[list[str]]) -> int:
    rows, cols = len(grid), len(grid[0]); count = 0
    def sink(r, c):
        if r < 0 or c < 0 or r >= rows or c >= cols or grid[r][c] != "1": return
        grid[r][c] = "0"
        for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)): sink(r + dr, c + dc)
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1": count += 1; sink(r, c)
    return count

def can_finish(n: int, prereqs: list[list[int]]) -> bool:
    adj = defaultdict(list); indeg = [0] * n
    for a, b in prereqs: adj[b].append(a); indeg[a] += 1
    q = deque(i for i in range(n) if indeg[i] == 0); seen = 0
    while q:
        u = q.popleft(); seen += 1
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0: q.append(v)
    return seen == n

def dijkstra(g: list[list[tuple[int, int]]], src: int) -> list[int]:
    dist = [float("inf")] * len(g); dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]: continue
        for v, w in g[u]:
            if d + w < dist[v]:
                dist[v] = d + w; heapq.heappush(pq, (dist[v], v))
    return dist

Heaps and PriorityQueue

A heap gives O(log n) insert and O(1) peek of the min/max — perfect for “top k”, “k-th largest”, merging k sorted lists, and scheduling. .NET 6 added PriorityQueue<TElement, TPriority> (min-heap; negate for max). Python's heapq is a min-heap over lists.

// K-th largest: keep a min-heap of size k
static int KthLargest(int[] nums, int k)
{
    var pq = new PriorityQueue<int, int>();
    foreach (var n in nums) { pq.Enqueue(n, n); if (pq.Count > k) pq.Dequeue(); }
    return pq.Peek();
}

// Merge k sorted lists
static ListNode? MergeK(ListNode?[] lists)
{
    var pq = new PriorityQueue<ListNode, int>();
    foreach (var l in lists) if (l is not null) pq.Enqueue(l, l.Val);
    var dummy = new ListNode(0); var tail = dummy;
    while (pq.TryDequeue(out var node, out _))
    {
        tail.Next = node; tail = node;
        if (node.Next is not null) pq.Enqueue(node.Next, node.Next.Val);
    }
    return dummy.Next;
}

// Running median: max-heap for the low half, min-heap for the high half
public class MedianFinder
{
    private readonly PriorityQueue<int, int> _low = new(Comparer<int>.Create((a, b) => b.CompareTo(a)));
    private readonly PriorityQueue<int, int> _high = new();
    public void Add(int x)
    {
        _low.Enqueue(x, x);
        _high.Enqueue(_low.Peek(), _low.Dequeue());
        if (_high.Count > _low.Count) _low.Enqueue(_high.Peek(), _high.Dequeue());
    }
    public double Median => _low.Count > _high.Count ? _low.Peek() : (_low.Peek() + _high.Peek()) / 2.0;
}

def merge_k(lists):
    heap = [(l.val, i, l) for i, l in enumerate(lists) if l]   # i breaks ties (nodes aren't comparable)
    heapq.heapify(heap)
    dummy = tail = ListNode(0)
    while heap:
        _, i, node = heapq.heappop(heap)
        tail.next = tail = node
        if node.next: heapq.heappush(heap, (node.next.val, i, node.next))
    return dummy.next

class MedianFinder:
    def __init__(self): self.low, self.high = [], []      # low is a max-heap (negated)
    def add(self, x):
        heapq.heappush(self.low, -x)
        heapq.heappush(self.high, -heapq.heappop(self.low))
        if len(self.high) > len(self.low):
            heapq.heappush(self.low, -heapq.heappop(self.high))
    @property
    def median(self):
        return -self.low[0] if len(self.low) > len(self.high) else (-self.low[0] + self.high[0]) / 2

Dynamic programming — an introduction

DP = recursion + memoisation over overlapping subproblems. Approach: define the state, write the recurrence, decide base cases, then either memoise top-down or fill a table bottom-up. Canonical starters: climbing stairs, coin change, longest increasing subsequence, 0/1 knapsack, edit distance.

// Climbing stairs: f(n) = f(n-1) + f(n-2), O(1) space
static int ClimbStairs(int n)
{
    int a = 1, b = 1;
    for (int i = 2; i <= n; i++) (a, b) = (b, a + b);
    return b;
}

// Coin change (min coins): bottom-up
static int CoinChange(int[] coins, int amount)
{
    var dp = new int[amount + 1]; Array.Fill(dp, amount + 1); dp[0] = 0;
    for (int a = 1; a <= amount; a++)
        foreach (var c in coins) if (c <= a) dp[a] = Math.Min(dp[a], dp[a - c] + 1);
    return dp[amount] > amount ? -1 : dp[amount];
}

// Top-down with memo: number of unique paths in a grid
static long UniquePaths(int r, int c, Dictionary<(int, int), long>? memo = null)
{
    memo ??= new();
    if (r == 1 || c == 1) return 1;
    if (memo.TryGetValue((r, c), out var v)) return v;
    return memo[(r, c)] = UniquePaths(r - 1, c, memo) + UniquePaths(r, c - 1, memo);
}

// Longest increasing subsequence: O(n log n) with patience sorting
static int LIS(int[] nums)
{
    var tails = new List<int>();
    foreach (var x in nums)
    {
        int i = tails.BinarySearch(x);
        if (i < 0) i = ~i;
        if (i == tails.Count) tails.Add(x); else tails[i] = x;
    }
    return tails.Count;
}

// 0/1 knapsack: 1-D table, iterate capacity downwards
static int Knapsack(int[] w, int[] v, int cap)
{
    var dp = new int[cap + 1];
    for (int i = 0; i < w.Length; i++)
        for (int c = cap; c >= w[i]; c--) dp[c] = Math.Max(dp[c], dp[c - w[i]] + v[i]);
    return dp[cap];
}
from functools import lru_cache
import bisect

def climb_stairs(n: int) -> int:
    a = b = 1
    for _ in range(2, n + 1): a, b = b, a + b
    return b

def coin_change(coins: list[int], amount: int) -> int:
    dp = [amount + 1] * (amount + 1); dp[0] = 0
    for a in range(1, amount + 1):
        for c in coins:
            if c <= a: dp[a] = min(dp[a], dp[a - c] + 1)
    return -1 if dp[amount] > amount else dp[amount]

@lru_cache(maxsize=None)          # top-down memo for free
def unique_paths(r: int, c: int) -> int:
    if r == 1 or c == 1: return 1
    return unique_paths(r - 1, c) + unique_paths(r, c - 1)

def lis(nums: list[int]) -> int:
    tails = []
    for x in nums:
        i = bisect.bisect_left(tails, x)
        if i == len(tails): tails.append(x)
        else: tails[i] = x
    return len(tails)

def knapsack(w: list[int], v: list[int], cap: int) -> int:
    dp = [0] * (cap + 1)
    for wi, vi in zip(w, v):
        for c in range(cap, wi - 1, -1):
            dp[c] = max(dp[c], dp[c - wi] + vi)
    return dp[cap]
