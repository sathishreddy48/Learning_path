Authentication vs authorization

Authentication (authn) answers “who are you?” and produces a ClaimsPrincipal. Authorization (authz) answers “are you allowed?” by evaluating roles, claims or policies against that principal. In ASP.NET Core they are two middlewares — UseAuthentication() must run first. Schemes (Cookie, JWT Bearer, OpenIdConnect) are how identity is proven; policies are how access is decided.

// 1. Register an authentication scheme (how identity is proven)
builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(/* see next section */);

// 2. Register authorization (how access is decided)
builder.Services.AddAuthorization(o =>
{
    o.AddPolicy("AdminOnly", p => p.RequireRole("admin"));
    o.FallbackPolicy = new AuthorizationPolicyBuilder().RequireAuthenticatedUser().Build(); // secure by default
});

var app = builder.Build();
app.UseAuthentication();     // sets HttpContext.User
app.UseAuthorization();      // enforces [Authorize] / RequireAuthorization

app.MapGet("/me", (ClaimsPrincipal user) => new
{
    name = user.Identity?.Name,
    id = user.FindFirstValue(ClaimTypes.NameIdentifier),
    roles = user.FindAll(ClaimTypes.Role).Select(c => c.Value)
});
app.MapGet("/public", () => "anyone").AllowAnonymous();
app.MapGet("/admin", () => "admins").RequireAuthorization("AdminOnly");

@dataclass
class Principal:                       # ≈ ClaimsPrincipal
    id: str; name: str; roles: list[str] = field(default_factory=list); claims: dict = field(default_factory=dict)

# Authentication: turn credentials into a principal (or None)
async def current_user(creds: HTTPAuthorizationCredentials | None = Depends(bearer)) -> Principal | None:
    if creds is None:
        return None
    return decode_token(creds.credentials)      # see JWT section

# Authorization: policies as dependencies
def require_user(user: Principal | None = Depends(current_user)) -> Principal:
    if user is None:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, headers={"WWW-Authenticate": "Bearer"})
    return user

def require_role(role: str):
    def check(user: Principal = Depends(require_user)) -> Principal:
        if role not in user.roles:
            raise HTTPException(status.HTTP_403_FORBIDDEN)
        return user
    return check

@app.get("/me")
def me(user: Principal = Depends(require_user)):
    return {"name": user.name, "id": user.id, "roles": user.roles}

@app.get("/public")
def public(): return "anyone"

@app.get("/admin", dependencies=[Depends(require_role("admin"))])
def admin(): return "admins"
Interview angle

401 vs 403: 401 = not authenticated (and include WWW-Authenticate), 403 = authenticated but not allowed. Also: what a claim is, and why roles are just claims of a particular type.

Practice
  • Configure a fallback policy that requires authentication everywhere, then explicitly open /health and /login with AllowAnonymous.

JWT bearer tokens

A JSON Web Token carries signed claims. The API validates the signature (symmetric key or the issuer's public keys), issuer, audience and expiry — then trusts the claims without a database round-trip. Keep tokens short-lived and use refresh tokens for longevity. Never put secrets in the payload: it is only base64-encoded, not encrypted.

// Validate tokens (API side)
builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(o =>
    {
        o.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,   ValidIssuer = "https://auth.example.com",
            ValidateAudience = true, ValidAudience = "orders-api",
            ValidateLifetime = true, ClockSkew = TimeSpan.FromSeconds(30),
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(builder.Configuration["Jwt:Key"]!))
        };
        // For an external IdP, set o.Authority = "https://login.microsoftonline.com/{tenant}/v2.0"
        // and the handler downloads the signing keys (JWKS) automatically.
    });

// Issue tokens (auth server / login endpoint) — use an IdP in production
app.MapPost("/login", (LoginDto dto, IConfiguration cfg, IUserStore users) =>
{
    var user = users.Verify(dto.Username, dto.Password);
    if (user is null) return Results.Unauthorized();

    var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(cfg["Jwt:Key"]!));
    var token = new JwtSecurityToken(
        issuer: "https://auth.example.com",
        audience: "orders-api",
        claims: new[]
        {
            new Claim(JwtRegisteredClaimNames.Sub, user.Id),
            new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString()),
            new Claim(ClaimTypes.Role, "admin")
        },
        expires: DateTime.UtcNow.AddMinutes(15),
        signingCredentials: new SigningCredentials(key, SecurityAlgorithms.HmacSha256));

    return Results.Ok(new { access_token = new JwtSecurityTokenHandler().WriteToken(token), token_type = "Bearer", expires_in = 900 });
});

app = FastAPI()

# Validate tokens (API side)
def decode_token(token: str) -> Principal:
    try:
        payload = jwt.decode(token, SECRET, algorithms=["HS256"],
                             issuer=ISSUER, audience=AUDIENCE, leeway=30)
    except jwt.ExpiredSignatureError:
        raise HTTPException(401, "token expired")
    except jwt.InvalidTokenError:
        raise HTTPException(401, "invalid token")
    return Principal(id=payload["sub"], name=payload.get("name", ""), roles=payload.get("roles", []))

def current_user(token: str = Depends(oauth2)) -> Principal:
    return decode_token(token)

# Issue tokens
@app.post("/login")
def login(form: OAuth2PasswordRequestForm = Depends()):
    user = users.verify(form.username, form.password)
    if user is None:
        raise HTTPException(401)
    now = datetime.now(timezone.utc)
    token = jwt.encode({
        "sub": user.id, "jti": uuid.uuid4().hex, "roles": ["admin"],
        "iss": ISSUER, "aud": AUDIENCE,
        "iat": now, "exp": now + timedelta(minutes=15),
    }, SECRET, algorithm="HS256")
    return {"access_token": token, "token_type": "bearer", "expires_in": 900}

OAuth 2.0, OpenID Connect and Microsoft Entra ID

OAuth 2.0 is delegated authorization: a client gets an access token to call an API on a user's behalf. OpenID Connect adds an ID token and standard user info on top — that is what “login with Microsoft/Google” is. Flows: Authorization Code + PKCE (web/SPA/mobile), Client Credentials (service-to-service), On-Behalf-Of (API calling another API as the user). Entra ID is Microsoft's IdP; Microsoft.Identity.Web wires it into ASP.NET Core.

// API protected by Entra ID (validates tokens issued to your app registration)
// appsettings.json: "AzureAd": { "Instance": "https://login.microsoftonline.com/", "TenantId": "...", "ClientId": "api://orders" }
builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddMicrosoftIdentityWebApi(builder.Configuration.GetSection("AzureAd"))
    .EnableTokenAcquisitionToCallDownstreamApi()           // On-Behalf-Of
    .AddDownstreamApi("Graph", builder.Configuration.GetSection("Graph"))
    .AddInMemoryTokenCaches();

app.MapGet("/profile", [Authorize] async (IDownstreamApi graph) =>
    await graph.CallApiForUserAsync<User>("Graph", o => o.RelativePath = "me"));

// Require a scope (delegated) or app role (client credentials)

app.MapGet("/jobs", () => ...).RequireAuthorization(p => p.RequireRole("Jobs.Run"));

Role-, claim- and policy-based authorization

Roles are the simplest check but do not scale to business rules. Policies compose requirements; a handler evaluates each requirement against the user and (optionally) the resource. Resource-based authorization (“can this user edit this document?”) uses IAuthorizationService inside the endpoint.

builder.Services.AddAuthorization(o =>
{
    o.AddPolicy("Adult", p => p.RequireClaim("age_verified", "true"));
    o.AddPolicy("EngOrAdmin", p => p.RequireAssertion(ctx =>
        ctx.User.IsInRole("admin") || ctx.User.HasClaim("department", "eng")));
    o.AddPolicy("MinLevel3", p => p.Requirements.Add(new MinLevelRequirement(3)));
});
builder.Services.AddSingleton<IAuthorizationHandler, MinLevelHandler>();
builder.Services.AddSingleton<IAuthorizationHandler, DocumentOwnerHandler>();

public record MinLevelRequirement(int Level) : IAuthorizationRequirement;

public class MinLevelHandler : AuthorizationHandler<MinLevelRequirement>
{
    protected override Task HandleRequirementAsync(AuthorizationHandlerContext ctx, MinLevelRequirement req)
    {
        var lvl = int.TryParse(ctx.User.FindFirstValue("level"), out var l) ? l : 0;
        if (lvl >= req.Level) ctx.Succeed(req);
        return Task.CompletedTask;         // not calling Succeed = fail (unless another handler succeeds)
    }
}

// Resource-based: the handler sees the resource
public record OwnerRequirement : IAuthorizationRequirement;
public class DocumentOwnerHandler : AuthorizationHandler<OwnerRequirement, Document>
{
    protected override Task HandleRequirementAsync(AuthorizationHandlerContext ctx, OwnerRequirement req, Document doc)
    {
        if (doc.OwnerId == ctx.User.FindFirstValue(ClaimTypes.NameIdentifier)) ctx.Succeed(req);
        return Task.CompletedTask;
    }
}

app.MapPut("/docs/{id}", async (int id, DocDto dto, ClaimsPrincipal user, IAuthorizationService authz, IDocRepo repo) =>
{
    var doc = await repo.Get(id);
    if (doc is null) return Results.NotFound();
    var result = await authz.AuthorizeAsync(user, doc, new OwnerRequirement());
    if (!result.Succeeded) return Results.Forbid();
    await repo.Update(doc with { Title = dto.Title });
    return Results.NoContent();
});
from fastapi import FastAPI, Depends, HTTPException

app = FastAPI()

# Policies as dependency factories
def require_claim(name: str, value: str):
    def check(user: Principal = Depends(require_user)):
        if user.claims.get(name) != value:
            raise HTTPException(403)
        return user
    return check

def require_any(*checks):
    def check(user: Principal = Depends(require_user)):
        if not any(c(user) for c in checks):
            raise HTTPException(403)
        return user
    return check

is_admin = lambda u: "admin" in u.roles
is_eng = lambda u: u.claims.get("department") == "eng"

def min_level(level: int):
    def check(user: Principal = Depends(require_user)):
        if int(user.claims.get("level", 0)) < level:
            raise HTTPException(403)
        return user
    return check

@app.get("/adult", dependencies=[Depends(require_claim("age_verified", "true"))])
def adult(): ...

@app.get("/eng", dependencies=[Depends(require_any(is_admin, is_eng))])
def eng(): ...

# Resource-based: check inside the handler
@app.put("/docs/{id}", status_code=204)
async def update_doc(id: int, dto: DocDto, user: Principal = Depends(require_user)):
    doc = await repo.get(id)
    if doc is None:
        raise HTTPException(404)
    if doc.owner_id != user.id and "admin" not in user.roles:
        raise HTTPException(403)
    await repo.update(doc.model_copy(update={"title": dto.title}))

Password hashing and ASP.NET Core Identity

Never store or compare plain-text passwords. Use a slow, salted hash: PBKDF2 (what Identity uses), bcrypt, or Argon2id. ASP.NET Core Identity provides users, roles, password policies, lockout, 2FA and email confirmation on top of EF Core — use it instead of rolling your own. In Python, passlib with bcrypt/argon2 is the standard.

// Manual hashing (when you can't use Identity)
public static class PasswordHasher
{
    public static string Hash(string password)
    {
        byte[] salt = RandomNumberGenerator.GetBytes(16);
        byte[] hash = Rfc2898DeriveBytes.Pbkdf2(password, salt, iterations: 600_000, HashAlgorithmName.SHA256, outputLength: 32);
        return $"{Convert.ToBase64String(salt)}.{Convert.ToBase64String(hash)}";
    }
    public static bool Verify(string password, string stored)
    {
        var parts = stored.Split('.');
        byte[] salt = Convert.FromBase64String(parts[0]);
        byte[] expected = Convert.FromBase64String(parts[1]);
        byte[] actual = Rfc2898DeriveBytes.Pbkdf2(password, salt, 600_000, HashAlgorithmName.SHA256, 32);
        return CryptographicOperations.FixedTimeEquals(actual, expected);   // constant-time compare
    }
}

# Manual PBKDF2 with the stdlib (if you can't add a dependency)
def hash_password(password: str) -> str:
    salt = os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 600_000, dklen=32)
    return f"{base64.b64encode(salt).decode()}.{base64.b64encode(dk).decode()}"

def verify_password(password: str, stored: str) -> bool:
    s, h = stored.split(".")
    salt, expected = base64.b64decode(s), base64.b64decode(h)
    actual = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 600_000, dklen=32)
    return hmac.compare_digest(actual, expected)

Secrets and configuration

Secrets never go in source control. Locally: dotnet user-secrets (stored outside the repo). In Azure: Key Vault via Managed Identity, with AddAzureKeyVault mapping secrets to configuration keys. Environment variables for containers. The configuration system layers all of these, so code just reads IConfiguration["ConnectionStrings:Default"].

// Local development
//   dotnet user-secrets init
//   dotnet user-secrets set "ConnectionStrings:Default" "Server=...;Password=..."
//   dotnet user-secrets set "Jwt:Key" "dev-only-long-random-string"

// Configuration layering (later wins): appsettings.json → appsettings.{Env}.json →
// user secrets (Dev) → environment variables → command line
var builder = WebApplication.CreateBuilder(args);

// Azure Key Vault in non-development
if (!builder.Environment.IsDevelopment())
{
    builder.Configuration.AddAzureKeyVault(
        new Uri($"https://{builder.Configuration["KeyVault:Name"]}.vault.azure.net/"),
        new DefaultAzureCredential());          // Managed Identity in Azure, az login locally
    // secret "ConnectionStrings--Default" in Key Vault → "ConnectionStrings:Default" in config
}

// Strongly-typed access

HTTPS, HSTS and security headers

Redirect HTTP to HTTPS, send Strict-Transport-Security so browsers never try HTTP again, and add the standard hardening headers: X-Content-Type-Options, X-Frame-Options/frame-ancestors, Content-Security-Policy, Referrer-Policy. Behind a reverse proxy, configure forwarded headers so the app knows the original scheme.

var app = builder.Build();

// Behind Azure App Service / nginx / Front Door
app.UseForwardedHeaders(new ForwardedHeadersOptions
{
    ForwardedHeaders = ForwardedHeaders.XForwardedFor | ForwardedHeaders.XForwardedProto
});

if (!app.Environment.IsDevelopment())
{
    app.UseHsts();                       // Strict-Transport-Security: max-age=2592000
}
app.UseHttpsRedirection();

// Security headers middleware
app.Use(async (ctx, next) =>
{
    var h = ctx.Response.Headers;
    h["X-Content-Type-Options"] = "nosniff";
    h["X-Frame-Options"] = "DENY";
    h["Referrer-Policy"] = "strict-origin-when-cross-origin";
    h["Permissions-Policy"] = "camera=(), microphone=(), geolocation=()";
    h["Content-Security-Policy"] = "default-src 'self'; frame-ancestors 'none'; object-src 'none'";
    await next();
});

// Cookie hardening
builder.Services.ConfigureApplicationCookie(o =>

OWASP Top 10 in .NET: injection, XSS, CSRF, SSRF

The bugs that actually get exploited. SQL injection: use parameters — EF Core and Dapper do this; string-built SQL does not. XSS: Razor encodes by default; Html.Raw and manual string building do not. CSRF: antiforgery tokens for cookie-authenticated forms; bearer-token APIs are immune. SSRF: validate and allow-list any URL you fetch on behalf of a user. Plus mass assignment, path traversal and insecure deserialization.

// SQL injection
var bad  = db.Users.FromSqlRaw($"SELECT * FROM Users WHERE Name = '{name}'");          // ✗ injectable
var good = db.Users.FromSqlInterpolated($"SELECT * FROM Users WHERE Name = {name}");  // ✓ parameterised
var ok   = db.Users.Where(u => u.Name == name);                                        // ✓ LINQ
var dapper = conn.Query<User>("SELECT * FROM Users WHERE Name = @Name", new { Name = name }); // ✓

Data Protection API and encryption

ASP.NET Core's Data Protection stack encrypts cookies, antiforgery tokens and anything you Protect(), with automatic key rotation. In a web farm the key ring must be shared (blob storage + Key Vault). For general encryption use AesGcm (authenticated) — never ECB, never a home-made scheme; hash with SHA-256 for integrity, HMAC for signatures.

// Shared key ring for multiple instances (Azure)
builder.Services.AddDataProtection()
    .SetApplicationName("orders")                                         // same name on every instance
    .PersistKeysToAzureBlobStorage(new Uri(blobUri), new DefaultAzureCredential())
    .ProtectKeysWithAzureKeyVault(new Uri(keyId), new DefaultAzureCredential())
    .SetDefaultKeyLifetime(TimeSpan.FromDays(90));

// Protect arbitrary data (purpose strings isolate uses)
public class TokenService(IDataProtectionProvider provider)
{
    private readonly IDataProtector _p = provider.CreateProtector("password-reset");
    private readonly ITimeLimitedDataProtector _t = provider.CreateProtector("password-reset").ToTimeLimitedDataProtector();

    public string Create(string userId) => _t.Protect(userId, TimeSpan.FromMinutes(30));
    public string? Read(string token)
    {
        try { return _t.Unprotect(token); } catch (CryptographicException) { return null; }
    }
}

// AES-GCM for data at rest you manage yourself
public static (byte[] cipher, byte[] nonce, byte[] tag) Encrypt(byte[] key, byte[] plain)
{
    var nonce = RandomNumberGenerator.GetBytes(AesGcm.NonceByteSizes.MaxSize);
    var cipher = new byte[plain.Length];
    var tag = new byte[AesGcm.TagByteSizes.MaxSize];
    using var aes = new AesGcm(key, tag.Length);
    aes.Encrypt(nonce, plain, cipher, tag);
    return (cipher, nonce, tag);
}

Input validation and safe defaults

Validate at the boundary, allow-list rather than deny-list, limit sizes, and fail closed. Set request body limits, restrict file uploads by type and size, use strongly-typed IDs to avoid IDOR, and log security events without logging secrets. Most of this is configuration, not code.

// Request size limits
builder.WebHost.ConfigureKestrel(k => k.Limits.MaxRequestBodySize = 10 * 1024 * 1024);   // 10 MB
app.MapPost("/upload", ...).WithMetadata(new RequestSizeLimitAttribute(2 * 1024 * 1024));

// File upload validation
app.MapPost("/avatar", async (IFormFile file) =>
{
    var allowed = new Dictionary<string, byte[]> { [".png"] = new byte[] { 0x89, 0x50, 0x4E, 0x47 } };
    var ext = Path.GetExtension(file.FileName).ToLowerInvariant();
    if (!allowed.ContainsKey(ext) || file.Length > 1_000_000) return Results.BadRequest();
    using var s = file.OpenReadStream();
    var header = new byte[4]; await s.ReadExactlyAsync(header);
    if (!header.SequenceEqual(allowed[ext])) return Results.BadRequest("not a png");      // check magic bytes, not just extension
    var safeName = $"{Guid.NewGuid()}{ext}";                                              // never use the client's file name
    await using var dst = File.Create(Path.Combine("uploads", safeName));
    s.Position = 0; await s.CopyToAsync(dst);
    return Results.Ok(safeName);
});

// IDOR: scope every query by the caller
app.MapGet("/orders/{id}", async (int id, ClaimsPrincipal user, AppDb db) =>
{
    var uid = user.FindFirstValue(ClaimTypes.NameIdentifier);
    var order = await db.Orders.FirstOrDefaultAsync(o => o.Id == id && o.OwnerId == uid);   // not just o.Id == id
    return order is null ? Results.NotFound() : Results.Ok(order);
});

MAGIC = {".png": b"\x89PNG"}
@app.post("/avatar")
async def avatar(file: UploadFile):
    ext = Path(file.filename or "").suffix.lower()
    header = await file.read(4)
    if ext not in MAGIC or header != MAGIC[ext]:
        raise HTTPException(400, "not a png")
    data = header + await file.read(1_000_000 + 1)
