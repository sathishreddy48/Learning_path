Minimal APIs vs controllers

Both styles run on the same pipeline. Minimal APIs (.NET 6+) map lambdas straight onto routes — less ceremony, great for small services. Controllers give you attribute routing, filters, model-state conventions and a familiar structure for large APIs. Know both; interviewers ask when you would pick each. FastAPI is closest to minimal APIs.

// Program.cs — minimal API
var builder = WebApplication.CreateBuilder(args);
builder.Services.AddSingleton<ITodoRepo, InMemoryTodoRepo>();
var app = builder.Build();

app.MapGet("/todos", (ITodoRepo repo) => repo.All());
app.MapGet("/todos/{id:int}", (int id, ITodoRepo repo) =>
    repo.Get(id) is { } t ? Results.Ok(t) : Results.NotFound());
app.MapPost("/todos", (Todo todo, ITodoRepo repo) =>
{
    var created = repo.Add(todo);
    return Results.Created($"/todos/{created.Id}", created);
});
app.MapDelete("/todos/{id:int}", (int id, ITodoRepo repo) =>
    repo.Delete(id) ? Results.NoContent() : Results.NotFound());

// Group + shared config

class InMemoryTodoRepo:
    def __init__(self): self._items: dict[int, Todo] = {}; self._next = 1
    def all(self): return list(self._items.values())
    def get(self, id): return self._items.get(id)
    def add(self, t: Todo):
        t.id = self._next; self._next += 1; self._items[t.id] = t; return t
    def delete(self, id): return self._items.pop(id, None) is not None

Routing and model binding

Route templates with constraints ({id:int}, {slug:alpha}, {*path}) decide which endpoint runs. Model binding then fills parameters from the route, query string, headers, body (JSON) or services. Minimal APIs infer the source; attributes ([FromQuery], [FromBody], [FromHeader], [FromServices]) make it explicit. FastAPI uses parameter annotations for the same purpose.

// Route constraints and catch-all
app.MapGet("/users/{id:int:min(1)}", (int id) => $"user {id}");
app.MapGet("/posts/{year:int}/{slug}", (int year, string slug) => $"{year}/{slug}");
app.MapGet("/files/{*path}", (string path) => path);

// Sources
app.MapGet("/search", (
    [FromQuery(Name = "q")] string query,
    [FromQuery] int page = 1,
    [FromHeader(Name = "X-Correlation-Id")] string? corr = null) =>
    new { query, page, corr });

app.MapPost("/orders", ([FromBody] OrderDto dto, [FromServices] IOrderService svc) => svc.Place(dto));

// Bind a whole object from the query string (AsParameters)
app.MapGet("/products", ([AsParameters] ProductQuery q) => q);
record ProductQuery(string? Category, int Page = 1, int Size = 20);

// Custom binding: TryParse or BindAsync on your type
public record Money(decimal Amount, string Currency)
{
    public static bool TryParse(string s, out Money? m)
    {
        var parts = s.Split(' ');
        m = parts.Length == 2 && decimal.TryParse(parts[0], out var a) ? new(a, parts[1]) : null;
        return m is not null;
    }
}
app.MapGet("/price/{money}", (Money money) => money);    // GET /price/9.99%20USD

Validation

Data annotations ([Required], [Range], [StringLength]) work automatically in controllers with [ApiController] (400 + ProblemDetails). Minimal APIs need a filter or a library — FluentValidation is the de facto standard for complex rules. Pydantic does this natively in FastAPI.

// Data annotations — automatic in controllers
public class CreateUserDto
{
    [Required, StringLength(50, MinimumLength = 2)] public string Name { get; set; } = "";
    [Required, EmailAddress] public string Email { get; set; } = "";
    [Range(18, 120)] public int Age { get; set; }
}

// FluentValidation — expressive, testable
public class CreateUserValidator : AbstractValidator<CreateUserDto>
{
    public CreateUserValidator()
    {
        RuleFor(x => x.Name).NotEmpty().Length(2, 50);
        RuleFor(x => x.Email).NotEmpty().EmailAddress();
        RuleFor(x => x.Age).InclusiveBetween(18, 120);
        RuleFor(x => x).Must(BeUniqueEmail).WithMessage("Email already used");
    }
}
builder.Services.AddValidatorsFromAssemblyContaining<CreateUserValidator>();

// Minimal API endpoint filter that validates any argument with a registered validator
public class ValidationFilter<T> : IEndpointFilter
{
    public async ValueTask<object?> InvokeAsync(EndpointFilterInvocationContext ctx, EndpointFilterDelegate next)
    {
        var validator = ctx.HttpContext.RequestServices.GetService<IValidator<T>>();
        var arg = ctx.Arguments.OfType<T>().FirstOrDefault();
        if (validator is not null && arg is not null)
        {
            var result = await validator.ValidateAsync(arg);
            if (!result.IsValid)
                return Results.ValidationProblem(result.ToDictionary());   // 400 + ProblemDetails
        }
        return await next(ctx);
    }
}

app.MapPost("/users", (CreateUserDto dto) => Results.Created("/users/1", dto))
   .AddEndpointFilter<ValidationFilter<CreateUserDto>>();

Dependency injection and service lifetimes

ASP.NET Core has a built-in container. Three lifetimes: Transient (new every time), Scoped (one per HTTP request), Singleton (one for the process). The classic bug: injecting a scoped service (like DbContext) into a singleton — a “captive dependency”. FastAPI's Depends is per-request by default; singletons are module-level objects.

builder.Services.AddTransient<IEmailSender, SmtpEmailSender>();   // stateless helpers
builder.Services.AddScoped<IOrderService, OrderService>();        // per request (uses DbContext)
builder.Services.AddSingleton<ICache, MemoryCache>();             // shared, must be thread-safe
builder.Services.AddDbContext<AppDb>(o => o.UseSqlServer(cs));   // scoped by default

// Options pattern
builder.Services.Configure<SmtpOptions>(builder.Configuration.GetSection("Smtp"));
public class SmtpEmailSender(IOptions<SmtpOptions> opts, ILogger<SmtpEmailSender> log) : IEmailSender { }

// Constructor injection (primary constructor, C# 12)
public class OrderService(AppDb db, IEmailSender email) : IOrderService
{
    public async Task Place(OrderDto dto) { db.Add(dto.ToEntity()); await db.SaveChangesAsync(); await email.Send(); }
}

// Captive dependency — BUG: singleton holds a scoped DbContext forever
builder.Services.AddSingleton<ReportCache>();     // ctor takes AppDb → wrong
// Fix: inject IServiceScopeFactory and create a scope per operation
public class ReportCache(IServiceScopeFactory scopes)
{
    public async Task Refresh()
    {
        using var scope = scopes.CreateScope();
        var db = scope.ServiceProvider.GetRequiredService<AppDb>();
        // ...
    }
}

// Keyed services (.NET 8)
builder.Services.AddKeyedSingleton<IPayment, StripePayment>("stripe");
app.MapPost("/pay", ([FromKeyedServices("stripe")] IPayment p) => p.Charge());

// Validate lifetimes at startup in Development
builder.Host.UseDefaultServiceProvider(o => { o.ValidateScopes = true; o.ValidateOnBuild = true; });

    db = SessionLocal()

    return SmtpEmailSender(settings)

    def __init__(self, db, email): self.db, self.email = db, email
    async def place(self, dto): ...

def get_order_service(db=Depends(get_db), email=Depends(get_email_sender)):
    return OrderService(db, email)

@app.post("/orders")

Middleware pipeline

A request passes through an ordered chain of middleware; each can short-circuit or call next. Order matters: exception handling → HSTS/HTTPS → static files → routing → CORS → authentication → authorization → endpoints. Write your own with a class or inline app.Use. FastAPI/Starlette middleware is the same idea.

// Order matters
app.UseExceptionHandler();          // outermost: catches everything below
app.UseHsts();
app.UseHttpsRedirection();
app.UseStaticFiles();
app.UseRouting();
app.UseCors();
app.UseAuthentication();
app.UseAuthorization();
app.MapControllers();

// Inline middleware
app.Use(async (ctx, next) =>
{
    var sw = Stopwatch.StartNew();
    await next(ctx);
    ctx.Response.Headers["X-Elapsed-Ms"] = sw.ElapsedMilliseconds.ToString();   // careful: headers must be set before body starts
});

// Class-based middleware
public class CorrelationIdMiddleware(RequestDelegate next, ILogger<CorrelationIdMiddleware> log)
{
    public async Task InvokeAsync(HttpContext ctx)
    {
        var id = ctx.Request.Headers["X-Correlation-Id"].FirstOrDefault() ?? Guid.NewGuid().ToString("N");
        ctx.Items["CorrelationId"] = id;
        ctx.Response.OnStarting(() => { ctx.Response.Headers["X-Correlation-Id"] = id; return Task.CompletedTask; });
        using (log.BeginScope(new Dictionary<string, object> { ["CorrelationId"] = id }))
            await next(ctx);
    }
}
app.UseMiddleware<CorrelationIdMiddleware>();

// Short-circuit / branch
app.Map("/ping", b => b.Run(async ctx => await ctx.Response.WriteAsync("pong")));
app.UseWhen(ctx => ctx.Request.Path.StartsWithSegments("/admin"),
            b => b.UseMiddleware<AdminAuditMiddleware>());

    t0 = time.perf_counter()
    response = await call_next(request)
    response.headers["X-Elapsed-Ms"] = f"{(time.perf_counter() - t0) * 1000:.0f}"
    return response

# Class-based middleware
class CorrelationIdMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        cid = request.headers.get("X-Correlation-Id") or uuid.uuid4().hex
        request.state.correlation_id = cid
        response = await call_next(request)
        response.headers["X-Correlation-Id"] = cid
        return response

Filters (MVC) and endpoint filters

Filters run inside the endpoint, after routing and model binding, so they know which action is executing and can see its arguments. MVC has authorization, resource, action, exception and result filters. Minimal APIs have a single IEndpointFilter pipeline. FastAPI achieves the same with dependencies and decorators.

// MVC action filter: log arguments and timing
public class LogActionFilter(ILogger<LogActionFilter> log) : IAsyncActionFilter
{
    public async Task OnActionExecutionAsync(ActionExecutingContext ctx, ActionExecutionDelegate next)
    {
        log.LogInformation("→ {Action} {@Args}", ctx.ActionDescriptor.DisplayName, ctx.ActionArguments);
        var executed = await next();                       // runs the action
        if (executed.Exception is not null) log.LogError(executed.Exception, "action failed");
    }
}
builder.Services.AddControllers(o => o.Filters.Add<LogActionFilter>());   // global
[ServiceFilter(typeof(LogActionFilter))] public class OrdersController : ControllerBase { }   // per controller

// Exception filter → custom response
public class DomainExceptionFilter : IExceptionFilter
{
    public void OnException(ExceptionContext ctx)
    {
        if (ctx.Exception is DomainException de)
        {
            ctx.Result = new BadRequestObjectResult(new { error = de.Message });
            ctx.ExceptionHandled = true;
        }
    }
}

// Minimal API endpoint filter
app.MapPost("/orders", (OrderDto dto) => Results.Ok(dto))
   .AddEndpointFilter(async (ctx, next) =>
   {
       var dto = ctx.GetArgument<OrderDto>(0);

       return await next(ctx);
   });

# Or a plain decorator around the handler function
def timed(fn):
    @functools.wraps(fn)
    async def wrapper(*a, **kw):
        import time; t0 = time.perf_counter()
        try: return await fn(*a, **kw)
        finally: logging.info("%s took %.1fms", fn.__name__, (time.perf_counter()-t0)*1000)
    return wrapper

Error handling and ProblemDetails

Return RFC 7807 ProblemDetails for every error so clients get a consistent shape. .NET 8 adds IExceptionHandler plus AddProblemDetails(), replacing hand-rolled exception middleware. Map domain exceptions to status codes in one place; never leak stack traces in production. FastAPI's exception handlers do the same job.

builder.Services.AddProblemDetails(o => o.CustomizeProblemDetails = ctx =>
{
    ctx.ProblemDetails.Extensions["traceId"] = ctx.HttpContext.TraceIdentifier;
});
builder.Services.AddExceptionHandler<GlobalExceptionHandler>();
var app = builder.Build();
app.UseExceptionHandler();          // uses the handlers above
app.UseStatusCodePages();           // ProblemDetails for bare 404/405 etc.

public class GlobalExceptionHandler(ILogger<GlobalExceptionHandler> log) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(HttpContext ctx, Exception ex, CancellationToken ct)
    {
        var (status, title) = ex switch
        {
            NotFoundException   => (StatusCodes.Status404NotFound, "Not found"),
            ValidationException => (StatusCodes.Status400BadRequest, "Validation failed"),
            UnauthorizedAccessException => (StatusCodes.Status403Forbidden, "Forbidden"),
            _                   => (StatusCodes.Status500InternalServerError, "Unexpected error")
        };
        if (status == 500) log.LogError(ex, "Unhandled exception");

        await Results.Problem(title: title, statusCode: status,
                              detail: status == 500 ? null : ex.Message,
                              extensions: new Dictionary<string, object?> { ["traceId"] = ctx.TraceIdentifier })
                     .ExecuteAsync(ctx);
        return true;                  // handled
    }
}

// Explicit results in endpoints
app.MapGet("/orders/{id}", (int id) =>
    id > 0 ? Results.Ok() : Results.Problem("id must be positive", statusCode: 400));

// Typed results make OpenAPI accurate
app.MapGet("/x/{id}", Results<Ok<Todo>, NotFound> (int id) => id == 1 ? TypedResults.Ok(new Todo(1, "a", false)) : TypedResults.NotFound());
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse
import logging, uuid

app = FastAPI()

def problem(status: int, title: str, detail: str | None = None, **ext):
    body = {"type": "about:blank", "title": title, "status": status}
    if detail: body["detail"] = detail
    body.update(ext)
    return JSONResponse(status_code=status, content=body,
                        media_type="application/problem+json")

class NotFoundError(Exception): ...
class ValidationError(Exception): ...

@app.exception_handler(NotFoundError)
async def not_found(req: Request, exc): return problem(404, "Not found", str(exc))

@app.exception_handler(ValidationError)
async def invalid(req: Request, exc): return problem(400, "Validation failed", str(exc))

@app.exception_handler(Exception)             # catch-all
async def unhandled(req: Request, exc):
    trace = uuid.uuid4().hex
    logging.exception("Unhandled exception trace=%s", trace)
    return problem(500, "Unexpected error", traceId=trace)   # no details leaked

# HTTPException still works for quick cases
@app.get("/orders/{id}")
def order(id: int):
    if id <= 0:
        raise HTTPException(400, "id must be positive")
    return {"id": id}

# Document responses for OpenAPI
@app.get("/x/{id}", responses={404: {"description": "Not found"}})
def x(id: int):
    if id != 1: raise NotFoundError(f"todo {id}")
    return {"id": 1}

API versioning

Once clients depend on you, breaking changes need a new version. Strategies: URL segment (/v2/orders, most visible), query string (?api-version=2), or header. The Asp.Versioning package handles all three and integrates with OpenAPI. Prefer additive, non-breaking changes so you version rarely.

// dotnet add package Asp.Versioning.Http
builder.Services.AddApiVersioning(o =>
{
    o.DefaultApiVersion = new ApiVersion(1, 0);
    o.AssumeDefaultVersionWhenUnspecified = true;
    o.ReportApiVersions = true;                               // api-supported-versions header
    o.ApiVersionReader = ApiVersionReader.Combine(
        new UrlSegmentApiVersionReader(),
        new HeaderApiVersionReader("X-Api-Version"));
});

var versioned = app.NewVersionedApi("Orders");

var v1 = versioned.MapGroup("/api/v{version:apiVersion}/orders").HasApiVersion(1, 0);
v1.MapGet("/", () => new[] { new { id = 1, total = 10m } });

var v2 = versioned.MapGroup("/api/v{version:apiVersion}/orders").HasApiVersion(2, 0);
v2.MapGet("/", () => new[] { new { id = 1, total = 10m, currency = "USD" } });   // additive change

// Deprecate
v1.HasDeprecatedApiVersion(1, 0);

HttpClient and IHttpClientFactory

Do not new HttpClient() per request (socket exhaustion) and do not keep one forever (stale DNS). IHttpClientFactory pools handlers and rotates them. Use named or typed clients, configure base address and headers once, and add resilience (retry, circuit breaker, timeout) with Microsoft.Extensions.Http.Resilience / Polly. Python's httpx.AsyncClient should likewise be created once and reused.

// Typed client
public class GitHubClient(HttpClient http)
{
    public async Task<Repo?> GetRepo(string owner, string name, CancellationToken ct = default)
        => await http.GetFromJsonAsync<Repo>($"repos/{owner}/{name}", ct);
}

builder.Services.AddHttpClient<GitHubClient>(c =>
{
    c.BaseAddress = new Uri("https://api.github.com/");
    c.DefaultRequestHeaders.UserAgent.ParseAdd("learning-path/1.0");
    c.Timeout = TimeSpan.FromSeconds(10);
})
.AddStandardResilienceHandler();          // retry + circuit breaker + timeout (Microsoft.Extensions.Http.Resilience)

// Named client
builder.Services.AddHttpClient("weather", c => c.BaseAddress = new Uri("https://api.weather.example/"));
public class WeatherService(IHttpClientFactory factory)
{
    public async Task<string> Today()
    {
        var client = factory.CreateClient("weather");
        using var resp = await client.GetAsync("today");
        resp.EnsureSuccessStatusCode();
        return await resp.Content.ReadAsStringAsync();
    }
}

// POST JSON, read JSON, handle non-success without exceptions
using var response = await http.PostAsJsonAsync("orders", new { sku = "A1", qty = 2 });
if (!response.IsSuccessStatusCode)
{
    var problem = await response.Content.ReadFromJsonAsync<ProblemDetails>();

// Delegating handler: add auth header to every call
public class AuthHandler(ITokenProvider tokens) : DelegatingHandler
{
    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage req, CancellationToken ct)
    {
        req.Headers.Authorization = new("Bearer", await tokens.GetAsync(ct));
        return await base.SendAsync(req, ct);
    }
}
builder.Services.AddTransient<AuthHandler>();
builder.Services.AddHttpClient<GitHubClient>().AddHttpMessageHandler<AuthHandler>();

CORS

Browsers block cross-origin fetch unless the server opts in with Access-Control-Allow-* headers. Configure named policies, allow only known origins in production, and remember preflight (OPTIONS) requests. CORS is a browser rule — it does nothing for server-to-server calls.

builder.Services.AddCors(o =>
{
    o.AddPolicy("spa", p => p
        .WithOrigins("https://app.example.com", "http://localhost:5173")
        .AllowAnyHeader()
        .WithMethods("GET", "POST", "PUT", "DELETE")
        .AllowCredentials()                       // cookies; cannot combine with AllowAnyOrigin
        .SetPreflightMaxAge(TimeSpan.FromHours(1)));
});

var app = builder.Build();
app.UseRouting();
app.UseCors("spa");             // after routing, before auth
app.UseAuthorization();

// Per-endpoint
app.MapGet("/public", () => "hi").RequireCors(p => p.AllowAnyOrigin());
[EnableCors("spa")] public class OrdersController : ControllerBase { }
[DisableCors] public IActionResult Internal() => Ok();

Caching and rate limiting

Three caching layers: in-process IMemoryCache, distributed IDistributedCache (Redis) and HTTP output caching / Cache-Control headers. .NET 7+ ships a rate-limiting middleware with fixed-window, sliding-window, token-bucket and concurrency limiters. Cache what is expensive and rarely changes; rate-limit what is abusable.

// In-memory cache
builder.Services.AddMemoryCache();
public class ProductService(IMemoryCache cache, AppDb db)
{
    public Task<Product?> Get(int id) => cache.GetOrCreateAsync($"product:{id}", async e =>
    {
        e.AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5);
        e.SlidingExpiration = TimeSpan.FromMinutes(1);
        return await db.Products.FindAsync(id);
    });
}

// Distributed (Redis)
builder.Services.AddStackExchangeRedisCache(o => o.Configuration = "localhost:6379");
await cache.SetStringAsync("k", json, new DistributedCacheEntryOptions { AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10) });

// HybridCache (.NET 9): L1 memory + L2 distributed with stampede protection
builder.Services.AddHybridCache();

// Output caching (server-side HTTP response cache)
builder.Services.AddOutputCache(o => o.AddPolicy("short", b => b.Expire(TimeSpan.FromSeconds(30)).SetVaryByQuery("page")));
app.UseOutputCache();
app.MapGet("/catalog", () => ...).CacheOutput("short");

// Response headers for client/CDN caching
app.MapGet("/logo", () => Results.File("logo.png")).WithMetadata(new ResponseCacheAttribute { Duration = 86400 });

// Rate limiting (.NET 7+)
builder.Services.AddRateLimiter(o =>
{
    o.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
    o.AddFixedWindowLimiter("fixed", l => { l.PermitLimit = 100; l.Window = TimeSpan.FromMinutes(1); l.QueueLimit = 0; });
    o.AddTokenBucketLimiter("burst", l => { l.TokenLimit = 20; l.TokensPerPeriod = 5; l.ReplenishmentPeriod = TimeSpan.FromSeconds(1); });
    o.GlobalLimiter = PartitionedRateLimiter.Create<HttpContext, string>(ctx =>
        RateLimitPartition.GetSlidingWindowLimiter(ctx.Connection.RemoteIpAddress?.ToString() ?? "anon",
            _ => new() { PermitLimit = 1000, Window = TimeSpan.FromMinutes(1), SegmentsPerWindow = 6 }));
});
app.UseRateLimiter();
app.MapPost("/login", ...).RequireRateLimiting("fixed");

# Hand-rolled token bucket for reference
class TokenBucket:
    def __init__(self, capacity, refill_per_sec):
        self.cap, self.rate, self.tokens, self.ts = capacity, refill_per_sec, capacity, time.monotonic()
    def allow(self) -> bool:
        now = time.monotonic(); self.tokens = min(self.cap, self.tokens + (now - self.ts) * self.rate); self.ts = now
        if self.tokens >= 1: self.tokens -= 1; return True
        return False
